##############################################################
### 朝堂权力角逐 COURT STRUGGLE — 效果层
### 议题1 继任之争 / 议题2 共治夺权
###
### 作用域约定（重要）：
###   本文件多数 effect 要求在【story 作用域】执行。
###   内部统一使用 scope:ftr_court_story（故事）与 scope:ftr_court_monarch（君主），
###   不依赖 root / prev，避免域链歧义。
###   单人向的 effect（站队、承诺等）在角色自身域执行。
##############################################################

######################################################
### 一、初始化
######################################################

# 在 story 域执行：建立初始状态
ftr_court_setup_effect = {
	save_scope_as = ftr_court_story
	story_owner = { save_scope_as = ftr_court_monarch }

	# 初始变量
	set_variable = { name = ftr_court_candidate_count  value = 0 }
	set_variable = { name = ftr_court_top_score  value = 0 }
	set_variable = { name = ftr_court_second_score  value = 0 }
	set_variable = { name = ftr_court_top_gap  value = 999 }
	# 烈度替代原 ftr_court_phase（0-3 离散阶段）：连续 0-200，由参与者加入/手段/君主状态驱动
	# 注：议题2 权力天平不再自建 swing 变量，统一走官方 diarchy swing（diarchy_swing / change_diarchy_swing）
	set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_min_value }

	# 建立参与角逐者列表（初始为空，由启动询问事件填充）。
	# 注意：必须用 clear_variable_list 初始化列表变量，不能用 remove_variable——
	# remove_variable 会删除整个列表变量，导致后续 add_to_variable_list 无法写入，参与者列表恒为空。
	clear_variable_list = ftr_court_participant_list
	set_variable = { name = ftr_court_participant_count  value = 0 }

	# 启动：询问利益相关方是否参与角逐
	ftr_court_launch_participation_asks_effect = yes
}

# 在 story 域执行：结束后清理，移除一切残留
ftr_court_cleanup_effect = {
	save_scope_as = ftr_court_story

	# 清理临时公国（兜底，防止异常路径泄漏）
	every_in_list = {
		variable = ftr_court_participant_list
		ftr_court_destroy_temp_duchy_effect = yes
	}

	# 清理候选人个体变量
	every_in_list = {
		variable = ftr_court_participant_list
		remove_variable = ftr_court_backer_of
		remove_variable = ftr_court_participant_of
		# 候选人也可能被他人支持：解除支持者背书并清空支持者名单
		ftr_court_release_all_backers_effect = yes
		remove_character_flag = ftr_court_pooled
		remove_character_flag = ftr_court_notified
	}
	# 清理参与角逐者（参与 / 拒绝标记在角色上，此处一并清除）
	every_in_list = {
		variable = ftr_court_participant_list
		remove_character_flag = ftr_court_participating
		remove_variable = ftr_court_backer_of
		remove_variable = ftr_court_participant_of
		ftr_court_release_all_backers_effect = yes
	}
	# 清理君主与朝堂成员的开始通知标记，以及最高统治者身上的个人参与者/支持者列表
	story_owner = {
		remove_character_flag = ftr_court_notified
		every_councillor = { remove_character_flag = ftr_court_notified }
		every_powerful_vassal = { remove_character_flag = ftr_court_notified }
		clear_variable_list = ftr_court_claimants
		# 君主亦是被支持权贵（GUI 读其 ftr_court_supporters）：解除其支持者背书
		ftr_court_release_all_backers_effect = yes
	}
	# 若最高统治者本人或共治者也在角逐（罕见），一并清理其支持者背书
	story_owner = {
		diarch ?= { ftr_court_release_all_backers_effect = yes }
	}

	remove_variable = ftr_court_candidate_pool
	remove_variable = ftr_court_claimant_list
	remove_variable = ftr_court_candidate_count
	remove_variable = ftr_court_top_score
	remove_variable = ftr_court_second_score
	remove_variable = ftr_court_top_gap
	remove_variable = ftr_court_struggle_intensity
	remove_variable = ftr_court_participant_list
	remove_variable = ftr_court_participant_count
}

# 在 story 域执行：故事落幕通知（须在 cleanup 前调用，因 cleanup 会清空名单）。
# 向三方各发一则落幕通知（0021）：最高领主（story_owner）、共治者（diarch，若存在）、
# 全部竞争者（ftr_court_participant_list 中的在世者）。
ftr_court_notify_struggle_end_effect = {
	save_scope_as = ftr_court_story
	# 1) 最高领主（故事主人仍在世时通知）
	story_owner ?= {
		if = {
			limit = { is_alive = yes }
			trigger_event = { id = ftr_court_struggle.0021  days = 2 }
		}
	}
	# 2) 共治者（若存在且仍在世）
	story_owner ?= {
		diarch ?= {
			if = {
				limit = { is_alive = yes }
				trigger_event = { id = ftr_court_struggle.0021  days = 2 }
			}
		}
	}
	# 3) 全部竞争者（在世者；曾参与角逐者即便已退出也在名单外——名单只留当前参与者）
	every_in_list = {
		variable = ftr_court_participant_list
		if = {
			limit = { is_alive = yes }
			trigger_event = { id = ftr_court_struggle.0021  days = 2 }
		}
	}
}

# 在 story 域执行：收场判定（供各 effect_group 的 fallback 复用）。
# 收紧后的规则：只有当故事持有者（story_owner）不再是最高领主
# （失去独立，或等级跌至王国以下）时才 end_story；
# 被囚禁 / 暂时无继承顺位 / 战争中 等临时失效不在此收场（fallback 落空即保留）。
# 注意：death 交接（有继承 → transfer；无继承 → end_story）由 on_owner_death 独立处理，
# 不经此 effect。
ftr_court_end_story_if_owner_deposed_effect = {
	# 持有者已不再是最高领主 → 结束故事（触发 on_end → 落幕通知 + cleanup）
	if = {
		limit = {
			story_owner ?= {
				NOT = { ftr_court_story_owner_is_supreme_liege_trigger = yes }
			}
		}
		end_story = yes
	}
}

######################################################
### 一之二、权力交接：故事随政权延续
######################################################

# 【通用交接】在 story 域执行：把朝堂故事移交给新君，角逐不落幕。
# 调用方需预先设置 scope:ftr_court_new_monarch（接管政权者）。
# 适用场景：君主驾崩由继承人继位 / 政变夺位 / 让位 / 被派系推翻等一切政权易主。
# 设计（对齐原版 seljuk / black_death 的 make_story_owner 范式）：
#   - 竞争者名单主存储挂在旧君主身上，须整体迁移到新君身上
#     （GUI 从 GetPlayer.GetTopLiege 读取），避免换主后窗口空白。
#   - 若新君本人此前就是角逐参与者，继承大位后不再是竞争者：
#     把他从竞争者名单（新君主存储 + story participant_list）中移除，
#     并清理其竞争者个体标记；其名下旧支持者论功行赏，
#     仅联盟/好友/挚友关系者留任为「君主支持者」，其余和平退出（无好感惩罚）。
#   - 若新君曾站队支持某权贵（旧朝参与者），登基后解除该关系。
ftr_court_transfer_story_effect = {
	# 记录旧君（当前 story owner，已失位/驾崩）与新君
	story_owner = { save_scope_as = ftr_court_old_monarch }

	if = {
		limit = { exists = scope:ftr_court_new_monarch }

		# 1) 迁移竞争者名单（旧君 → 新君）；新君本人若曾是参与者则在此剔除
		scope:ftr_court_new_monarch = {
			clear_variable_list = ftr_court_claimants
		}
		scope:ftr_court_old_monarch = {
			every_in_list = {
				variable = ftr_court_claimants
				limit = {
					is_alive = yes
					NOT = { this = scope:ftr_court_new_monarch }
				}
				save_scope_as = ftr_court_inherited_claimant
				scope:ftr_court_new_monarch = {
					add_to_variable_list = {
						name = ftr_court_claimants
						target = scope:ftr_court_inherited_claimant
					}
				}
				# 政权易主后，参与者登记的所属君主 backlink 更新为新君（供死亡清理反查）
				set_variable = { name = ftr_court_participant_of  value = scope:ftr_court_new_monarch }
			}
		}
		# 旧君名单已迁走，清理残留
		scope:ftr_court_old_monarch = { clear_variable_list = ftr_court_claimants }

		# 2) story 参与者名单同步剔除新君（若他在名单中）
		if = {
			limit = {
				is_target_in_variable_list = {
					name = ftr_court_participant_list
					target = scope:ftr_court_new_monarch
				}
			}
			remove_list_variable = {
				name = ftr_court_participant_list
				target = scope:ftr_court_new_monarch
			}
			# 2b) 竞争者成为故事所有者：对其旧支持者论功行赏，
			#     并只保留联盟/好友/挚友关系的支持者，其余和平退出（无好感惩罚）。
			# 防重入：同一角色一次登基仅结算一次（若有多个朝堂 story 逐一触发 transfer）。
			scope:ftr_court_new_monarch = {
				if = {
					limit = { NOT = { has_character_flag = ftr_court_ascension_backers_rewarded } }
					ftr_court_reward_ascended_claimant_backers_effect = yes
					add_character_flag = ftr_court_ascension_backers_rewarded
				}
			}
		}

		# 3) 新君不再是竞争者：清理参与/站队个体标记与临时公国
		#    （自身 ftr_court_supporters 保留，GUI 读作君主支持者）
		scope:ftr_court_new_monarch = {
			# 若新君曾站队支持某权贵（旧朝参与者），登基后解除该关系：
			# 从被支持者的支持者列表移除自己，再清掉自己的站队变量。
			if = {
				limit = { has_variable = ftr_court_backer_of }
				var:ftr_court_backer_of ?= {
					ftr_court_remove_backer_effect = { BACKER = prev }
				}
			}
			remove_character_flag = ftr_court_participating
			remove_character_flag = ftr_court_asked_participation
			remove_variable = ftr_court_backer_of
			remove_variable = ftr_court_participant_of
			remove_character_flag = ftr_court_pooled
			ftr_court_revoke_temp_duchy_effect = yes
		}

		# 4) 刷新参与计数（variable list 数值计数引用 count value）
		set_variable = {
			name = ftr_court_participant_count
			value = ftr_court_participant_list_count_value
		}
		set_variable = { name = ftr_court_candidate_count  value = var:ftr_court_participant_count }

		# 5) 故事所有权移交新君
		make_story_owner = scope:ftr_court_new_monarch
	}
}

# 在 story 域执行（由 story_cycle 的 on_owner_death 调用）：君主驾崩时的交接入口。
# 找出主要继承人 → 复用通用交接；无主要继承人 → 收场（end_story → on_end 统一清理）。
ftr_court_owner_death_effect = {
	# 记录旧君（当前 story owner，已驾崩）与主要继承人
	story_owner = { save_scope_as = ftr_court_old_monarch }
	story_owner.primary_heir ?= { save_scope_as = ftr_court_new_monarch }

	if = {
		limit = { exists = scope:ftr_court_new_monarch }
		ftr_court_transfer_story_effect = yes
	}
	else = {
		end_story = yes
	}
}

######################################################
### 一之三、角色死亡：清理朝堂名单残留
######################################################

# 在死者域执行（root = 死者，由 on_death 调用）。
# 名单存储：竞争者主名单 = 最高统治者身上的 ftr_court_claimants + story 的 ftr_court_participant_list
# （两者保持同步）；支持者名单 = 被支持者身上的 ftr_court_supporters。
# 反查方式：参与者身上登记 var:ftr_court_participant_of（所属君主 target）；
#           支持者身上已有 var:ftr_court_backer_of（被支持权贵 target）。
# 场景1：死者是某竞争者的支持者 → 从被支持者的支持者名单移除；
# 场景2：死者本人是竞争者 → 从所属政权名单（君主主名单 + story 名单）移除并刷新计数。
ftr_court_death_cleanup_effect = {
	# 1) 支持者除名：死者曾站队某权贵 → 从该权贵的 ftr_court_supporters 移除
	if = {
		limit = { has_variable = ftr_court_backer_of }
		var:ftr_court_backer_of ?= {
			ftr_court_remove_backer_effect = { BACKER = root }
		}
		remove_variable = ftr_court_backer_of
	}

	# 2) 竞争者除名：经 backlink 找到所属政权，同步清君主主名单与 story 名单
	if = {
		limit = { has_variable = ftr_court_participant_of }
		var:ftr_court_participant_of ?= {
			save_scope_as = ftr_court_dead_owner
			# 从最高统治者个人参与者名单移除（GUI 主存储）
			if = {
				limit = {
					is_target_in_variable_list = {
						name = ftr_court_claimants
						target = root
					}
				}
				remove_list_variable = {
					name = ftr_court_claimants
					target = root
				}
			}
			# 进入该政权的朝堂故事，清理 story 名单并刷新计数
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				if = {
					limit = {
						is_target_in_variable_list = {
							name = ftr_court_participant_list
							target = root
						}
					}
					remove_list_variable = {
						name = ftr_court_participant_list
						target = root
					}
				}
				# 刷新参与计数（variable list 计数引用 count value）
				set_variable = {
					name = ftr_court_participant_count
					value = ftr_court_participant_list_count_value
				}
				set_variable = { name = ftr_court_candidate_count  value = var:ftr_court_participant_count }
			}
		}
		# 死者作为竞争者死亡：解除其名下支持者们的背书（支持者不再"已站队"）
		ftr_court_release_all_backers_effect = yes
		# 清除死者的竞争者个体状态（含 backlink 与临时公国）
		remove_variable = ftr_court_participant_of
		remove_character_flag = ftr_court_participating
		remove_character_flag = ftr_court_pooled
		remove_character_flag = ftr_court_asked_participation
		ftr_court_revoke_temp_duchy_effect = yes
	}
}

######################################################
### 二、候选池与竞争力排名
######################################################

# 在参与角逐者域执行：加入最高统治者个人参与者列表 ftr_court_claimants（去重）。
# 依赖 scope:ftr_court_monarch（最高统治者）。
ftr_court_add_participant_effect = {
	save_scope_as = ftr_court_participant
	scope:ftr_court_monarch = {
		every_in_list = {
			variable = ftr_court_claimants
			limit = { this = scope:ftr_court_participant }
			add_character_flag = ftr_court_already_in_participant_list
		}
	}
	if = {
		limit = { NOT = { has_character_flag = ftr_court_already_in_participant_list } }
		save_scope_as = ftr_court_participant
		scope:ftr_court_monarch = {
			add_to_variable_list = {
				name = ftr_court_claimants
				target = scope:ftr_court_participant
			}
		}
	}
	remove_character_flag = ftr_court_already_in_participant_list
	# 登记所属朝堂君主（角色 target），供角色死亡 / 政权易主时反查并清理名单
	set_variable = { name = ftr_court_participant_of  value = scope:ftr_court_monarch }
}

# 在 story 域执行：参与者加入角逐时的烈度抬升。
# 烈度替代原 ftr_court_phase 离散定档。有人决定参与角逐即象征争夺成形，
# 若烈度仍处平静区则抬到酝酿区起点（至少进入对抗酝酿）。
ftr_court_on_participant_join_effect = {
	save_scope_as = ftr_court_story
	# 有人参与角逐，对抗升温：若低于酝酿起点则至少拉过 quiet_bound
	if = {
		limit = {
			has_variable = ftr_court_struggle_intensity
			var:ftr_court_struggle_intensity < ftr_court_intensity_quiet_bound_value
		}
		set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_quiet_bound_value }
	}
	else_if = {
		limit = {
			has_variable = ftr_court_struggle_intensity
			var:ftr_court_struggle_intensity >= ftr_court_intensity_quiet_bound_value
		}
		# 已入酝酿则温和累积张力
		change_variable = {
			name = ftr_court_struggle_intensity
			add = ftr_court_intensity_tension_bias_value
		}
	}
	# 夹紧到 0-200 区间，防止越界
	if = {
		limit = { var:ftr_court_struggle_intensity > ftr_court_intensity_max_value }
		set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_max_value }
	}
}

# 在 story 域执行：烈度年度维护（自然回落 + 君主状态偏置 + 竞争者张力）。
# 替代原阶段定档逻辑，是烈度随时间自然演化的主循环（承接 §12.3.2）。
# 无参与者（candidate_count < min）时向 0 回落，避免长期无人对抗仍高温。
ftr_court_refresh_intensity_effect = {
	save_scope_as = ftr_court_story
	story_owner = { save_scope_as = ftr_court_monarch }

	if = {
		limit = { scope:ftr_court_monarch = { ftr_court_has_succession_order_trigger = yes } }
		if = {
			limit = { ftr_court_has_active_struggle_trigger = yes }
			# 有参与者在角逐：向低区缓慢自然回落（防单向锁死不回落）
			if = {
				limit = { var:ftr_court_struggle_intensity > ftr_court_intensity_quiet_bound_value }
				change_variable = {
					name = ftr_court_struggle_intensity
					subtract = ftr_court_intensity_yearly_decay_value
				}
			}
			# 君主失能/弱势：逐年推高烈度（承接 ftr_court_monarch_weak_trigger 语义）
			if = {
				limit = { scope:ftr_court_monarch = { ftr_court_monarch_infirm_trigger = yes } }
				change_variable = {
					name = ftr_court_struggle_intensity
					add = ftr_court_intensity_weak_bias_value
				}
			}
			else_if = {
				limit = {
					scope:ftr_court_monarch = {
						prestige >= ftr_court_monarch_weak_prestige_value
						piety >= ftr_court_monarch_weak_piety_value
					}
				}
				# 君主强势：逐年压低烈度
				change_variable = {
					name = ftr_court_struggle_intensity
					add = ftr_court_intensity_strong_bias_value
				}
			}
		}
		else = {
			# 无参与者：向 0 回落到平静，避免残温空转
			if = {
				limit = { var:ftr_court_struggle_intensity > ftr_court_intensity_quiet_bound_value }
				change_variable = {
					name = ftr_court_struggle_intensity
					subtract = ftr_court_intensity_yearly_decay_value
				}
			}
		}
	}

	# 夹紧到 0-200 区间
	if = {
		limit = { var:ftr_court_struggle_intensity < ftr_court_intensity_min_value }
		set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_min_value }
	}
	else_if = {
		limit = { var:ftr_court_struggle_intensity > ftr_court_intensity_max_value }
		set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_max_value }
	}
}

# 在 story 域执行：启动/定期询问潜在继任者是否参与角逐。
# 血缘继承：询问子女 / 兄弟姐妹（按性别继承法过滤）；选举继承：主头衔继承顺位。
# 已明确表示不参与者（ftr_court_declined_contest）不询问。
# 参与人数达到上限（5）后不再询问。

# 在被支持者（权贵）域执行：从自己的 ftr_court_supporters 支持者列表移除指定角色。
# 参数：BACKER = 要移除的支持者（角色 scope）。
# 注意：ftr_court_supporters 是 add_to_variable_list 维护的 variable list（存在被支持者身上），
# 移除成员用 remove_list_variable（对齐原版 tests/event_target_lists_tests.txt），
# 不能用 local list 的 remove_from_list。先 has_variable_list 保护避免对空列表操作。
ftr_court_remove_backer_effect = {
	if = {
		limit = { has_variable_list = ftr_court_supporters }
		remove_list_variable = {
			name = ftr_court_supporters
			target = $BACKER$
		}
	}
}

# 在支持者域执行：公开支持某权贵的完整记账（供 0014 五个选项与主动决议共用，替代原交互 on_accept）。
# 参数：$CLAIMANT$ = 被支持者（角色 scope 表达式，如 scope:ftr_court_claimant_1）。
# 依赖：scope:ftr_court_story 已保存（0014 immediate 已保存；主动决议同样经 0014 触发）。
# 含：改投清理（曾支持他人则先从旧主账目移除并令旧主/其支持者记恨）→ 新主记账（backer_of /
# supporters / 双向好感）→ 公开树敌（其余参与者）→ 支持君主子女/继承人的背书声望 →
# 君主警觉 0041（概率随双方谋略浮动）。
ftr_court_apply_backing_effect = {
	# 支持者即当前 root，登记为 ftr_court_backer 供警觉事件使用
	save_scope_as = ftr_court_backer
	# ---- 1) 改投清理：此前已支持他人且非本目标 ----
	if = {
		limit = {
			has_variable = ftr_court_backer_of
			NOT = { var:ftr_court_backer_of ?= { this = $CLAIMANT$ } }
		}
		save_scope_as = ftr_court_old_backed
		# 旧主从自己的支持者名单移除本支持者
		var:ftr_court_backer_of ?= {
			ftr_court_remove_backer_effect = { BACKER = root }
		}
		# 旧主记恨
		scope:ftr_court_old_backed = {
			add_opinion = { target = root  modifier = ftr_court_betrayed_opinion }
			# 旧主的支持者们（有地与廷臣，除自己外）同仇敌忾
			every_vassal = {
				limit = {
					has_variable = ftr_court_backer_of
					var:ftr_court_backer_of ?= { this = scope:ftr_court_old_backed }
					NOT = { this = root }
				}
				add_opinion = { target = root  modifier = ftr_court_rival_claimant_opinion }
			}
			every_courtier = {
				limit = {
					has_variable = ftr_court_backer_of
					var:ftr_court_backer_of ?= { this = scope:ftr_court_old_backed }
					NOT = { this = root }
				}
				add_opinion = { target = root  modifier = ftr_court_rival_claimant_opinion }
			}
		}
	}
	# ---- 2) 支持者侧记账：更新站队对象 + 对受支持者表达好感 ----
	set_variable = { name = ftr_court_backer_of  value = $CLAIMANT$ }
	add_opinion = { target = $CLAIMANT$  modifier = ftr_court_backer_opinion }
	# ---- 3) 被支持者侧记账：登记支持者（防重）、回赠政治盟友好感、另存 debug 与被警觉用作用域 ----
	$CLAIMANT$ = {
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = ftr_court_supporters
						target = root
					}
				}
			}
			add_to_variable_list = {
				name = ftr_court_supporters
				target = root
			}
		}
		add_opinion = { target = root  modifier = ftr_political_alliance_opinion }
		save_scope_as = ftr_court_debug_backed
		save_scope_as = ftr_court_supported
	}
	# ---- 4) 公开树敌：其余参与者同生芥蒂（对齐 0014 / 交互公开分支）----
	scope:ftr_court_story ?= {
		every_in_list = {
			variable = ftr_court_participant_list
			limit = {
				NOT = { this = $CLAIMANT$ }
				NOT = { this = root }
			}
			add_opinion = { target = root  modifier = ftr_court_rival_claimant_opinion }
		}
	}
	# ---- 5) 背书声望：支持对象是君主的子女 / 继承人 ----
	$CLAIMANT$ = {
		if = {
			limit = {
				exists = scope:ftr_court_monarch
				OR = {
					is_child_of = scope:ftr_court_monarch
					is_heir_of = scope:ftr_court_monarch
				}
			}
			add_prestige = ftr_court_heir_backing_prestige_value
		}
	}
	# ---- 6) 君主警觉：支持君主本人不构成"惊君"；其余对象按谋略对抗概率触发 0041 ----
	if = {
		limit = {
			exists = scope:ftr_court_monarch
			NOT = { $CLAIMANT$ = scope:ftr_court_monarch }
			$CLAIMANT$ = { NOT = { ftr_court_is_struggle_monarch_trigger = yes } }
		}
		random = {
			chance = ftr_court_backing_alert_chance_value
			custom_tooltip = {
				text = ftr_court_apply_backing_effect.liege_notify_text
				scope:ftr_court_monarch ?= {
					save_scope_as = ftr_court_monarch
					trigger_event = { id = ftr_court_struggle.0041  days = 1 }
				}
			}
		}
	}
	# ---- 7) DEBUG 通知「谁支持了谁」（后续统一移除）----
	ftr_court_debug_backing_notice_effect = yes
}

# 在被支持者（权贵）域执行：解除自己名下所有支持者的背书关系（供本权贵被移出角逐时调用）。
# 背书是双向记账：被支持者身上的 ftr_court_supporters 名单 + 支持者本人的 ftr_court_backer_of。
# 若本权贵（被支持对象）死亡 / 被捕清算 / 退出角逐，其支持者不应继续"已站队"——
# 清空名单的同时须逐个抹去支持者身上的 ftr_court_backer_of，避免悬空背书占用其站队资格。
# 注意：新君登基（ftr_court_transfer_story_effect）复用本 effect 逻辑前须确认——
# 新君旧日的支持者会保留并读作「君主支持者」，故 transfer 场景不调用本 effect。
ftr_court_release_all_backers_effect = {
	if = {
		limit = { has_variable_list = ftr_court_supporters }
		every_in_list = {
			variable = ftr_court_supporters
			# 清空所有支持者的背书记忆
			remove_variable = ftr_court_backer_of
		}
		clear_variable_list = ftr_court_supporters
	}
}

# 在新君（原角逐参与者）域执行：竞争者成为故事所有者后的论功行赏。
# 适用场景：ftr_court_transfer_story_effect 把新君从参与者名单移除（新君曾是竞争者）时调用。
# 逻辑：
#   1) 逐个遍历其名下 ftr_court_supporters 旧支持者，在世者发赏：
#      威望 / 影响力（行政制支持者按官方影响力档、其余威望）+ 金钱 + 税收加成 modifier。
#   2) 只保留与本人有联盟 / 好友 / 挚友关系的支持者；其余和平退出
#      （解除背书、移出名单），不施加任何好感惩罚。
# 依赖：进入时 this = 新君，其名下有 ftr_court_supporters（variable list）。
ftr_court_reward_ascended_claimant_backers_effect = {
	save_scope_as = ftr_court_ascended_monarch
	# 初始化临时「留任者」名单（防旧残留）
	clear_variable_list = ftr_court_keep_backers
	if = {
		limit = { has_variable_list = ftr_court_supporters }
		# ---- 1) 论功：逐名在世支持者发赏 + 甄别留任者 ----
		every_in_list = {
			variable = ftr_court_supporters
			limit = { is_alive = yes }
			# 威望 / 影响力：行政制支持者获官方影响力，其余获威望
			if = {
				limit = { government_allows = administrative }
				change_influence = medium_influence_gain
			}
			else = {
				add_prestige = ftr_court_ascension_backer_prestige_value
			}
			# 金钱
			add_gold = ftr_court_ascension_backer_gold_value
			# 税收加成（持续若干年）
			add_character_modifier = {
				modifier = ftr_court_ascension_backer_modifier
				years = ftr_court_ascension_backer_tax_years_value
			}
			# 亲密关系留任：联盟 / 好友 / 挚友
			if = {
				limit = {
					OR = {
						is_allied_to = scope:ftr_court_ascended_monarch
						has_relation_friend = scope:ftr_court_ascended_monarch
						has_relation_best_friend = scope:ftr_court_ascended_monarch
					}
				}
				add_character_flag = ftr_court_keep_backer
				save_scope_as = ftr_court_kept_backer
				scope:ftr_court_ascended_monarch = {
					add_to_variable_list = {
						name = ftr_court_keep_backers
						target = scope:ftr_court_kept_backer
					}
				}
			}
		}
		# ---- 2) 清场：其余支持者和平退出（解除背书，无好感惩罚）----
		every_in_list = {
			variable = ftr_court_supporters
			limit = {
				is_alive = yes
				NOT = { has_character_flag = ftr_court_keep_backer }
			}
			remove_variable = ftr_court_backer_of
		}
		# ---- 3) 名单重建为留任者（亲密者继续作为君主支持者）----
		clear_variable_list = ftr_court_supporters
		if = {
			limit = { has_variable_list = ftr_court_keep_backers }
			every_in_list = {
				variable = ftr_court_keep_backers
				# 双记账一致：恢复留任者「我支持新君」的背书，再挂回新君支持者名单
				set_variable = { name = ftr_court_backer_of  value = scope:ftr_court_ascended_monarch }
				save_scope_as = ftr_court_kept_backer
				scope:ftr_court_ascended_monarch = {
					add_to_variable_list = {
						name = ftr_court_supporters
						target = scope:ftr_court_kept_backer
					}
				}
				remove_character_flag = ftr_court_keep_backer
			}
		}
	}
	# 清理临时「留任者」名单（无论本权贵此前是否有支持者，均不残留）
	remove_variable = ftr_court_keep_backers
}

# 在被支持者（权贵）域执行：其支持者受压制而离心。
# 替代旧的"直接扣 ftr_claim_strength"做法（实时评分已无存储变量可扣）——
# 随机一名支持者退出支持，其余支持者对该权贵好感下降。
# 依赖：进入时 this = 被支持者（权贵），其身上有 ftr_court_supporters 名单。
ftr_court_discourage_backers_effect = {
	save_scope_as = ftr_court_patron_holder
	if = {
		limit = { has_variable_list = ftr_court_supporters }
		# 1) 随机一名支持者动摇退出支持（退出者解除背书、对原主交恶）
		random_in_list = {
			variable = ftr_court_supporters
			save_scope_as = ftr_court_defecting_backer
			scope:ftr_court_defecting_backer = {
				remove_variable = ftr_court_backer_of
				add_opinion = { target = scope:ftr_court_patron_holder  modifier = ftr_court_backer_discouraged_opinion }
			}
		}
		# 2) 把退出者移出被支持者的支持者名单
		scope:ftr_court_patron_holder = {
			if = {
				limit = { exists = scope:ftr_court_defecting_backer }
				ftr_court_remove_backer_effect = { BACKER = scope:ftr_court_defecting_backer }
			}
		}
		# 3) 其余支持者离心（对权贵好感下降）
		scope:ftr_court_patron_holder = {
			every_in_list = {
				variable = ftr_court_supporters
				limit = { NOT = { this = scope:ftr_court_defecting_backer } }
				add_opinion = { target = scope:ftr_court_patron_holder  modifier = ftr_court_backer_discouraged_opinion }
			}
		}
	}
}

# 在无地竞争者域执行：创建纯虚拟的临时公国。
# 依赖：scope:ftr_court_monarch（在位君主）。
# 该头衔在战争结束时必定销毁，不论胜负。
ftr_court_create_temp_duchy_effect = {
	save_scope_as = ftr_court_claimant

	# 在王国境内挑一个伯爵领作为临时公国首都
	scope:ftr_court_monarch = {
		random_sub_realm_county = {
			save_scope_as = ftr_court_temp_capital
		}
	}

	if = {
		limit = { exists = scope:ftr_court_temp_capital }

		create_dynamic_title = {
			tier = duchy
			name = FTR_COURT_TEMP_DUCHY_NAME
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = ftr_court_title_change
			add_claim_on_loss = no
		}
		scope:new_title = {
			set_capital_county = scope:ftr_court_temp_capital
			# 纯虚拟头衔：不产生宣称、不传承、销毁即彻底删除
			set_landless_title = yes
			set_destroy_on_succession = yes
			set_delete_on_destroy = yes
			set_no_automatic_claims = yes
			set_definitive_form = yes
			set_can_be_named_after_dynasty = no
			change_title_holder = {
				holder = scope:ftr_court_claimant
				change = scope:ftr_court_title_change
			}
		}
		resolve_title_and_vassal_change = scope:ftr_court_title_change

		# 记录以便战后回收
		set_variable = {
			name = ftr_court_temp_duchy
			value = scope:new_title
		}
	}
}

# 在角色域执行：销毁临时公国（纯虚拟头衔，战后一律移除）
ftr_court_destroy_temp_duchy_effect = {

}

# 在起兵者域执行：起兵前准备。
# 关键点：无地皇子/王子无法宣战，必须先创建临时公国赋予其宣战资格，
# 然后立即发动继位战争。有地者直接宣战。
# 参数：TARGET = 宣战对象（现任君主）。
ftr_court_prepare_rebellion_effect = {
	save_scope_as = ftr_court_warleader
	# 供 create_temp_duchy_effect 选定临时首都（宣战对象即现任君主）
	$TARGET$ ?= { save_scope_as = ftr_court_monarch }

	# 无地者先备虚拟头衔（已备则跳过）

}

# 内斗出手（§12.4 手段 2.1 诬陷下毒链，首个可用手段族；其余手段族随后接入）：
# story 域。参与者 >=2 且烈度进入对立(>=90)后，随机一名竞争者向另一名竞争者出手。
# 手手段随烈度升级：对立区(90-140)→0050 放谣；濒危区(>=140)→0051 伪造 / 0052 下毒。
ftr_court_strife_slander_effect = {
	save_scope_as = ftr_court_strife_story
	set_variable = { name = ftr_court_strife_pick  value = 0 }
	random_in_list = {
		variable = ftr_court_participant_list
		limit = {
			NOT = { has_character_flag = ftr_court_strife_active }
			NOT = { this = scope:ftr_court_strife_story.story_owner }
		}
		save_scope_as = ftr_court_strife_actor
		if = {
			limit = { var:ftr_court_strife_pick = 0 }
			# 目标：另一名竞争者（参与者 >=2 时才成立）
			set_variable = { name = ftr_court_strife_pick  value = 1 }
			random_in_list = {
				variable = ftr_court_participant_list
				limit = {
					NOT = { this = scope:ftr_court_strife_actor }
					NOT = { this = scope:ftr_court_strife_story.story_owner }
				}
				save_scope_as = ftr_court_strife_victim
			}
			# 把 victim 存到 actor 身上，供手段事件读取
			scope:ftr_court_strife_actor ?= {
				set_variable = { name = ftr_court_strife_victim  value = scope:ftr_court_strife_victim }
				# 手段随烈度升级（诬陷链 / 牵制 / 绑架 / 谋杀），按概率触发，保留"不触发"
				if = {
					limit = {
						scope:ftr_court_strife_story.var:ftr_court_struggle_intensity >= ftr_court_intensity_critical_bound_value
					}
					random = { chance = 35  trigger_event = { id = ftr_court_struggle.0052 } }
					random = { chance = 35  trigger_event = { id = ftr_court_struggle.0090 } }
					random = { chance = 25  trigger_event = { id = ftr_court_struggle.0080 } }
				}
				else_if = {
					limit = {
						scope:ftr_court_strife_story.var:ftr_court_struggle_intensity >= ftr_court_intensity_high_bound_value
					}
					random = { chance = 30  trigger_event = { id = ftr_court_struggle.0051 } }
					random = { chance = 30  trigger_event = { id = ftr_court_struggle.0090 } }
					random = { chance = 30  trigger_event = { id = ftr_court_struggle.0080 } }
				}
				else = {
					random = { chance = 40  trigger_event = { id = ftr_court_struggle.0050 } }
					random = { chance = 35  trigger_event = { id = ftr_court_struggle.0070 } }
				}
			}
		}
	}
	remove_variable = ftr_court_strife_pick
}

# 内斗强出口轮（§4.3 主时钟强出口分支）：
# story 域，由内斗主时钟在互殴手段派发后调用。按 ftr_court_strife_exit_chance_value 决定本轮是否推进。
# 烈度档：90-139（对立）→ 0003 通知 + 0014 站队询问；
#         ≥140（濒危） → 0005 通知 + 0014 + 0015（评分最高者）；
#         ≥170（临界） 且满足废黜条件 → 0018（朝堂最强势力）。各出口沿用既有 3 年防骚扰 flag。
ftr_court_strife_exit_pulse_effect = {
	save_scope_as = ftr_court_strife_exit_story
	# 保存君主命名域，供下方 order_by = ftr_court_claim_strength_value 实时评估使用
	story_owner = { save_scope_as = ftr_court_monarch }
	random = {
		chance = ftr_court_strife_exit_chance_value
		# 通知事件按烈度区分：濒危以上 0005（一触即发）、对立区 0003（势均力敌）
		if = {
			limit = { var:ftr_court_struggle_intensity >= ftr_court_intensity_high_bound_value }
			story_owner = { trigger_event = { id = ftr_court_struggle.0005 } }
		}
		else_if = {
			limit = { var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value }
			story_owner = { trigger_event = { id = ftr_court_struggle.0003 } }
		}
		# 0014 站队：询问一名未站队的封臣（random_courtier 路径覆盖无地廷臣）
		story_owner = {
			random_vassal = {
				limit = {
					is_capable_adult = yes
					NOT = { has_variable = ftr_court_backer_of }
					NOT = { has_character_flag = ftr_court_asked_stance }
				}
				add_character_flag = { flag = ftr_court_asked_stance  years = 3 }
				trigger_event = { id = ftr_court_struggle.0014  days = 2 }
			}
			random_courtier = {
				limit = {
					is_capable_adult = yes
					NOT = { has_variable = ftr_court_backer_of }
					NOT = { has_character_flag = ftr_court_asked_stance }
				}
				add_character_flag = { flag = ftr_court_asked_stance  years = 3 }
				trigger_event = { id = ftr_court_struggle.0014  days = 2 }
			}
		}
		# 0015 良机已至：仅濒危（≥140），推给竞争力最高且未询问过者
		if = {
			limit = { var:ftr_court_struggle_intensity >= ftr_court_intensity_high_bound_value }
			ordered_in_list = {
				variable = ftr_court_participant_list
				order_by = ftr_court_claim_strength_value
				save_scope_as = ftr_court_exit_ringleader
				scope:ftr_court_exit_ringleader ?= {
					if = {
						limit = { NOT = { has_character_flag = ftr_court_asked_to_rebel } }
						trigger_event = { id = ftr_court_struggle.0015  days = 4 }
					}
				}
			}
		}
		# 0018 废黜昏君：临界（≥170）且满足废黜条件 → 推给朝堂最强势力
		if = {
			limit = {
				var:ftr_court_struggle_intensity >= ftr_court_intensity_critical_bound_value
				ftr_court_can_attempt_usurp_trigger = yes
			}
			story_owner = {
				# usurp_candidates 为作用域内临时 local list（add_to_list → 遍历用 list =），
				# 不需要（也不应）用 variable list 的 clear_variable_list 清理。
				every_vassal = {
					limit = {
						is_capable_adult = yes
						NOT = { has_character_flag = ftr_court_asked_usurp }
					}
					add_to_list = ftr_court_usurp_candidates
				}
				every_courtier = {
					limit = {
						is_capable_adult = yes
						NOT = { has_character_flag = ftr_court_asked_usurp }
					}
					add_to_list = ftr_court_usurp_candidates
				}
				set_variable = { name = ftr_court_usurp_pick  value = 0 }
				ordered_in_list = {
					list = ftr_court_usurp_candidates
					order_by = max_military_strength
					save_scope_as = ftr_court_exit_usurper
					root = {
						if = {
							limit = { var:ftr_court_usurp_pick = 0 }
							scope:ftr_court_exit_usurper ?= {
								add_character_flag = { flag = ftr_court_asked_usurp  years = 3 }
								trigger_event = { id = ftr_court_struggle.0018  days = 2 }
							}
						}
						change_variable = { name = ftr_court_usurp_pick  add = 1 }
					}
				}
				remove_variable = ftr_court_usurp_pick
			}
		}
	}
}

# 议题 2 阵营拉锯：无地参与者的「暗中策动」等价链入口（§9.6）
# story 域。当权力天平偏向共治者（≥70）且存在共治者时，随机一名无地参与者暗中联络朝臣削弱共治（00A1）。
ftr_court_landless_swing_effect = {
	save_scope_as = ftr_court_strife_story
	random = {
		chance = 25
		if = {
			limit = {
				# 官方 swing：天平偏共治 ≥70 时才暗中策动削弱
				story_owner ?= {
					exists = diarch
					diarchy_swing >= ftr_court_swing_regency_threshold_value
				}
			}
			random_in_list = {
				variable = ftr_court_participant_list
				limit = {
					is_landed = no
					NOT = { has_character_flag = ftr_court_strife_active }
				}
				save_scope_as = ftr_court_landless_actor
				scope:ftr_court_landless_actor ?= {
					add_character_flag = { flag = ftr_court_strife_active  years = 1 }
					trigger_event = { id = ftr_court_struggle.00A1  days = 2 }
				}
			}
		}
	}
}

# 议题 2：共治者反击（§9.6）——swing ≤30（辅政稳固）时，共治者阵营暗中游说拉回（00A2）。
ftr_court_diarch_backlash_effect = {

		}
	}
