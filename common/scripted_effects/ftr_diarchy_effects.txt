### 〇、天平偏移通知（§7.1：任何 swing 变动都要能追溯到原因）
######################################################

# 求值域：共治关系中的**君主（领主）**侧。
# 参数：$KEY$ = 原因本地化键（同一个键同时用作消息的 custom_tooltip，键内写明方向与点数等细节）
# 作用：向**君主**与**共治者**各发一条「天平偏移」消息；send_interface_message 仅在接收者为当前玩家时
#       显示，故 AI 政权之间不会产生可见噪音（双方都被通知，各自以自己为视角）。
# 用法：紧跟每一处 swing 写入（update_diarchy_swing_with_perspective_effect / change_diarchy_swing /
#       set_diarchy_swing）之后调用，把"天平为什么动"直接告诉当事双方。
ftr_diarchy_swing_notice_effect = {
	if = {
		limit = {
			has_active_diarchy = yes
			exists = diarch
		}
		save_scope_as = ftr_diarchy_notice_liege
		diarch ?= { save_scope_as = ftr_diarchy_notice_diarch }
		# 领主侧回执（左图标 = 共治者）
		send_interface_message = {
			type = event_generic_neutral
			title = ftr_diarchy_swing_notice_title
			desc = ftr_diarchy_swing_notice_desc
			left_icon = scope:ftr_diarchy_notice_diarch
			right_icon = scope:ftr_diarchy_notice_liege
			custom_tooltip = $KEY$
		}
		# 共治者侧回执（同一原因，视角相反）
		scope:ftr_diarchy_notice_diarch ?= {
			send_interface_message = {
				type = event_generic_neutral
				title = ftr_diarchy_swing_notice_title
				desc = ftr_diarchy_swing_notice_desc
				left_icon = scope:ftr_diarchy_notice_liege
				right_icon = scope:ftr_diarchy_notice_diarch
				custom_tooltip = $KEY$
			}
		}
	}
}

### 一、立储接线（§2.2）
######################################################

# 求值域：主君。参数 PRINCE = 被指定的继承人。
ftr_diarchy_try_start_crown_prince_effect = {
	save_scope_as = ftr_diarchy_liege
	if = {
		limit = { scope:ftr_diarchy_liege = { ftr_diarchy_can_start_crown_prince_trigger = yes } }
		$PRINCE$ ?= { save_scope_as = ftr_diarchy_prince_candidate }
		if = {
			limit = {
				exists = scope:ftr_diarchy_prince_candidate
				scope:ftr_diarchy_prince_candidate = {
					is_alive = yes
					is_adult = yes
					is_imprisoned = no
				}
			}
			scope:ftr_diarchy_liege = {
				try_start_diarchy = ftr_diarchy_crown_prince
				set_diarch = scope:ftr_diarchy_prince_candidate
				# 主动立储：低起点（区别于被逼立储的 40）
				set_diarchy_swing = ftr_diarchy_crown_prince_voluntary_initial_swing_value
				# 天平偏移通知（§7.1）：说明初值来由（主动立储基线）
				ftr_diarchy_swing_notice_effect = { KEY = ftr_diarchy_swing_notice_initial_voluntary }
			}
		}
	}
}

# 求值域：君主（由 01 故事循环主时钟每拍调用，§2.2 补立）。
# 口径：未成年指定继承人**不即建**共治（立储接线的 is_adult 守卫保留）；
#       待其成年后由本效果补建太子共治，并触发 0700 事件告知君主。
# 注：已有任何共治（含摄政）时 `ftr_diarchy_can_start_crown_prince_trigger` 不成立 →
#     补立不发生（储位冻结口径，须先结束共治）。
ftr_diarchy_promote_adult_heir_effect = {
	if = {
		limit = {
			ftr_diarchy_can_start_crown_prince_trigger = yes
			exists = designated_heir
			designated_heir ?= {
				is_alive = yes
				is_adult = yes
				is_imprisoned = no
				is_diarch = no
			}
		}
		ftr_diarchy_try_start_crown_prince_effect = { PRINCE = designated_heir }
		if = {
			limit = {
				has_active_diarchy = yes
				diarch ?= { this = designated_heir }
			}
			trigger_event = { id = ftr_diarchy.0700  days = 2 }
		}
	}
}

### 二、无地共治者授土（§4.0）
######################################################

# 求值域：主君。参数 RECIPIENT = 受地者（共治者）。
# 优先划出「非首都伯爵领」；无则转移一个「公爵头衔」，
# 该公国下属伯爵领与封臣随 type = granted 一并转移（D1b）。
ftr_diarchy_grant_land_effect = {
	save_scope_as = ftr_diarchy_grant_liege
	$RECIPIENT$ ?= { save_scope_as = ftr_diarchy_grant_target }

	if = {
		limit = { exists = scope:ftr_diarchy_grant_target }

		# 1) 非首都伯爵领
		random_held_title = {
			limit = {
				tier = tier_county
				NOT = { this = scope:ftr_diarchy_grant_liege.capital_county }
			}
			save_scope_as = ftr_diarchy_grant_title
		}

		# 2) 无直辖伯爵领则取公爵头衔
		if = {
			limit = { NOT = { exists = scope:ftr_diarchy_grant_title } }
			random_held_title = {
				limit = { tier = tier_duchy }
				save_scope_as = ftr_diarchy_grant_title
			}
		}

		# 3) 转移
		if = {
			limit = { exists = scope:ftr_diarchy_grant_title }
			scope:ftr_diarchy_grant_liege = {
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = ftr_diarchy_grant_change
					add_claim_on_loss = no
				}
			}
			scope:ftr_diarchy_grant_title = {
				change_title_holder = {
					holder = scope:ftr_diarchy_grant_target
					change = scope:ftr_diarchy_grant_change
				}
			}
			scope:ftr_diarchy_grant_liege = {
				resolve_title_and_vassal_change = scope:ftr_diarchy_grant_change
			}
			scope:ftr_diarchy_grant_target = {
				add_character_flag = ftr_diarchy_granted_land
			}
		}
	}
}

### 三、废黜 / 篡位 —— 共用结局（§4.1 与 §4.4 只差「谁上位」）
######################################################

# 求值域：共治者本人。参数 NEW_HOLDER = 继位者。
# 结局：旧主失去所有头衔 → 由 NEW_HOLDER 接手 → 旧主被共治者囚禁 → 事件发落。
ftr_diarchy_oust_liege_impl_effect = {
	save_scope_as = ftr_diarchy_ouser
	liege = { save_scope_as = ftr_diarchy_old_liege }

	$NEW_HOLDER$ ?= { save_scope_as = ftr_diarchy_new_holder }

	# 未指定 / 已失效的继位者 → 回退到指定继承人，再回退到原版继承法继承人
	if = {
		limit = { NOT = { exists = scope:ftr_diarchy_new_holder } }
		scope:ftr_diarchy_old_liege = {
			designated_heir ?= { save_scope_as = ftr_diarchy_new_holder }
		}
	}
	if = {
		limit = { NOT = { exists = scope:ftr_diarchy_new_holder } }
		scope:ftr_diarchy_old_liege = {
			primary_heir ?= { save_scope_as = ftr_diarchy_new_holder }
		}
	}

	# 旧主失去所有头衔
	if = {
		limit = { exists = scope:ftr_diarchy_new_holder }
		scope:ftr_diarchy_old_liege = {
			create_title_and_vassal_change = {
				type = usurped
				save_scope_as = ftr_diarchy_oust_change
				add_claim_on_loss = no
			}
			every_held_title = {
				change_title_holder = {
					holder = scope:ftr_diarchy_new_holder
					change = scope:ftr_diarchy_oust_change
				}
			}
			resolve_title_and_vassal_change = scope:ftr_diarchy_oust_change
			add_opinion = {
				target = scope:ftr_diarchy_ouser
				modifier = ftr_diarchy_ousted_liege_opinion
			}
		}
	}

	# 囚禁旧主，随后交共治者发落
	scope:ftr_diarchy_ouser = {
		if = {
			limit = {
				exists = scope:ftr_diarchy_old_liege
				scope:ftr_diarchy_old_liege = { is_imprisoned = no }
			}
			imprison = {
				target = scope:ftr_diarchy_old_liege
				type = house_arrest
			}
		}
		# 事件间传参：把旧主存成变量（scope 不跨事件）
		if = {
			limit = { exists = scope:ftr_diarchy_old_liege }
			set_variable = {
				name = ftr_diarchy_deposed_liege
				value = scope:ftr_diarchy_old_liege
				days = 30
			}
		}
		trigger_event = { id = ftr_diarchy.0100  days = 1 }
	}
}

# 求值域：共治者本人。§4.1 废黜：由「指定继承人」接手（共治者隐于幕后）
ftr_diarchy_oust_liege_to_heir_effect = {
	save_scope_as = ftr_diarchy_ouser
	liege = { save_scope_as = ftr_diarchy_old_liege }
	scope:ftr_diarchy_old_liege = {
		designated_heir ?= { save_scope_as = ftr_diarchy_new_holder }
	}
	ftr_diarchy_oust_liege_impl_effect = { NEW_HOLDER = scope:ftr_diarchy_new_holder }
	# 从龙功：复用 02 既有记账（求值域 = 效劳方）
	ftr_personal_merit_crown_add_effect = { AMOUNT = ftr_qingjunce_merit_value }
}

# 求值域：共治者本人。§4.4 篡位：共治者自立
ftr_diarchy_oust_liege_to_self_effect = {
	save_scope_as = ftr_diarchy_ouser
	save_scope_as = ftr_diarchy_new_holder
	liege = { save_scope_as = ftr_diarchy_old_liege }
	ftr_diarchy_oust_liege_impl_effect = { NEW_HOLDER = scope:ftr_diarchy_ouser }
}

### 四、废太子（§3.2）
######################################################

# 求值域：主君。参数 TARGET = 被废的太子。
ftr_diarchy_abolish_crown_prince_effect = {
	save_scope_as = ftr_diarchy_liege
	$TARGET$ ?= { save_scope_as = ftr_diarchy_deposed_prince }

	if = {
		limit = { exists = scope:ftr_diarchy_deposed_prince }

		# 1) 原版废除继承权机制（加 disinherited trait，与 §2.2 候选过滤闭环）
		scope:ftr_diarchy_deposed_prince = {
			disinherit_effect = { DISINHERITOR = scope:ftr_diarchy_liege }
			# 双向固定值好感惩罚：被废者 → 主君
			add_opinion = {
				target = scope:ftr_diarchy_liege
				modifier = passed_me_over_for_regent_opinion
				opinion = ftr_diarchy_abolish_opinion_mutual_value
			}
			# 复位入口标记：coup 计谋据此放行「前共治者」
			add_character_flag = ftr_diarchy_deposed_prince
		}

		scope:ftr_diarchy_liege = {
			# 双向固定值好感惩罚：主君 → 被废者
			add_opinion = {
				target = scope:ftr_diarchy_deposed_prince
				modifier = designated_bad_regent_opinion
				opinion = ftr_diarchy_abolish_opinion_mutual_value
			}
			# 太子党羽不满：曾为太子背书者（ftr_court_supporters 名单，01 双记账）记恨废储之举
			# 注：designated_bad_regent_opinion 无静态数值，必须显式给 opinion。
			scope:ftr_diarchy_deposed_prince = {
				if = {
					limit = { has_variable_list = ftr_court_supporters }
					every_in_list = {
						variable = ftr_court_supporters
						limit = { NOT = { this = scope:ftr_diarchy_liege } }
						add_opinion = {
							target = scope:ftr_diarchy_liege
							modifier = designated_bad_regent_opinion
							opinion = ftr_diarchy_abolish_backer_opinion_value
						}
					}
				}
			}
			# 2) 代价：废储动摇国本，损失正统性
			if = {
				limit = { has_legitimacy = yes }
				add_legitimacy = ftr_diarchy_abolish_legitimacy_value
			}
		}
		scope:ftr_diarchy_deposed_prince = {
			# 3) 结束共治
			if = {
				limit = { is_diarch  = yes }
				depose_diarch = yes
			}
			# 4) 清除储位：引擎不支持 set_designated_heir = none，须用原版 clear_designated_heir
			clear_designated_heir = yes
		}
	}
}

### 五、大赦天下（§4.6）
######################################################

# 求值域：共治者本人。
# 行为：① 固定值削减自身不和；
#      ② 借原版赦免机制，把**主君全部直属封臣的罪行**一律「变成已处罚」；
#      ③ 释放**主君监狱**与**共治者自己监狱**中的全部囚犯；
#      ④ 所有受益者（被赦封臣 / 被释囚犯）大幅提升对共治者的好感，
#         并按好感分档有概率触发 ftr_diarchy.0300 引导其为共治者背书；
#      ⑤ 代价 = 主君不满。
ftr_diarchy_amnesty_effect = {
	save_scope_as = ftr_diarchy_amnestying_diarch
	# 固定值削减自身不和
	change_strife_opinion = ftr_diarchy_amnesty_strife_reduction_value

	liege ?= {
		# 代价：主君不满
		add_opinion = {
			target = scope:ftr_diarchy_amnestying_diarch
			modifier = ftr_diarchy_amnesty_liege_opinion
		}

		# ② 赦免：主君所有直属封臣的罪行一律记为「已处罚」
		# （原版效果内部自带 has_*_reason 判空，对无罪行者也安全）
		custom_tooltip = {
			text = ftr_diarchy_amnesty_pardon_tt
			every_vassal = {
				consume_all_criminal_reasons_effect = {
					LIEGE = prev
					CRIMINAL = this
				}
				ftr_diarchy_amnesty_beneficiary_effect = {
					AMNESTIER = scope:ftr_diarchy_amnestying_diarch
				}
			}
		}

		# ③-a 释放主君监狱中的所有囚犯
		custom_tooltip = {
			text = ftr_diarchy_amnesty_release_tt
			every_prisoner = {
				release_from_prison = yes
				ftr_diarchy_amnesty_beneficiary_effect = {
					AMNESTIER = scope:ftr_diarchy_amnestying_diarch
				}
			}
		}
	}

	# ③-b 释放共治者自己监狱中的所有囚犯
	custom_tooltip = {
		text = ftr_diarchy_amnesty_release_tt
		every_prisoner = {
			release_from_prison = yes
			ftr_diarchy_amnesty_beneficiary_effect = {
				AMNESTIER = scope:ftr_diarchy_amnestying_diarch
			}
		}
	}
}

# 求值域：受益者本人。参数 AMNESTIER = 大赦发起者（共治者）。
# ④ 受益者：大幅提升对共治者的好感 + 按好感分档概率触发背书引导事件。
ftr_diarchy_amnesty_beneficiary_effect = {
	$AMNESTIER$ ?= { save_scope_as = ftr_diarchy_amnesty_diarch }
	if = {
		limit = {
			exists = scope:ftr_diarchy_amnesty_diarch
			NOT = { this = scope:ftr_diarchy_amnesty_diarch }
		}
		# 大幅增加对共治者的好感
		add_opinion = {
			target = scope:ftr_diarchy_amnesty_diarch
			modifier = ftr_diarchy_amnesty_beneficiary_opinion
		}
		# 供事件读取的施恩者指针（scope 不跨事件）
		set_variable = {
			name = ftr_diarchy_amnesty_patron
			value = scope:ftr_diarchy_amnesty_diarch
			days = 60
		}
		# 依好感度分档概率：高好感 25%，中好感 10%
		if = {
			limit = {
				opinion = {
					target = scope:ftr_diarchy_amnesty_diarch
					value >= ftr_diarchy_amnesty_backing_opinion_high_value
				}
			}
			random_list = {
				# 25% 引导背书
				25 = {
					trigger_event = {
						id = ftr_diarchy.0300
						days = { 3 12 }
					}
				}
				75 = { }
			}
		}
		else_if = {
			limit = {
				opinion = {
					target = scope:ftr_diarchy_amnesty_diarch
					value >= ftr_diarchy_amnesty_backing_opinion_mid_value
				}
			}
			random_list = {
				# 10% 引导背书
				10 = {
					trigger_event = {
						id = ftr_diarchy.0300
						days = { 3 12 }
					}
				}
				90 = { }
			}
		}
	}
}

# 求值域：受益者本人。参数 CHAMPION = 被背书者（共治者）。
# 为共治者背书：按 01 的双记账规范登记（支持者侧 + 被支持者名单），并按 02 记拥立功。
ftr_diarchy_amnesty_back_benefactor_effect = {
	$CHAMPION$ ?= { save_scope_as = ftr_diarchy_backing_champion }
	if = {
		limit = { exists = scope:ftr_diarchy_backing_champion }
		# 01 记账 ①：支持者侧——我背书的对象
		set_variable = {
			name = ftr_court_backer_of
			value = scope:ftr_diarchy_backing_champion
		}
		# 01 记账 ②：被支持者侧名单（双记账一致）
		scope:ftr_diarchy_backing_champion = {
			add_to_variable_list = {
				name = ftr_court_supporters
				target = prev
			}
		}
		# 好感：背书亲近
		add_opinion = {
			target = scope:ftr_diarchy_backing_champion
			modifier = ftr_court_backer_opinion
		}
		# 02 记账：拥立功（求值域 = 效劳方 = 受益者本人）
		ftr_personal_merit_backing_add_effect = {
			CLAIMANT = scope:ftr_diarchy_backing_champion
			AMOUNT = ftr_diarchy_amnesty_backing_merit_value
		}
	}
}

### 六、准备篡位：阶段 A（§4.4-A）
######################################################

# 求值域：共治者本人。广播表态 + 开启阶段 A 计时。
ftr_diarchy_phase_a_start_effect = {
	save_scope_as = ftr_diarchy_preparing_diarch
	remove_variable = ftr_diarchy_phase_a_opposed
	remove_character_flag = ftr_diarchy_phase_a_done
	remove_character_flag = ftr_diarchy_phase_a_timeout
	# 阶段 A 双计时（D6 双通道）：
	#   ① 表态窗口结束即结算 → ftr_diarchy.0021 置 ftr_diarchy_phase_a_done
	#      （「全部表态」的代理，解锁「无人反对」路径）
	#   ② 真·超时兜底 → ftr_diarchy.0022 置 ftr_diarchy_phase_a_timeout
	#      （有清君侧战争久拖不决时，期满自动解锁）
	trigger_event = {
		id = ftr_diarchy.0021
		days = ftr_diarchy_phase_a_settle_days_value
	}
	trigger_event = {
		id = ftr_diarchy.0022
		days = ftr_diarchy_phase_a_timeout_days_value
	}

	liege ?= {
		save_scope_as = ftr_diarchy_prep_liege
		# 通知主君：共治者准备篡位（公开警告，非秘密）
		trigger_event = { id = ftr_diarchy.0010  days = 1 }
		# 向全部直属封臣广播表态请求
		every_vassal = {
			limit = {
				is_landed = yes
				NOT = { this = scope:ftr_diarchy_preparing_diarch }
			}
			trigger_event = { id = ftr_diarchy.0020  days = 1 }
		}
	}

	# 党羽游说对垒（§4.4-A D13 重做：双方各抽一人能力对垒，一次仅 ±5，不再按人数逐名堆叠）
	ftr_diarchy_backer_lobby_pk_effect = yes
}

# 求值域：共治者本人。阶段 A 收尾：记录完成标记并评估解锁。

	add_character_flag = {

		days = ftr_diarchy_phase_a_done_days_value
	}
}

# 求值域：共治者本人。党羽游说对垒（§4.4-A D13；2026-09 重做）。
# 旧版：每名党羽各自 +5 → 人数直接换算 swing，实测把天平推得过快（人数堆叠激进）。
# 新版：双方从各自 ftr_court_supporters 名单中**各随机抽一人**对垒——
#   评分 = 智谋 + 外交（§4.4-A 的两项关键属性），
#   共治者方胜率 = 基准 50 +（评分差 × 2.5），夹在 10~90；
#   胜方一次推动天平 ±5（共治者方胜 = 向共治者，君主方胜 = 向君主）。
# 四重频率闸门：① 仅阶段 A 开局调用一次；② 同政权冷却（ftr_diarchy_lobby_pk_cooldown_years_value）内不再对垒；
#   ③ 任一方无人可派（名单为空 / 无合格者）时不出手，杜绝单方面白拿 swing；
#   ④ 参战者个人冷却（ftr_diarchy_lobby_pk_lobbyist_cooldown_days_value）：阵营内部轮换——
#      避免「人少但能力强」的阵营每次都派同一强手碾压对方弱手，把对垒变成一边倒。
ftr_diarchy_backer_lobby_pk_effect = {
	save_scope_as = ftr_diarchy_pk_diarch

	# 抽签①：共治者阵营一人
	if = {
		limit = { scope:ftr_diarchy_pk_diarch = { has_variable_list = ftr_court_supporters } }
		scope:ftr_diarchy_pk_diarch = {
			random_in_list = {
				variable = ftr_court_supporters
				limit = { ftr_diarchy_lobby_pk_candidate_trigger = yes }
				save_scope_as = ftr_diarchy_pk_diarch_lobbyist
			}
		}
	}

	liege ?= {
		save_scope_as = ftr_diarchy_pk_liege
		# 抽签②：君主阵营一人
		if = {
			limit = { has_variable_list = ftr_court_supporters }
			random_in_list = {
				variable = ftr_court_supporters
				limit = { ftr_diarchy_lobby_pk_candidate_trigger = yes }
				save_scope_as = ftr_diarchy_pk_liege_lobbyist
			}
		}
		# 对垒：双方齐备 + 共治仍在 + 不在冷却，才发生
		if = {
			limit = {
				exists = scope:ftr_diarchy_pk_diarch_lobbyist
				exists = scope:ftr_diarchy_pk_liege_lobbyist
				has_active_diarchy = yes
				exists = diarch
				NOT = { has_character_flag = ftr_diarchy_lobby_pk_cooldown }
			}
			add_character_flag = {
				flag = ftr_diarchy_lobby_pk_cooldown
				years = ftr_diarchy_lobby_pk_cooldown_years_value
			}
			# 个人冷却：出战双方一年内不再被抽中（阵营内轮换，防强手反复出战造成一边倒）
			scope:ftr_diarchy_pk_diarch_lobbyist = {
				add_character_flag = {
					flag = ftr_diarchy_lobby_pk_lobbied
					days = ftr_diarchy_lobby_pk_lobbyist_cooldown_days_value
				}
			}
			scope:ftr_diarchy_pk_liege_lobbyist = {
				add_character_flag = {
					flag = ftr_diarchy_lobby_pk_lobbied
					days = ftr_diarchy_lobby_pk_lobbyist_cooldown_days_value
				}
			}
			# 掷骰：在共治者方游说者域求值其胜率（评分差加权）
			scope:ftr_diarchy_pk_diarch_lobbyist = {
				random = {
					chance = ftr_diarchy_lobby_pk_diarch_win_chance_value
					scope:ftr_diarchy_pk_liege = { add_character_flag = ftr_diarchy_lobby_pk_diarch_won }
				}
			}
			# 结算：共治者方得手 → 天平向共治者；否则君主方得手 → 天平向君主
			if = {
				limit = { has_character_flag = ftr_diarchy_lobby_pk_diarch_won }
				remove_character_flag = ftr_diarchy_lobby_pk_diarch_won
				update_diarchy_swing_with_perspective_effect = {
					SWING = ftr_diarchy_lobby_pk_swing_gain
					LIEGE = liege
				}
				ftr_diarchy_swing_notice_effect = { KEY = ftr_diarchy_swing_notice_lobby_win }
			}
			else = {
				update_diarchy_swing_with_perspective_effect = {
					SWING = ftr_diarchy_lobby_pk_swing_loss
					LIEGE = liege
				}
				ftr_diarchy_swing_notice_effect = { KEY = ftr_diarchy_swing_notice_lobby_lose }
			}
		}
	}
}

### 七、清君侧（§5）
######################################################

# 求值域：表态造反的封臣本人（§4.4-A 选项 b）。
# 已有清君侧战争 → 直接加入（不重复开战）；否则对主君宣战。
ftr_qingjunce_try_declare_war_effect = {
	save_scope_as = ftr_qingjunce_declarer
	liege ?= { save_scope_as = ftr_qingjunce_target_liege }
	if = {
		limit = {
			exists = scope:ftr_qingjunce_target_liege
			scope:ftr_qingjunce_target_liege = { is_alive = yes }
		}
		if = {
			limit = { scope:ftr_qingjunce_target_liege = { has_variable = ftr_qingjunce_current_war } }
			scope:ftr_qingjunce_target_liege = {
				var:ftr_qingjunce_current_war ?= {
					add_attacker = scope:ftr_qingjunce_declarer
				}
			}
		}
		else = {
			scope:ftr_qingjunce_declarer = {
				start_war = {
					target = scope:ftr_qingjunce_target_liege
					casus_belli = ftr_qingjunce
				}
			}
		}
	}
}

# 求值域：清君侧宣战方（封臣）本人。胜利：顶替现任共治者。
# 依赖外层：scope:ftr_qingjunce_liege（主君）、scope:ftr_qingjunce_old_diarch（现任共治者）
ftr_qingjunce_win_effect = {
	save_scope_as = ftr_qingjunce_winner
	scope:ftr_qingjunce_liege ?= {
		# 免去现任共治者
		if = {
			limit = { has_active_diarchy = yes }
			depose_diarch = yes
		}
		# 顶替：清君侧者成为新摄政（E1：depose + set_diarch）
		if = { limit = { NOT = { has_active_diarchy = yes } }  try_start_diarchy = regency }
		if = {
			limit = { has_active_diarchy = yes }
			set_diarch = scope:ftr_qingjunce_winner
			update_diarchy_swing_with_perspective_effect = {
				SWING = ftr_qingjunce_win_swing_value
				LIEGE = liege
			}
			# 天平偏移通知（§7.1）：新共治者以战功上位，说明天平上移的来由
			ftr_diarchy_swing_notice_effect = { KEY = ftr_diarchy_swing_notice_qingjunce }
		}
	}
	# 拥立功：复用 02 既有记账（求值域 = 效劳方 = 胜方本人）
	ftr_personal_merit_crown_add_effect = { AMOUNT = ftr_qingjunce_merit_value }
	# 现任共治者被囚（E2）
	scope:ftr_qingjunce_old_diarch ?= {
		if = { limit = { is_imprisoned = no }  imprison = { target = this  type = house_arrest } }
		add_opinion = {
			target = scope:ftr_qingjunce_winner
			modifier = ftr_diarchy_ousted_liege_opinion
		}
	}
}

# 求值域：清君侧宣战方（封臣）本人。失败：永久监禁。
ftr_qingjunce_defeat_effect = {
	save_scope_as = ftr_qingjunce_loser
	scope:ftr_qingjunce_liege ?= {
		add_opinion = {
			target = scope:ftr_qingjunce_loser
			modifier = ftr_diarchy_deposed_prince_opinion
		}
		if = {
			limit = { scope:ftr_qingjunce_loser = { is_imprisoned = no } }
			imprison = {
				target = scope:ftr_qingjunce_loser
				type = house_arrest
			}
		}
	}
	# 现任共治者巩固权力：清君侧失败 ⇒ 篡位链解锁
	scope:ftr_qingjunce_old_diarch ?= {
		remove_variable = ftr_diarchy_phase_a_opposed
		add_character_flag = { flag = ftr_diarchy_phase_a_timeout  days = 30 }
		change_strife_opinion = ftr_qingjunce_defeat_strife_value
	}
}

### 八、季度背书博弈（§7.1 整合：swing 唯一的人工成长源）
######################################################

# 求值域：君主（季度脉冲，playable）。
# ① 按「背书共治者人数 − 背书君主人数 × 折算溢价」小幅移动 swing（无任何被动月度修正）；
# ② 依天平档位概率派发 01 的共治拉锯事件（0500 反共治 / 0510 共治反击）；
#    这两个事件已泛化：无朝堂角逐 story 时自动用共治关系补全作用域。
ftr_diarchy_quarterly_swing_drift_effect = {
	save_scope_as = ftr_diarchy_drift_liege

	# ① 背书差 → swing 漂移（夹在 ±上限内，幅度刻意压小）
	# 正值设门槛（加权领先 ≥ positive_min 才推共治者），负值不设门槛（君主领先必能拉回）
	if = {
		limit = {
			NOT = { ftr_diarchy_backer_swing_drift_value = 0 }
			OR = {
				ftr_diarchy_backer_swing_drift_value < 0
				ftr_diarchy_backer_swing_drift_value >= ftr_diarchy_swing_drift_positive_min_value
			}
		}
		# 诊断（临时，排完可删）：mod 侧唯一的周期性 swing 写入，每季度至多一次。
		# 抓不到本行却仍见天平上升 ⇒ 增量来自 mod 之外（原版 mandate 链 / 共治者的摆动天平 / 引擎月度回落）。
		debug_log = "FTR_SWING_DRIFT: quarterly backer drift applied"
		debug_log_scopes = yes
		hidden_effect = {
			change_diarchy_swing = ftr_diarchy_backer_swing_drift_value
		}
		# 天平偏移通知（§7.1）：按方向给不同原因键（正 = 共治者党羽占优 / 负 = 君主党羽占优）
		if = {
			limit = { ftr_diarchy_backer_swing_drift_value > 0 }
			ftr_diarchy_swing_notice_effect = { KEY = ftr_diarchy_swing_notice_drift_up }
		}
		else = {
			ftr_diarchy_swing_notice_effect = { KEY = ftr_diarchy_swing_notice_drift_down }
		}
	}

	# ② 共治者占优（≥ 中性 50）：君主阵营有人暗中削弱共治（0500）
	if = {
		limit = {
			diarchy_swing >= ftr_diarch_counter_swing_threshold_value
			# 单政权冷却：冷却期内不再派发（见 ftr_diarch_swing_event_cooldown_days_value）
			NOT = { has_character_flag = ftr_diarch_swing_event_cooldown }
		}
		random = {
			chance = ftr_diarchy_swing_event_chance_value
			random_vassal = {
				limit = {
					is_landed = yes
					is_adult = yes
					is_imprisoned = no
					NOT = { this = prev.diarch }
					# 排除共治者党羽（他们不会替君主出手）
					NOT = {
						has_variable = ftr_court_backer_of
						var:ftr_court_backer_of ?= { this = prev.diarch }
					}
				}
				trigger_event = {
					id = ftr_court_struggle.0500
					days = { 1 10 }
				}
				# 打冷却 + 派发溯源（诊断，排完可删）
				scope:ftr_diarchy_drift_liege = {
					add_character_flag = {
						flag = ftr_diarch_swing_event_cooldown
						days = ftr_diarch_swing_event_cooldown_days_value
					}
				}
				debug_log = "FTR_DISPATCH_0500_DRIFT"
				debug_log_scopes = yes
			}
		}
	}
	# ③ 君主强势（≤30）：共治者阵营游说反扑——2026-09 起**不再派发 0510 事件**，
	#    统一改由 03 的「党羽游说对垒」(ftr_diarchy_backer_lobby_pk_effect) 承接：
	#    双方各抽一人按「智谋 + 外交」对垒，胜方一次推动 ±5；另有政权 1 年 + 个人 1 年冷却。
	# 前置：共治者党羽不落后于君主（有党羽撑腰才谈反扑）；频率低于君主侧，避免「只涨不跌」
	else_if = {
		limit = {
			diarchy_swing <= ftr_swing_tier_loyal_value
			ftr_diarchy_backer_lead_value >= 0
			NOT = { has_character_flag = ftr_diarch_swing_event_cooldown }
		}
		diarch ?= {
			random = {
				chance = ftr_diarch_backlash_event_chance_value
				# 打冷却 + 派发溯源（诊断，排完可删）
				scope:ftr_diarchy_drift_liege = {
					add_character_flag = {
						flag = ftr_diarch_swing_event_cooldown
						days = ftr_diarch_swing_event_cooldown_days_value
					}
				}
				debug_log = "FTR_DISPATCH_LOBBY_PK_DRIFT"
				debug_log_scopes = yes
				ftr_diarchy_backer_lobby_pk_effect = yes
			}
		}
	}
}

# 求值域：共治者本人。把参数 VASSAL 拉入自己所在的我方战线（§4.5）
ftr_diarchy_levy_vassal_to_wars_effect = {
	save_scope_as = ftr_diarchy_levy_diarch
	$VASSAL$ ?= { save_scope_as = ftr_diarchy_levy_vassal }
	if = {
		limit = { exists = scope:ftr_diarchy_levy_vassal }
		scope:ftr_diarchy_levy_diarch = {
			every_character_war = {
				if = {
					limit = { is_attacker = scope:ftr_diarchy_levy_diarch }
					add_attacker = scope:ftr_diarchy_levy_vassal
				}
				else_if = {
					limit = { is_defender = scope:ftr_diarchy_levy_diarch }
					add_defender = scope:ftr_diarchy_levy_vassal
				}
			}
		}
	}
}
