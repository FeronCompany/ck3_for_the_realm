
# Coup success effect
ftr_coup_success_effect = {
	$SCHEME$ = {
		save_scope_as = coup_scheme
	}
	scope:coup_scheme = {
		scheme_owner = {
			save_scope_as = owner
		}
		scheme_target_character = {
			save_scope_as = target
		}
		hidden_effect = {
			every_scheme_agent_character = {
				add_opinion = {
					modifier = claimant_faction_member_opinion
					target = scope:owner
				}
				# 从龙功（coup_scheme）：参与者成功后对密谋发起人（新君）获得 +20 拥立功
				ftr_personal_merit_backing_add_effect = {
					CLAIMANT = scope:owner
					AMOUNT = ftr_personal_merit_coup_support_value
				}
			}
		}
	}
	scope:owner = {
		imprison = {
			target = scope:target
			type = dungeon
		}
		add_legitimacy = massive_legitimacy_loss
		add_dread = 50
	}
	ftr_coup_transfer_realm_effect = yes
}

# 政变夺权共用段（05 §3.3）：头衔转移 + 封臣改宗 + 宫廷宏伟度继承。
# 依赖：scope:owner = 政变发起者；scope:target = 被夺位者。
ftr_coup_transfer_realm_effect = {
	create_title_and_vassal_change = {
		type = usurped
		save_scope_as = change
		add_claim_on_loss = yes
	}
	scope:target = {
		add_opinion = {
			target = scope:owner
			modifier = ftr_success_coup_on_me_opinion
		}
		every_held_title = {
			limit = {
				OR = {
					tier >= tier_kingdom
					tier = scope:target.primary_title.tier
					AND = {
						tier = tier_county
						exists = scope:target.capital_county
						this = scope:target.capital_county
					}
				}
			}
			change_title_holder_include_vassals = {
				holder = scope:owner
				change = scope:change
				government_base = scope:target
			}
		}
		every_vassal = {
			change_liege = {
				liege = scope:owner
				change = scope:change
			}
		}
		change_liege = {
			liege = scope:owner
			change = scope:change
		}
	}
	if = {
		limit = {
			scope:target = {
				has_royal_court = yes
			}
			scope:owner = {
				has_royal_court = no
			}
		}
		scope:owner = {
			set_variable = {
				name = previous_holder_grandeur_value
				value = scope:target.court_grandeur_current
			}
		}
	}
	resolve_title_and_vassal_change = scope:change
	scope:owner = {
		if = {
			limit = {
				exists = var:previous_holder_grandeur_value
				has_royal_court = yes
			}
			hidden_effect = {
				change_current_court_grandeur = -100 # Reset to 0
				change_current_court_grandeur = var:previous_holder_grandeur_value
				remove_variable = previous_holder_grandeur_value
			}
			change_current_court_grandeur = massive_court_grandeur_gain # Gain a bonus to Grandeur for winning a claim faction
		}
	}
}

# Coup failure effect
ftr_coup_failure_effect = {
	$SCHEME$ = {
		save_scope_as = coup_scheme
	}
	scope:coup_scheme = {
		scheme_target_character = {
			save_scope_as = target
		}
		every_scheme_agent_character = {
			limit = { NOT = { this = scope:coup_scheme.scheme_owner } }
			scope:target = {
				imprison = {
					target = prev
					type = dungeon
				}
				add_opinion = {
					target = prev
					modifier = ftr_attempted_coup_crime
				}
			}
		}
		scope:target = {
			imprison = {
				target = scope:coup_scheme.scheme_owner
				type = dungeon
			}
			add_opinion = {
				target = scope:coup_scheme.scheme_owner
				modifier = ftr_attempted_coup_crime
			}
		}
		end_scheme = yes
	}
}

ftr_coup_scheme_prep_effect = {
	save_scope_as = scheme
	scheme_owner = {
		save_scope_as = owner
	}
	scheme_target_character = {
		save_scope_as = target
	}
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:ftr_coup_outcoming.4000
	}
	if = {
		limit = {
			NOT = { exists = scope:suppress_next_event }
		}
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

# 彻底夺产（05 §3.3）：把被废君主名下剩余头衔一并收归新君，使其彻底无地。
# 依赖：scope:owner = 新君；scope:target = 被废君主。
ftr_coup_strip_all_titles_effect = {
	create_title_and_vassal_change = {
		type = usurped
		save_scope_as = ftr_coup_strip_change
		add_claim_on_loss = yes
	}
	scope:target = {
		every_held_title = {
			change_title_holder_include_vassals = {
				holder = scope:owner
				change = scope:ftr_coup_strip_change
				government_base = scope:target
			}
		}
	}
	resolve_title_and_vassal_change = scope:ftr_coup_strip_change
}

# 君主在外·政变成功（05 §3.3）：矫诏废君——不擒获，剥夺全部头衔，目标转无地冒险者。
ftr_coup_away_success_effect = {
	$SCHEME$ = {
		save_scope_as = coup_scheme
	}
	scope:coup_scheme = {
		scheme_owner = {
			save_scope_as = owner
		}
		scheme_target_character = {
			save_scope_as = target
		}
		hidden_effect = {
			every_scheme_agent_character = {
				add_opinion = {
					modifier = claimant_faction_member_opinion
					target = scope:owner
				}
				ftr_personal_merit_backing_add_effect = {
					CLAIMANT = scope:owner
					AMOUNT = ftr_personal_merit_coup_support_value
				}
			}
		}
	}
	scope:owner = {
		add_legitimacy = massive_legitimacy_loss
		add_dread = 50
	}
	ftr_coup_transfer_realm_effect = yes
	ftr_coup_strip_all_titles_effect = yes
	# 被废君主转无地冒险者（EP3 守卫；无该 DLC 时保留无地状态）
	if = {
		limit = {
			scope:target ?= { is_alive = yes }
			has_dlc_feature = roads_to_power
		}
		scope:target = {
			set_landless_government_effect = yes
			create_landless_adventurer_title_effect = {
				REASON = flag:deposition
				FLAVOR_CHAR = scope:owner
			}
			create_landless_adventurer_title_tooltip_effect = yes
			adventurer_prestige_reset_effect = yes
		}
	}
}
