
# 参数 XP = 本场战斗应得经验，由调用方（战斗方域）先按敌方参战规模算好传入
ftr_battle_commander_rank_up_effect = {
	if = {
		limit = {
			NOT = { has_trait = ftr_battle_commander }
		}
		add_trait = ftr_battle_commander
	}
	else_if = {
		limit = {
			has_trait_xp = {
				trait = ftr_battle_commander
				value < 80
			}
		}
		add_trait_xp = {
			trait = ftr_battle_commander
			value = $XP$
		}
	}
}
