##############################################################
### 朝堂权力角逐 COURT STRUGGLE — 效果层
### 议题1 继任之争 / 议题2 共治夺权
###
### 作用域约定（重要）：
###   本文件多数 effect 要求在【story 作用域】执行。
###   内部统一使用 scope:ftr_court_story（故事）与 scope:ftr_court_monarch（君主），
###   不依赖 root / prev，避免域链歧义。
###   单人向的 effect（站队、承诺等）在角色自身域执行。
##############################################################

######################################################
### 一、初始化
######################################################

# 在 story 域执行：建立初始状态
ftr_court_setup_effect = {
	save_scope_as = ftr_court_story
	story_owner = { save_scope_as = ftr_court_monarch }

	# 初始变量
	set_variable = { name = ftr_court_candidate_count  value = 0 }
	set_variable = { name = ftr_court_top_score  value = 0 }
	set_variable = { name = ftr_court_second_score  value = 0 }
	set_variable = { name = ftr_court_top_gap  value = 999 }
	set_variable = { name = ftr_court_regency_swing  value = 0 }
	set_variable = { name = ftr_court_phase  value = 0 }

	# 建立候选池
	ftr_court_update_candidates_effect = yes
	# 刷新支持者计数
	ftr_court_refresh_backer_count_effect = yes
}

# 在 story 域执行：结束后清理，移除一切残留
ftr_court_cleanup_effect = {
	save_scope_as = ftr_court_story

	# 清理临时公国（兜底，防止异常路径泄漏）
	every_in_list = {
		list = ftr_court_claimant_list
		ftr_court_destroy_temp_duchy_effect = yes
	}

	# 清理候选人个体变量
	every_in_list = {
		list = ftr_court_claimant_list
		remove_variable = ftr_claim_strength
		remove_variable = ftr_court_backer_count
		remove_variable = ftr_court_backer_of
		remove_character_flag = ftr_court_pooled
	}

	remove_variable = ftr_court_candidate_pool
	remove_variable = ftr_court_claimant_list
	remove_variable = ftr_court_candidate_count
	remove_variable = ftr_court_top_score
	remove_variable = ftr_court_second_score
	remove_variable = ftr_court_top_gap
	remove_variable = ftr_court_regency_swing
	remove_variable = ftr_court_phase
}

######################################################
### 二、候选池与竞争力排名
######################################################

# 在 story 域执行：重建候选池、重算评分、选出前 N 名
ftr_court_update_candidates_effect = {
	save_scope_as = ftr_court_story
	story_owner = { save_scope_as = ftr_court_monarch }

	# 清空旧池（先清标记，避免 flag 残留）
	every_in_list = {
		list = ftr_court_candidate_pool
		remove_character_flag = ftr_court_pooled
	}
	remove_variable = ftr_court_candidate_pool
	remove_variable = ftr_court_claimant_list

	# ---- 1. 收集候选人：继承人 + 成年子女 + 成年手足 ----
	scope:ftr_court_monarch = {
		every_heir = {
			limit = { ftr_court_claimant_valid_trigger = yes }
			ftr_court_try_pool_candidate_effect = yes
		}
		every_child = {
			limit = { ftr_court_claimant_valid_trigger = yes }
			ftr_court_try_pool_candidate_effect = yes
		}
		every_sibling = {
			limit = { ftr_court_claimant_valid_trigger = yes }
			ftr_court_try_pool_candidate_effect = yes
		}
	}

	# ---- 2. 计算竞争力评分 ----
	every_in_list = {
		list = ftr_court_candidate_pool
		set_variable = {
			name = ftr_claim_strength
			value = ftr_court_claim_strength_value
		}
	}

	# ---- 3. 统计支持者 ----
	ftr_court_refresh_backer_count_effect = yes

	# ---- 4. 按评分降序取前 N 名 ----
	scope:ftr_court_story = {
		set_variable = { name = ftr_court_rank_counter  value = 0 }
		set_variable = { name = ftr_court_top_score  value = 0 }
		set_variable = { name = ftr_court_second_score  value = 0 }
	}

	ordered_in_list = {
		list = ftr_court_candidate_pool
		order_by = var:ftr_claim_strength
		save_scope_as = ftr_court_candidate
		scope:ftr_court_story = {
			# 记录第一名分数
			if = {
				limit = { var:ftr_court_rank_counter = 0 }
				set_variable = {
					name = ftr_court_top_score
					value = scope:ftr_court_candidate.var:ftr_claim_strength
				}
			}
			# 记录第二名分数
			else_if = {
				limit = { var:ftr_court_rank_counter = 1 }
				set_variable = {
					name = ftr_court_second_score
					value = scope:ftr_court_candidate.var:ftr_claim_strength
				}
			}
			# 前 N 名进入朝堂展示名单
			if = {
				limit = { var:ftr_court_rank_counter < ftr_court_max_candidates_value }
				add_to_variable_list = {
					name = ftr_court_claimant_list
					target = scope:ftr_court_candidate
				}
			}
			change_variable = { name = ftr_court_rank_counter  add = 1 }
		}
	}

	# ---- 5. 写入数量与前两名差距 ----
	scope:ftr_court_story = {
		set_variable = {
			name = ftr_court_candidate_count
			value = list_size:ftr_court_claimant_list
		}
		set_variable = { name = ftr_court_top_gap  value = var:ftr_court_top_score }
		change_variable = { name = ftr_court_top_gap  subtract = var:ftr_court_second_score }
		remove_variable = ftr_court_rank_counter
		ftr_court_update_phase_effect = yes
	}
}

# 在 story 域执行：依据竞争者人数与前两名差距，更新阶段号
ftr_court_update_phase_effect = {
	save_scope_as = ftr_court_story
	set_variable = { name = ftr_court_phase  value = 0 }

	if = {
		limit = { var:ftr_court_candidate_count >= ftr_court_min_candidates_value }
		set_variable = { name = ftr_court_phase  value = 1 }
		if = {
			limit = { var:ftr_court_top_gap <= ftr_court_contested_gap_value }
			set_variable = { name = ftr_court_phase  value = 2 }
			if = {
				limit = {
					var:ftr_court_top_gap <= ftr_court_crisis_gap_value
					story_owner ?= { ftr_court_monarch_weak_trigger = yes }
				}
				set_variable = { name = ftr_court_phase  value = 3 }
			}
		}
	}
}

# 在候选人域执行：去重后加入候选池
ftr_court_try_pool_candidate_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = ftr_court_pooled }
		}
		add_character_flag = ftr_court_pooled
		save_scope_as = ftr_court_candidate
		scope:ftr_court_story = {
			add_to_variable_list = {
				name = ftr_court_candidate_pool
				target = scope:ftr_court_candidate
			}
		}
	}
}

# 在任意角色域执行：刷新其自身、或其宗主所拥有的朝堂角逐
ftr_court_refresh_local_effect = {
	every_owned_story = {
		limit = { story_type = ftr_court_struggle }
		ftr_court_update_candidates_effect = yes
	}
	top_liege ?= {
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			ftr_court_update_candidates_effect = yes
		}
	}
}

######################################################
### 三、支持者记账
######################################################

# 在 story 域执行：重新统计每位候选人的支持者数量
ftr_court_refresh_backer_count_effect = {
	save_scope_as = ftr_court_story
	story_owner = { save_scope_as = ftr_court_monarch }

	# 清零（必须显式 set，避免 change_variable 对不存在变量报错）
	every_in_list = {
		list = ftr_court_claimant_list
		set_variable = { name = ftr_court_backer_count  value = 0 }
	}

	# 统计：直接封臣 + 廷臣
	scope:ftr_court_monarch = {
		every_vassal = {
			limit = { ftr_court_has_backing_trigger = yes }
			ftr_court_count_one_backer_effect = yes
		}
		every_courtier = {
			limit = { ftr_court_has_backing_trigger = yes }
			ftr_court_count_one_backer_effect = yes
		}
	}
}

# 在潜在支持者域执行：其人头计入所支持的对象
ftr_court_count_one_backer_effect = {
	save_scope_as = ftr_court_backer
	var:ftr_court_backer_of ?= {
		save_scope_as = ftr_court_backed_target
		scope:ftr_court_story = {
			every_in_list = {
				list = ftr_court_claimant_list
				limit = { this = scope:ftr_court_backed_target }
				change_variable = { name = ftr_court_backer_count  add = 1 }
			}
		}
	}
}

######################################################
### 四、共治拉锯（议题2）
######################################################

# 在 story 域执行：移动拉锯值。
# 参数通过变量 ftr_court_swing_amount 传入（正=倒向共治者，负=忠于君主），
# 调用方需先在本作用域设置该变量，调用结束后由调用方自行清理。
ftr_court_shift_swing_effect = {
	save_scope_as = ftr_court_story
	if = {
		limit = { has_variable = ftr_court_swing_amount }
		change_variable = {
			name = ftr_court_regency_swing
			add = var:ftr_court_swing_amount
		}
		# 夹紧到区间，避免无限累积
		if = {
			limit = { var:ftr_court_regency_swing > ftr_court_swing_max_value }
			set_variable = { name = ftr_court_regency_swing  value = ftr_court_swing_max_value }
		}
		else_if = {
			limit = { var:ftr_court_regency_swing < ftr_court_swing_min_value }
			set_variable = { name = ftr_court_regency_swing  value = ftr_court_swing_min_value }
		}
	}
}

# 在 story 域执行：年度自然回落 + 君主状态偏置
ftr_court_swing_yearly_effect = {
	save_scope_as = ftr_court_story
	story_owner = { save_scope_as = ftr_court_monarch }

	# 向 0 回落
	if = {
		limit = { var:ftr_court_regency_swing > 0 }
		change_variable = {
			name = ftr_court_regency_swing
			subtract = ftr_court_swing_yearly_decay_value
		}
	}
	else_if = {
		limit = { var:ftr_court_regency_swing < 0 }
		change_variable = {
			name = ftr_court_regency_swing
			add = ftr_court_swing_yearly_decay_value
		}
	}

	# 君主失能 -> 逐年向共治者倾斜
	if = {
		limit = { scope:ftr_court_monarch = { ftr_court_monarch_infirm_trigger = yes } }
		change_variable = {
			name = ftr_court_regency_swing
			add = ftr_court_swing_infirm_bias_value
		}
	}
	else_if = {
		limit = { scope:ftr_court_monarch = { prestige >= ftr_court_monarch_weak_prestige_value } }
		change_variable = {
			name = ftr_court_regency_swing
			add = ftr_court_swing_strong_bias_value
		}
	}
}

### 五、临时公国（无地竞争者起兵用）
######################################################

# 在无地竞争者域执行：创建纯虚拟的临时公国。
# 依赖：scope:ftr_court_monarch（在位君主）。
# 该头衔在战争结束时必定销毁，不论胜负。
ftr_court_create_temp_duchy_effect = {
	save_scope_as = ftr_court_claimant

	# 在王国境内挑一个伯爵领作为临时公国首都
	scope:ftr_court_monarch = {
		random_sub_realm_county = {
			save_scope_as = ftr_court_temp_capital
		}
	}

	if = {
		limit = { exists = scope:ftr_court_temp_capital }

		create_dynamic_title = {
			tier = duchy
			name = FTR_COURT_TEMP_DUCHY_NAME
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = ftr_court_title_change
			add_claim_on_loss = no
		}
		scope:new_title = {
			set_capital_county = scope:ftr_court_temp_capital
			# 纯虚拟头衔：不产生宣称、不传承、销毁即彻底删除
			set_landless_title = yes
			set_destroy_on_succession = yes
			set_delete_on_destroy = yes
			set_no_automatic_claims = yes
			set_definitive_form = yes
			set_can_be_named_after_dynasty = no
			change_title_holder = {
				holder = scope:ftr_court_claimant
				change = scope:ftr_court_title_change
			}
		}
		resolve_title_and_vassal_change = scope:ftr_court_title_change

		# 记录以便战后回收
		set_variable = {
			name = ftr_court_temp_duchy
			value = scope:new_title
		}
	}
}

# 在角色域执行：销毁临时公国（纯虚拟头衔，战后一律移除）
ftr_court_destroy_temp_duchy_effect = {
	if = {
		limit = { has_variable = ftr_court_temp_duchy }
		var:ftr_court_temp_duchy ?= {
			destroy_title = this
		}
		add_gold = ftr_court_temp_duchy_compensation_value
		remove_variable = ftr_court_temp_duchy
	}
}

# 在角色域执行：不论胜负，无条件回收临时公国（用于战争结算）
ftr_court_revoke_temp_duchy_effect = {
	if = {
		limit = { has_variable = ftr_court_temp_duchy }
		var:ftr_court_temp_duchy ?= {
			destroy_title = this
		}
		remove_variable = ftr_court_temp_duchy
	}
}

######################################################
### 五之二、起兵准备（无地者必须先有头衔才能宣战）
######################################################

# 在起兵者域执行：起兵前准备。
# 关键点：无地皇子/王子无法宣战，必须先创建临时公国赋予其宣战资格，
# 然后立即发动继位战争。有地者直接宣战。
# 参数：TARGET = 宣战对象（现任君主）。
ftr_court_prepare_rebellion_effect = {
	save_scope_as = ftr_court_warleader
	# 供 create_temp_duchy_effect 选定临时首都（宣战对象即现任君主）
	$TARGET$ ?= { save_scope_as = ftr_court_monarch }

	# 无地者先备虚拟头衔（已备则跳过）
	if = {
		limit = {
			is_landed = no
			NOT = { has_variable = ftr_court_temp_duchy }
		}
		ftr_court_create_temp_duchy_effect = yes
	}

	start_war = {
		casus_belli = ftr_succession_war
		target = $TARGET$
	}
}

### 六、内战：召集支持者
######################################################

# 在起兵者域执行：把所有已站队支持自己的封臣拉入战争。
# 依赖：scope:ftr_court_war（战争）、scope:ftr_court_monarch（君主）、scope:ftr_court_story（故事）
ftr_court_call_backers_to_war_effect = {
	save_scope_as = ftr_court_warleader

	scope:ftr_court_monarch = {
		every_vassal = {
			limit = {
				has_variable = ftr_court_backer_of
				var:ftr_court_backer_of ?= { this = scope:ftr_court_warleader }
				is_landed = yes
				NOT = { is_participant_in_war = scope:ftr_court_war }
			}
			save_scope_as = ftr_court_backer
			scope:ftr_court_war = {
				add_attacker = scope:ftr_court_backer
			}
		}
	}
}

# 在君主域执行：把保皇派拉入防御方
ftr_court_call_loyalists_to_war_effect = {
	save_scope_as = ftr_court_monarch

	every_vassal = {
		limit = {
			NOT = { is_participant_in_war = scope:ftr_court_war }
			OR = {
				NOT = { has_variable = ftr_court_backer_of }
				# 站队对象不是起兵者
				NOT = { var:ftr_court_backer_of ?= { this = scope:ftr_court_warleader } }
			}
			# 与君主关系良好者才会勤王
			opinion = { target = scope:ftr_court_monarch  value >= 0 }
		}
		save_scope_as = ftr_court_loyalist
		scope:ftr_court_war = {
			add_defender = scope:ftr_court_loyalist
		}
	}
}

######################################################
### 七、夺位与清算
######################################################

# 在新君域执行：取代君主并将其囚禁，随后跳清算事件。
# 依赖：scope:ftr_court_old_monarch（被推翻的君主）
ftr_court_usurp_throne_effect = {
	save_scope_as = ftr_court_new_monarch

	# 夺取主头衔
	scope:ftr_court_old_monarch = {
		save_scope_as = ftr_court_deposed
	}
	scope:ftr_court_deposed.primary_title ?= {
		save_scope_as = ftr_court_seized_title
	}

	if = {
		limit = { exists = scope:ftr_court_seized_title }
		create_title_and_vassal_change = {
			type = all
			save_scope_as = ftr_court_usurp_change
		}
		scope:ftr_court_seized_title = {
			change_title_holder = {
				holder = scope:ftr_court_new_monarch
				change = scope:ftr_court_usurp_change
			}
		}
		resolve_title_and_vassal_change = scope:ftr_court_usurp_change
	}

	# 囚禁老君主
	scope:ftr_court_new_monarch = {
		imprison = {
			target = scope:ftr_court_deposed
			type = house_arrest
		}
		# 清算事件：决定老君王与其它竞争者的命运
		trigger_event = {
			id = ftr_court_struggle.0010
			days = 3
		}
		# 内阁席位承诺：继承稍后提醒（随机延迟避免密集弹窗）
		ftr_court_schedule_council_reminder_effect = yes
		# 新君进入冷却期，避免刚上位就被连锁推翻
		add_character_flag = {
			flag = ftr_court_cooldown
			years = ftr_court_cooldown_years_value
		}
		# 朝堂通报：大位易主
		save_scope_as = ftr_court_monarch
		ftr_court_send_court_toast_effect = {
			TITLE = ftr_court_notice_usurp
			ICON = scope:ftr_court_new_monarch
		}
	}

	# 旧君被废后其名下的朝堂角逐收场（旧君尚存但已失位，story 的 owner 不会自行转移）
	scope:ftr_court_old_monarch = {
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			end_story = yes
		}
	}
}

# 在新君域执行：欠有内阁席位承诺时，安排首次提醒事件。
ftr_court_schedule_council_reminder_effect = {
	if = {
		limit = {
			has_variable = ftr_court_owes_council
			var:ftr_court_owes_council > 0
		}
		trigger_event = {
			id = ftr_court_struggle.0016
			days = { ftr_court_council_remind_delay_min_value  ftr_court_council_remind_delay_max_value }
		}
	}
}

# 在君主域执行：立储。抬高被立者的竞争力，其余竞争者同生怨望。
# 参数：NOMINEE
ftr_court_nominate_effect = {
	save_scope_as = ftr_court_monarch

	$NOMINEE$ ?= {
		save_scope_as = ftr_court_nominee
		add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_political_alliance_opinion }
		if = {
			limit = { has_variable = ftr_claim_strength }
			change_variable = { name = ftr_claim_strength  add = ftr_court_nominate_gain_value }
		}
	}

	every_owned_story = {
		limit = { story_type = ftr_court_struggle }
		save_scope_as = ftr_court_story
	}
	scope:ftr_court_story ?= {
		every_in_list = {
			list = ftr_court_claimant_list
			limit = { NOT = { this = scope:ftr_court_nominee } }
			add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_political_opponent_opinion }
			if = {
				limit = { has_variable = ftr_claim_strength }
				change_variable = { name = ftr_claim_strength  add = ftr_court_nominate_penalty_value }
			}
		}
		ftr_court_update_phase_effect = yes
	}
}

# 在君主域执行：压制某位竞争者，削其声势与人望。
# 参数：TARGET
ftr_court_suppress_effect = {
	save_scope_as = ftr_court_monarch

	$TARGET$ ?= {
		add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_court_rival_claimant_opinion }
		if = {
			limit = { has_variable = ftr_claim_strength }
			change_variable = { name = ftr_claim_strength  add = ftr_court_suppress_penalty_value }
		}
		if = {
			limit = { has_variable = ftr_court_backer_count }
			change_variable = { name = ftr_court_backer_count  add = ftr_court_suppress_backer_loss_value }
		}
	}

# 在任命者（新君）域执行：把被承诺者按其最强属性任命为内阁成员。
# 直接复用官方 assign_appropriate_councillor_position_effect（00_councillor_effects.txt:806）：
#   按最高技能 switch 首选职位 -> 职位空缺或现任可解雇则任命，否则四级 fallback。
# 首选/备选均为官方五席（chancellor/steward/spymaster/marshal/court_chaplain），
# 全政体通用，满足"必须是可任命的职位"。
# 注意：不用 root（本 effect 会被其它 scripted effect 嵌套调用，root 不可靠），
#       统一用 save_scope_as 固定任命者作用域。
# 参数：TARGET = 被承诺者
ftr_court_appoint_promised_councillor_effect = {
	save_scope_as = ftr_court_council_owner

	$TARGET$ ?= {
		# 非封臣的承诺对象先拉入宫廷（官方 petition_liege 范式），否则无法就任
		if = {
			limit = {
				NOT = { is_vassal_of = scope:ftr_court_council_owner }
			}
			scope:ftr_court_council_owner = {
				add_courtier = prev
			}
		}
	}
	assign_appropriate_councillor_position_effect = { NEW_COUNCILLOR = $TARGET$ }
	ftr_court_clear_council_promise_effect = { TARGET = $TARGET$ }
}

# 在任命者域执行：清除一笔内阁席位承诺（被承诺者记账 + 自身欠账计数）。
# 参数：TARGET = 被承诺者
ftr_court_clear_council_promise_effect = {
	$TARGET$ ?= {
		remove_variable = ftr_court_promise_council
		remove_variable = ftr_court_promised_by
	}
	if = {
		limit = { has_variable = ftr_court_owes_council }
		change_variable = { name = ftr_court_owes_council  subtract = 1 }
		if = {
			limit = { var:ftr_court_owes_council <= 0 }
			remove_variable = ftr_court_owes_council
		}
	}
}

# 在新君域执行：若还欠其它人内阁席位，串行排队下一个提醒（随机间隔避免密集弹窗）
ftr_court_queue_next_council_reminder_effect = {

					has_variable = ftr_court_promise_council
					var:ftr_court_promised_by ?= { this = prev }
				}
			}
		}
		trigger_event = {
			id = ftr_court_struggle.0016
			days = { ftr_court_council_queue_delay_min_value  ftr_court_council_queue_delay_max_value }
		}
	}
}

# 在成员角色域执行：把自身加入 root（玩家）域的指定列表。

# 在候选人域执行：把候选人加入玩家域继任者列表，最多保留 5 位。
# 核心政治逻辑：最高统治者的子女优先参与继位角逐（不分继承顺位高低），
# 孙辈顺位虽靠前但非角逐主角——本效果只由 every_child / every_sibling 调用，
# 天然只收集直系血亲，用 root 域计数器截断到最多 5 位。

ftr_court_add_heir_effect = {

# 在君主域执行：收集权臣 + 内阁成员（排除配偶，避免与后宫格重复），
# 参考 ftr_my_powerful_subject_list 的两段式：add_to_list 集中收集 → every_in_list 转存到 root（玩家）域。

ftr_court_collect_ministers_effect = {

# 在君主域执行：收集朝廷核心成员，按类别写入玩家域的独立列表。
# 布局（自上而下）：君主[+摄政] → 配偶/姬妾 → 权臣/内阁 → 继任者 → 追随者。
ftr_court_sync_courtiers_effect = {
	save_scope_as = ftr_court_court_monarch

	root = {
		# 复位标记与列表
		set_variable = { name = ftr_court_ui_has_diarch  value = 0 }
		# 清空共治者角色变量：无共治者时 GUI 常驻空位需读到无效角色，避免残留上一次的旧值
		clear_variable = ftr_court_ui_diarch_char
		clear_variable_list = ftr_court_ui_spouses
		clear_variable_list = ftr_court_ui_ministers
		clear_variable_list = ftr_court_ui_heirs
		clear_variable_list = ftr_court_ui_backers
		# 继任者计数器复位（ftr_court_add_heir_effect 用它截断到 5）
		set_variable = { name = ftr_court_heir_count  value = 0 }
	}

	# ---- 1. 君主（靠上居中，存单角色变量） ----
	save_scope_as = ftr_court_ui_rule
	root = {
		set_variable = {
			name = ftr_court_ui_monarch_char
			value = scope:ftr_court_ui_rule
		}
	}

	# ---- 2. 摄政/共治者（与君主并排居中；无则不显示，存单角色变量） ----
	if = {
		limit = { exists = diarch }
		diarch ?= {
			save_scope_as = ftr_court_ui_diarch_char_scope
		}
		root = {
			set_variable = {
				name = ftr_court_ui_diarch_char
				value = scope:ftr_court_ui_diarch_char_scope
			}
			set_variable = { name = ftr_court_ui_has_diarch  value = 1 }
		}
	}
	else = {
		# 无摄政/共治者：平衡条归零（显示在中点 = 无偏移）
		root = {
			set_variable = { name = ftr_court_ui_swing  value = 0 }
		}
	}

	# ---- 3. 配偶 + 姬妾（打标记供权臣/内阁排除） ----
	every_spouse = {
		add_character_flag = ftr_court_in_spouses
		ftr_court_add_ui_member_effect = { LIST = ftr_court_ui_spouses }
	}
	every_concubine = {
		add_character_flag = ftr_court_in_spouses
		ftr_court_add_ui_member_effect = { LIST = ftr_court_ui_spouses }
	}

	# ---- 4. 权臣 + 内阁成员（排除已列入后宫者，避免重复） ----
	# 收集逻辑集中在 ftr_court_collect_ministers_effect（参考 ftr_my_powerful_subject_list 模式）。
	ftr_court_collect_ministers_effect = yes

	# ---- 5. 继任者：按政体/继承法区分获取逻辑 ----
	# 血缘继承（分割/长子幼子）：优先血亲——有继承权的子女 + 同宗族亲兄弟姐妹。
	#            性别优先级：男性优先时，有 >=1 男性继承人则不添加女性；女性优先颠倒。
	#            有 >=1 成年子女时，不添加兄弟姐妹。
	# 选举/不稳定（选侯/部分头衔选举/欢呼）：直接按头衔继承顺位。
	if = {
		limit = { ftr_court_hereditary_succession_trigger = yes }

		# --- 有继承权的子女，按性别法优先级过滤 ---
		# CK3 性别法：male_only_law / female_only_law 是 title law（has_title_law 判断，
		# 见原版 00_auto_character_triggers.txt:7,13）。male_preference/female_preference/equal
		# 是继承法 gender_law 字段，无独立 realm law 可查。因此：
		#   - male_only_law  → 仅男性
		#   - female_only_law → 仅女性
		#   - 其余（默认 male_preference / equal）→ 先男后女（男性优先，避免女儿挤掉儿子）
		# 计数器控制：优先性别占满 5 位则另一性别不进入。
		if = {
			# 仅男性继承
			limit = {
				primary_title ?= { has_title_law = male_only_law }
			}
			every_child = {
				limit = {
					is_alive = yes
					is_child_of = scope:ftr_court_court_monarch
					is_male = yes
				}
				ftr_court_add_heir_effect = yes
			}
		}
		else_if = {
			# 仅女性继承
			limit = {
				primary_title ?= { has_title_law = female_only_law }
			}
			every_child = {
				limit = {
					is_alive = yes
					is_child_of = scope:ftr_court_court_monarch
					is_female = yes
				}
				ftr_court_add_heir_effect = yes
			}
		}
		else = {
			# 默认男性优先（male_preference / equal）：先男后女
			every_child = {
				limit = {
					is_alive = yes
					is_child_of = scope:ftr_court_court_monarch
					is_male = yes
				}
				ftr_court_add_heir_effect = yes
			}
			every_child = {
				limit = {
					is_alive = yes
					is_child_of = scope:ftr_court_court_monarch
					is_female = yes
				}
				ftr_court_add_heir_effect = yes
			}
		}

		# --- 同宗族亲兄弟姐妹：仅当无任何成年子女时参与 ---
		if = {
			limit = {
				NOT = {
					OR = {
						ftr_court_has_male_heir_trigger = yes
						ftr_court_has_female_heir_trigger = yes
					}
				}
			}
			every_sibling = {
				limit = {
					is_alive = yes
					is_adult = yes
					is_sibling_of = scope:ftr_court_court_monarch
					dynasty ?= {
						this = scope:ftr_court_court_monarch.dynasty
					}
				}
				ftr_court_add_heir_effect = yes
			}
		}
	}
	else = {
		# 选举/不稳定继承：按头衔继承顺位收集 title heir，计数器截断到 5
		primary_title = {
			save_scope_as = ftr_court_heir_title_scope
			every_title_heir = {
				limit = {
					is_alive = yes
				}
				ftr_court_add_heir_effect = yes
			}
		}
	}

	# ---- 6. 每个继任者各自的追随者（存继任者身上的 ftr_court_heir_backers 列表） ----
	# 供 GUI 继任者条目内嵌显示（hbox { 继任者头像 + 该继任者的追随者 + 空白 }）。
	# 追随者 = 朝堂封臣/廷臣中已站队（ftr_court_backer_of）支持某位竞争者者。
	# 仅在有朝堂 story（存在站队）时填充；无 story 则列表为空，GUI 端也以 ftr_court_ui_active 隐藏。
	# 先清空所有相关继任者的追随者列表，再统一填充。
	root = {
		every_in_list = {
			list = ftr_court_ui_heirs
			clear_variable_list = ftr_court_heir_backers
		}
	}
	every_vassal = {
		limit = { has_variable = ftr_court_backer_of }
		var:ftr_court_backer_of ?= {
			clear_variable_list = ftr_court_heir_backers
		}
	}
	every_courtier = {
		limit = { has_variable = ftr_court_backer_of }
		var:ftr_court_backer_of ?= {
			clear_variable_list = ftr_court_heir_backers
		}
	}
	every_vassal = {
		limit = { ftr_court_has_backing_trigger = yes }
		save_scope_as = ftr_court_backer_scope
		var:ftr_court_backer_of ?= {
			add_to_variable_list = {
				name = ftr_court_heir_backers
				target = scope:ftr_court_backer_scope
			}
		}
	}
	every_courtier = {
		limit = { ftr_court_has_backing_trigger = yes }
		save_scope_as = ftr_court_backer_scope
		var:ftr_court_backer_of ?= {
			add_to_variable_list = {
				name = ftr_court_heir_backers
				target = scope:ftr_court_backer_scope
			}
		}
	}

	# ---- 7. 追随者：朝堂封臣中已站队支持某位继任竞争者者（全局列表，供旧版展示） ----
	every_vassal = {
		limit = { ftr_court_has_backing_trigger = yes }
		ftr_court_add_ui_member_effect = { LIST = ftr_court_ui_backers }
	}

	# ---- 清理配偶标记 ----
	every_spouse = {
		remove_character_flag = ftr_court_in_spouses
	}
	every_concubine = {
		remove_character_flag = ftr_court_in_spouses
	}
}

# 在持有朝堂 story 的角色域执行：把 story 变量镜像到 root（玩家）域。

			}
		}
