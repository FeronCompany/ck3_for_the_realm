##############################################################
### 朝堂权力角逐 COURT STRUGGLE
### 挂在独立王国/帝国/霸权的最高统治者（story owner）身上
###
### 议题1 继任之争：竞争者以「竞争力评分」排名，取前五名进入朝堂名单
### 议题2 共治夺权：以 ftr_court_regency_swing 为拉锯轴
###
### 阶段（var:ftr_court_phase）：
###   0 dormant   —— 竞争者不足，静默
###   1 simmering —— 暗中角力
###   2 contested —— 势均力敌，公开对立
###   3 crisis    —— 政变/内战窗口开启
###
### 作用域约定：root / scope:story = 故事本体；owner = 在位君主
##############################################################

ftr_court_struggle = {
	visible = yes

	################## 生命周期 ##################

	on_setup = {
		ftr_court_setup_effect = yes
		# 朝堂权力角逐开始：通知最高统治者。
		story_owner = {
			add_character_flag = { flag = ftr_court_notified }
			trigger_event = { id = ftr_court_struggle.0019 }
		}
	}

	on_owner_death = {
		# 君主驾崩：本轮角逐落幕。
		# 清理统一交给 on_end 处理，此处只结束故事，避免清理两遍。
		# 新君是否重启角逐交由下一年的年度检查判定——刻意避开继承结算的时序风险。
		end_story = yes
	}

	on_end = {
		ftr_court_cleanup_effect = yes
	}

	################## 外观 ##################

	# 危机阶段改用剑图标，其余沿用宫廷政治图标
	icon = {
		trigger = { var:ftr_court_phase >= 3 }
		reference = "gfx/interface/icons/story_cycles/story_icon_conqueror.dds"
	}
	icon = {
		reference = "gfx/interface/icons/story_cycles/story_icon_palace_politics.dds"
	}

	background = {
		reference = "gfx/interface/illustrations/event_scenes/corridor_day.dds"
	}

	################## 面板可视化 ##################

	visualization = {
		# 继任竞争者名单（按竞争力降序，最多五人）
		character_list = {
			variable_name = "ftr_court_participant_list"
			label = "FTR_COURT_CLAIMANTS"
		}

		# 共治拉锯：0 = 完全忠于君主，50 = 平衡，100 = 完全倒向共治者
		tug_of_war_counter = {
			variable_name = "ftr_court_regency_swing"
			min = 0
			max = 100
			label = "FTR_COURT_SWING"
			min_label = "FTR_COURT_SWING_LOYAL"
			max_label = "FTR_COURT_SWING_DIARCH"
		}

		custom_string_key = "FTR_COURT_STRINGS"
	}

	################## 年度维护：刷新候选池、拉锯与阶段 ##################

	effect_group = {
		years = 1
		triggered_effect = {
			trigger = {
				story_owner ?= { ftr_court_struggle_valid_trigger = yes }
			}
			effect = {
				ftr_court_update_candidates_effect = yes
				ftr_court_swing_yearly_effect = yes
			}
		}
		# 政权不再符合条件（被征服、降级、绝嗣）时收场
		fallback = {
			end_story = yes
		}
	}

	################## 参与者刷新（30-60 天随机） ##################

	# 定期移除不符合条件的参与者（失能 / 被剥夺继承权 / 被囚禁 / 死亡），
	# 并让之前未参与且未明确表示不参与的潜在继承人有一次选择的机会。
	effect_group = {
		days = { ftr_court_refresh_interval_min_value  ftr_court_refresh_interval_max_value }
		triggered_effect = {
			trigger = {
				story_owner ?= { ftr_court_struggle_valid_trigger = yes }
			}
			effect = {
				ftr_court_update_candidates_effect = yes
				ftr_court_launch_participation_asks_effect = yes
				ftr_court_refresh_backer_count_effect = yes
			}
		}
		fallback = {
			end_story = yes
		}
	}

	################## 定期同步参与者（最高统治者 → story） ##################

	# 新方案：参与者列表主存储放在最高统治者（story_owner）的个人变量列表 ftr_court_claimants 上。
	# 此处定期从最高统治者列表同步到 story 的 ftr_court_participant_list，
	# 供 story 逻辑与侧栏可视化使用。
	effect_group = {
		months = 1
		triggered_effect = {
			trigger = {
				story_owner ?= { ftr_court_struggle_valid_trigger = yes }
			}
			effect = {
				ftr_court_update_candidates_effect = yes
				ftr_court_refresh_backer_count_effect = yes
			}
		}
		fallback = {
			end_story = yes
		}
	}

	################## 危机阶段脉冲 ##################

	# 注意：trigger_event 的 root 取当前作用域，而事件是 character_event，
	# 因此必须包在 owner 中执行，否则 root 会变成故事本体。
	effect_group = {
		years = 1
		chance = 20
		random_valid = {
			triggered_effect = {
				trigger = { ftr_court_is_crisis_trigger = yes }
				effect = {
					story_owner = { trigger_event = { id = ftr_court_struggle.0005 } }
					# 询问一名未站队的封臣（事件 0014 同时覆盖无地廷臣）
					story_owner = {
						random_vassal = {
							limit = {
								is_capable_adult = yes
								NOT = { has_variable = ftr_court_backer_of }
								NOT = { has_character_flag = ftr_court_asked_stance }
							}
							add_character_flag = { flag = ftr_court_asked_stance  years = 3 }
							trigger_event = { id = ftr_court_struggle.0014  days = 2 }
						}
						random_courtier = {
							limit = {
								is_capable_adult = yes
								NOT = { has_variable = ftr_court_backer_of }
								NOT = { has_character_flag = ftr_court_asked_stance }
							}
							add_character_flag = { flag = ftr_court_asked_stance  years = 3 }
							trigger_event = { id = ftr_court_struggle.0014  days = 2 }
						}
					}
					# 询问竞争力最高的竞争者是否起兵（事件 0015，无地者由此路径开战）
					ordered_in_list = {
						list = ftr_court_participant_list
						order_by = var:ftr_claim_strength
						save_scope_as = ftr_court_ringleader
						scope:ftr_court_ringleader ?= {
							if = {
								limit = { NOT = { has_character_flag = ftr_court_asked_to_rebel } }
								trigger_event = { id = ftr_court_struggle.0015  days = 4 }
							}
						}
					}
					# 废黜昏君窗口：拉锯达到夺位阈值且君主失能时，
					# 向朝堂中最强势力推送夺位抉择（事件 0018）
					if = {
						limit = { ftr_court_can_attempt_usurp_trigger = yes }
						story_owner = {
							every_vassal = {
								limit = {
									is_capable_adult = yes
									NOT = { has_character_flag = ftr_court_asked_usurp }
								}
								add_to_list = ftr_court_usurp_candidates
							}
							every_courtier = {
								limit = {
									is_capable_adult = yes
									NOT = { has_character_flag = ftr_court_asked_usurp }
								}
								add_to_list = ftr_court_usurp_candidates
							}
							set_variable = { name = ftr_court_usurp_pick  value = 0 }
							ordered_in_list = {
								list = ftr_court_usurp_candidates
								order_by = max_military_strength
								save_scope_as = ftr_court_usurper
								root = {
									if = {
										limit = { var:ftr_court_usurp_pick = 0 }
										scope:ftr_court_usurper ?= {
											add_character_flag = { flag = ftr_court_asked_usurp  years = 3 }
											trigger_event = { id = ftr_court_struggle.0018  days = 2 }
										}
									}
									change_variable = { name = ftr_court_usurp_pick  add = 1 }
								}
							}
							remove_variable = ftr_court_usurp_pick
						}
					}
				}
			}
		}
	}

	################## 对立阶段脉冲 ##################

	effect_group = {
		years = 1
		chance = 14
		random_valid = {
			triggered_effect = {
				trigger = {
					NOT = { ftr_court_is_crisis_trigger = yes }
					ftr_court_is_contested_trigger = yes
				}
				effect = {
					story_owner = { trigger_event = { id = ftr_court_struggle.0003 } }
					# 询问一名未站队的封臣或廷臣（0014，无地廷臣由此参与）
					story_owner = {
						random_vassal = {
							limit = {
								is_capable_adult = yes
								NOT = { has_variable = ftr_court_backer_of }
								NOT = { has_character_flag = ftr_court_asked_stance }
							}
							add_character_flag = { flag = ftr_court_asked_stance  years = 3 }
							trigger_event = { id = ftr_court_struggle.0014  days = 2 }
						}
						random_courtier = {
							limit = {
								is_capable_adult = yes
								NOT = { has_variable = ftr_court_backer_of }
								NOT = { has_character_flag = ftr_court_asked_stance }
							}
							add_character_flag = { flag = ftr_court_asked_stance  years = 3 }
							trigger_event = { id = ftr_court_struggle.0014  days = 2 }
						}
					}
				}
			}
		}
	}

	################## 潜伏阶段脉冲（两年一次，最稀疏） ##################

	effect_group = {
		years = 2
		chance = 8
		random_valid = {
			triggered_effect = {
				trigger = {
					NOT = { ftr_court_is_contested_trigger = yes }
					story_owner ?= { ftr_court_has_succession_order_trigger = yes }
				}
				effect = {
					story_owner = { trigger_event = { id = ftr_court_struggle.0001 } }
				}
			}
		}
	}
}
