########## 三、好感实例数值（求值域 = 记功方 / holder）##########
# 从征功净额：求值域 = 记功方（holder）；scope:ftr_pm_subject = 效劳方
ftr_personal_merit_crown_amount_value = {
	value = 0
	if = {
		limit = {
			scope:ftr_pm_subject ?= { has_variable = ftr_personal_merit_crown }
		}
		value = scope:ftr_pm_subject.var:ftr_personal_merit_crown
	}
}

# 拥立功净额：求值域 = 记功方（holder）；scope:ftr_pm_subject = 效劳方
ftr_personal_merit_backing_amount_value = {
	value = 0
	if = {
		limit = {
			scope:ftr_pm_subject ?= { has_variable = ftr_personal_merit_backing }
		}
		value = scope:ftr_pm_subject.var:ftr_personal_merit_backing
	}
}

########## 四、可兑余额 ##########
# 求值域 = 效劳方；scope:ftr_pm_holder = 现任最高领主（兑现方）
ftr_personal_merit_redeemable_value = {
	value = 0
	# 从征账（holder == crown_to）
	if = {
		limit = {
			scope:ftr_pm_holder ?= { always = yes }
			has_variable = ftr_personal_merit_crown_to
			var:ftr_personal_merit_crown_to ?= { this = scope:ftr_pm_holder }
		}
		add = var:ftr_personal_merit_crown
	}
	# 拥立账（holder == backing_to；fold 后应与现任领主不同，此处仍兼容）
	if = {
		limit = {
			scope:ftr_pm_holder ?= { always = yes }
			has_variable = ftr_personal_merit_backing_to
			var:ftr_personal_merit_backing_to ?= { this = scope:ftr_pm_holder }
		}
		add = var:ftr_personal_merit_backing
	}
}

ftr_personal_merit_ui_display_value = {

########## 五、消耗（影子运算辅助，见 consume 效果）##########
# 本步从拥立账可扣额度 = min(backing, remaining)。求值域 = 效劳方；
# ftr_pm_remaining 为 consume 效果设置的临时余量变量。
ftr_personal_merit_backing_take_value = {
	value = 0
	if = {
		limit = {
			has_variable = ftr_personal_merit_backing
			var:ftr_personal_merit_backing >= var:ftr_pm_remaining
		}
		value = var:ftr_pm_remaining
	}
	else_if = {
		limit = {
			has_variable = ftr_personal_merit_backing
			var:ftr_personal_merit_backing > 0
		}
		value = var:ftr_personal_merit_backing
	}
}

# 本步从从征账可扣额度 = min(crown, remaining)。求值域 = 效劳方。
ftr_personal_merit_crown_take_value = {
	value = 0
	if = {
		limit = {
			has_variable = ftr_personal_merit_crown
			var:ftr_personal_merit_crown >= var:ftr_pm_remaining
		}
		value = var:ftr_pm_remaining
	}
	else_if = {
		limit = {
			has_variable = ftr_personal_merit_crown
			var:ftr_personal_merit_crown > 0
		}
		value = var:ftr_personal_merit_crown
	}
}
