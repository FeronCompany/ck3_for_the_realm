
######### Charge ##########
ftr_charge_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	cooldown_against_recipient = { years = 10 }
	desc = ftr_charge_interaction_desc
	ai_targets = {
		ai_recipients = peer_vassals
		max = 2
	}
	ai_frequency = 36
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			liege = scope:recipient.liege
			is_independent_ruler = no
		}
		scope:recipient = {
			is_independent_ruler = no
		}
		scope:actor.liege = {
			has_imprisonment_reason = scope:recipient
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_capable_adult = yes
		}
		scope:recipient = {
			is_imprisoned = no
			is_capable_adult = yes
			is_at_war = no
		}
	}
	on_accept = {
		scope:recipient = {
			add_opinion = {
				modifier = angry_opinion
				target = scope:actor
				opinion = -60
			}
		}
		scope:actor.liege = {
			trigger_event = ftr_pg_interaction_event.0100
		}
		# 双方各自都有背书对象（分属党羽）→ 指控本身即党争摩擦；同最高领主且该领主有朝堂故事时抬升其烈度
		if = {
			limit = {
				scope:actor = { has_variable = ftr_court_backer_of }
				scope:recipient = { has_variable = ftr_court_backer_of }
			}
			scope:actor = {
				ftr_court_realm_intensity_add_effect = {
					OTHER = scope:recipient
					AMOUNT = ftr_court_interaction_clash_intensity_value
				}
			}
		}
	}
	on_decline = {}
	auto_accept = yes
	ai_will_do = {
		base = -20
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = -0.25
		}
		# 党羽政治：优先清除其他党羽的政敌
		modifier = {
			add = 60
			scope:actor = { has_variable = ftr_court_backer_of }
			scope:recipient = { has_variable = ftr_court_backer_of }
			NOT = { scope:actor = { ftr_court_same_camp_trigger = { OTHER = scope:recipient } } }
		}
		# 同一党羽（含本方党首本人）→ 不对自己人下手
		modifier = {
			add = -80
			scope:actor = { ftr_court_same_camp_trigger = { OTHER = scope:recipient } }
		}
	}
}

######### Sow Discord ##########
ftr_sow_discord_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	cooldown = { years = 2 }
	cooldown_against_recipient = { years = 5 }
	desc = ftr_sow_discord_interaction_desc
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = neighboring_rulers
		max = 3
	}
	ai_frequency = 64
	is_shown = {
		scope:actor = {
			OR = {
				intrigue >= decent_skill_rating
				has_trait = deceitful
			}
		}
		OR = {
			scope:actor = { is_vassal_of = scope:recipient }
			scope:recipient = { is_vassal_of = scope:actor }
			scope:actor = {
				is_landed = yes
				is_independent_ruler = yes
				character_is_realm_neighbor = scope:recipient
			}
			scope:actor = {
				liege = scope:recipient.liege
				NOT = { this = scope:recipient.liege }
				is_landed = yes
			}
		}
		scope:recipient = {
			primary_title.tier >= tier_county
			is_landed = yes
			NOT = { this = scope:actor }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_capable_adult = yes
		}
		scope:recipient = {
			is_imprisoned = no
			is_capable_adult = yes
		}
	}
	populate_actor_list = {
		scope:recipient = {
			every_vassal = {
				limit = {
					is_capable_adult = yes
					NOT = { scope:actor = this }
				}
				add_to_list = characters
			}
			every_courtier = {
				limit = {
					is_capable_adult = yes
					NOT = { scope:actor = this }
				}
				add_to_list = characters
			}
			if = {
				limit = { is_independent_ruler = no }
				liege = {
					if = {
						limit = { NOT = { scope:actor = this } }
						add_to_list = characters
					}
					every_vassal = {
						limit = {
							is_capable_adult = yes
							NOT = { scope:actor = this }
						}
						add_to_list = characters
					}
				}
			}
		}
	}
	can_be_picked = {
		NOT = { this = scope:recipient }
		NOT = { has_relation_rival = scope:recipient }
		NOT = { has_relation_nemesis = scope:recipient }
	}

	on_accept = {
		scope:actor = {
			stress_impact = {
				just = minor_stress_impact_gain
				honest = minor_stress_impact_gain
				deceitful = minor_stress_impact_loss
			}
		}
		custom_tooltip = {
			text = ftr_sow_discord_desc_text
		}
		random_list = {
			10 = { # Sow discord success
				modifier = {
					add = scope:actor.intrigue
				}
				scope:recipient = {
					trigger_event = {
						id = ftr_pg_interaction_event.0200
						days = { 15 30 }
					}
				}
				desc = ftr_sow_discord.success
			}
			10 = { # Failed but not discovered
				modifier = {
					add = scope:recipient.intrigue
					scope:recipient.intrigue < high_skill_rating
				}
				scope:recipient = {
					trigger_event = {
						id = ftr_pg_interaction_event.0201
						days = { 15 30 }
					}
				}
				desc = ftr_sow_discord.failed
			}
			0 = { # Failed and discovered
				modifier = {
					add = scope:recipient.intrigue
					scope:recipient.intrigue >= high_skill_rating
				}
				scope:recipient = {
					trigger_event = {
						id = ftr_pg_interaction_event.0203
						days = { 15 30 }
					}
				}
				desc = ftr_sow_discord.discovered
			}
		}
		# 双方各自都有背书对象（分属党羽）→ 散布不和本身即党争摩擦；同最高领主且该领主有朝堂故事时抬升其烈度
		if = {
			limit = {
				scope:actor = { has_variable = ftr_court_backer_of }
				scope:recipient = { has_variable = ftr_court_backer_of }
			}
			scope:actor = {
				ftr_court_realm_intensity_add_effect = {
					OTHER = scope:recipient
					AMOUNT = ftr_court_interaction_clash_intensity_value
				}
			}
		}
	}
	on_decline = {}
	auto_accept = yes
	ai_will_do = {
		base = 0
		opinion_modifier = {
			opinion_target = scope:recipient
			who = scope:actor
			multiplier = -0.25
		}
		opinion_modifier = {
			opinion_target = scope:secondary_actor
			who = scope:actor
			multiplier = -0.15
		}
		modifier = {
			add = -50
			OR = {
				has_trait = just
				has_trait = honest
				has_trait = content
			}
		}
		modifier = {
			add = 20
			has_trait = deceitful
		}
		modifier = {
			add = 20
			scope:actor = {
				opinion = {
					target = scope:recipient
					value < 30
				}
				is_vassal_of = scope:secondary_actor
			}
			scope:recipient = {
				is_vassal_of = scope:secondary_actor
			}
		}
		# 党羽政治：打击其他党羽（目标非我方党羽）→ 加权
		modifier = {
			add = 50
			scope:actor = { has_variable = ftr_court_backer_of }
			scope:recipient = { has_variable = ftr_court_backer_of }
			NOT = { scope:actor = { ftr_court_same_camp_trigger = { OTHER = scope:recipient } } }
		}
		# 在其他党羽之间散谣：被打散的两方同属一个外营 → 再加权
		modifier = {
			add = 50
			scope:recipient = { ftr_court_same_camp_trigger = { OTHER = scope:secondary_actor } }
			NOT = { scope:actor = { ftr_court_same_camp_trigger = { OTHER = scope:recipient } } }
		}
		# 党羽内不内斗：目标属我方党羽 → 减权
		modifier = {
			add = -50
			scope:actor = { ftr_court_same_camp_trigger = { OTHER = scope:recipient } }
		}
		# 被离间的两方都是我的人（纯内斗）→ 再减权
		modifier = {
			add = -70
			scope:actor = { ftr_court_same_camp_trigger = { OTHER = scope:recipient } }
			scope:actor = { ftr_court_same_camp_trigger = { OTHER = scope:secondary_actor } }
		}
	}
}
