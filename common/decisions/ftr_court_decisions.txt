### 朝堂权力角逐 COURT STRUGGLE — 决议层
### 均以最高统治者为触发入口。

######### 立储：公开确立继承人，压制其余竞争者 #########
ftr_court_nominate_heir_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = realm
	desc = ftr_court_nominate_heir_decision_desc
	selection_tooltip = ftr_court_nominate_heir_decision_tooltip
	cooldown = { years = 5 }

	is_shown = {
		ftr_court_is_struggle_monarch_trigger = yes
		any_owned_story = { type = ftr_court_struggle }
	}
	is_valid = {
		any_owned_story = {
			type = ftr_court_struggle
			ftr_court_has_enough_candidates_trigger = yes
		}
	}

	cost = {
		prestige = { value = ftr_court_nominate_heir_cost_value }
	}

	# 由事件列出候选人供挑选
	effect = {
		trigger_event = ftr_court_struggle.0030
	}

	ai_check_interval = 365
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = -30
		modifier = {
			add = 50
			any_owned_story = {
				type = ftr_court_struggle
				ftr_court_is_contested_trigger = yes
			}
		}
		modifier = {
			add = 30
			has_trait = impatient
		}
		modifier = {
			add = -40
			has_trait = patient
		}
	}
}

######### 整肃朝纲：巩固自身权威，把共治拉锯拉回自己一侧 #########
ftr_court_consolidate_authority_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = realm
	desc = ftr_court_consolidate_authority_decision_desc
	selection_tooltip = ftr_court_consolidate_authority_decision_tooltip
	cooldown = { years = 2 }

	is_shown = {
		ftr_court_is_struggle_monarch_trigger = yes
		any_owned_story = {
			type = ftr_court_struggle
			var:ftr_court_regency_swing > 0
		}
	}
	is_valid = {
		NOT = { ftr_court_monarch_infirm_trigger = yes }
	}

	effect = {
		save_scope_as = ftr_court_monarch
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			save_scope_as = ftr_court_story
		}
		if = {
			limit = { exists = scope:ftr_court_story }
			scope:ftr_court_story = {
				set_variable = {
					name = ftr_court_swing_amount
					value = ftr_court_swing_support_monarch_value
				}
				ftr_court_shift_swing_effect = yes
				ftr_court_swing_yearly_effect = yes
				remove_variable = ftr_court_swing_amount
			}
		}
		add_prestige = ftr_court_prestige_minor_value
	}

	ai_check_interval = 180
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = -10
		modifier = {
			add = 60
			any_owned_story = {
				type = ftr_court_struggle
				ftr_court_can_petition_coregency_trigger = yes
			}
		}
		modifier = {
			add = 25
			has_trait = diligent
		}
	}
}
