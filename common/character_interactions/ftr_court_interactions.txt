########## 定位所属朝堂（供各交互复用） ##########

### 议题 1：继任之争
############################################################################

######### 支持某权贵 / 君主的子女与继承人（公开或秘密） #########
# 支持对象：权贵（君主 / 共治者 / 参与角逐者）或君主的子女 / 继承人（提前押注，§7.3）。
# 公开分支 = 表态支持：朝堂通报、其余参与者记恨、小概率君主警觉 0041；对象为继承人时另授予背书声望。
# 秘密分支 = 只记朝堂支持账（backer_of / supporters），不通报、不树敌、不触发警觉。
# 原 ftr_support_heir_interaction（支持继承人）已并入本交互并废弃（§7.3）。
# 若此前已支持他人，改投新主会严重破坏与旧主及其支持者的关系（公开/秘密皆然）。
ftr_court_back_claimant_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_back_claimant_interaction_desc
	cooldown_against_recipient = { years = 5 }
	options_heading = ftr_back_claimant_options_tt
	send_option = { # Support in secret
		flag = ftr_backing_in_secret
		localization = ftr_support_in_secret_tt
	}
	send_option = { # Support in public
		flag = ftr_backing_in_public
		localization = ftr_support_in_public_tt
		starts_enabled = { always = yes }
	}
	send_options_exclusive = yes
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		ai_recipients = dynasty
		ai_recipients = liege
		ai_recipients = vassals
		ai_recipients = realm_characters
		max = 5
	}
	ai_frequency = 12
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		# 支持对象：权贵（君主 / 共治者 / 参与角逐的继任者）或君主的子女 / 继承人
		OR = {
			scope:recipient = { ftr_court_is_supported_person_trigger = yes }
			scope:recipient = {
				OR = {
					is_child_of = scope:actor.top_liege
					is_heir_of = scope:actor.top_liege
				}
			}
		}
		scope:actor = {
			# 无地廷臣无法主动发起交互，参与一律走事件（ftr_court_struggle.0014）
			is_landed = yes
			NOT = { ftr_court_is_struggle_monarch_trigger = yes }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_back_claimant_prestige_cost_value
				INFLUENCE = ftr_court_back_claimant_influence_cost_value
			}
		}
		scope:recipient = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			# 资格过滤（承接原 support_heir 语义）
			NOT = { has_trait = disinherited }
			NOT = { has_trait = disputed_heritage }
			NOT = { has_trait = denounced }
		}
	}
	can_send = {}
	on_accept = {
		# ---- 若已支持他人，改投新主：从旧主支持者列表移除自己，并与旧主及其支持者关系严重恶化 ----
		scope:actor = {
			if = {
				limit = {
					has_variable = ftr_court_backer_of
					NOT = { var:ftr_court_backer_of ?= { this = scope:recipient } }
				}
				save_scope_as = ftr_court_old_backed
				# 从旧主的支持者列表移除自己（新方案：支持者列表存于被支持者身上）
				scope:ftr_court_old_backed = {
					ftr_court_remove_backer_effect = { BACKER = scope:actor }
				}
				# 旧主记恨
				var:ftr_court_backer_of ?= {
					add_opinion = { target = scope:actor  modifier = ftr_court_betrayed_opinion }
				}
				# 旧主的支持者们（除了自己）也会敌视改投者
				scope:ftr_court_old_backed = {
					every_vassal = {
						limit = {
							has_variable = ftr_court_backer_of
							var:ftr_court_backer_of ?= { this = scope:ftr_court_old_backed }
							NOT = { this = scope:actor }
						}
						add_opinion = { target = scope:actor  modifier = ftr_court_rival_claimant_opinion }
					}
					every_courtier = {
						limit = {
							has_variable = ftr_court_backer_of
							var:ftr_court_backer_of ?= { this = scope:ftr_court_old_backed }
							NOT = { this = scope:actor }
						}
						add_opinion = { target = scope:actor  modifier = ftr_court_rival_claimant_opinion }
					}
				}
			}
			# 更新自己的被支持者变量
			set_variable = { name = ftr_court_backer_of  value = scope:recipient }
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_back_claimant_prestige_cost_value
				INFLUENCE = ftr_court_back_claimant_influence_cost_value
			}
			add_opinion = { target = scope:recipient  modifier = ftr_court_backer_opinion }
		}
		# 把 actor 加入 recipient（被支持权贵）的个人支持者列表（公开/秘密共用，记账唯一）
		scope:recipient = {
			add_to_variable_list = {
				name = ftr_court_supporters
				target = scope:actor
			}
			add_opinion = { target = scope:actor  modifier = ftr_political_alliance_opinion }
		}

		# ---- 公开分支：通报 + 树敌 + 背书声望 + 君主警觉 ----
		if = {
			limit = { NOT = { scope:ftr_backing_in_secret = yes } }
			# 公开支持君主的子女 / 继承人：授予背书声望（原 support_heir 公开效果）
			if = {
				limit = {
					scope:recipient ?= {
						OR = {
							is_child_of = scope:actor.top_liege
							is_heir_of = scope:actor.top_liege
						}
					}
				}
				scope:recipient = {
					add_prestige = ftr_court_heir_backing_prestige_value
				}
			}
			# 站队公开，其余竞争者同生芥蒂
			scope:actor.top_liege ?= {
				every_owned_story = {
					limit = { story_type = ftr_court_struggle }
					save_scope_as = ftr_court_story
				}
			}
			if = {
				limit = { exists = scope:ftr_court_story }
				scope:ftr_court_story = {
					every_in_list = {
						list = ftr_court_participant_list
						limit = { NOT = { this = scope:recipient } }
						add_opinion = { target = scope:actor  modifier = ftr_court_rival_claimant_opinion }
					}
					# 朝堂通报：公开表态
					owner = {
						save_scope_as = ftr_court_monarch
						ftr_court_send_court_toast_effect = {
							TITLE = ftr_court_notice_backing
							ICON = scope:actor
						}
					}
					# ### DEBUG ### 通知玩家「谁支持了谁」（后续统一移除）
					scope:recipient = { save_scope_as = ftr_court_debug_backed }
					ftr_court_debug_backing_notice_effect = yes
				}
			}

			# ---- 公开支持小概率触发君主警觉事件，恶化君主与被支持者的关系 ----
			if = {
				limit = {
					NOT = { scope:recipient = scope:actor }
					# 被支持者不是君主本人（支持君主不会让君主"警觉"）
					scope:recipient = {
						NOT = { ftr_court_is_struggle_monarch_trigger = yes }
					}
				}
				# 概率触发（random + chance 作为概率门）
				random = {
					chance = ftr_court_monarch_alert_chance_value
					# 保存被支持者与背书者供警觉事件使用
					scope:recipient = {
						save_scope_as = ftr_court_supported
					}
					scope:actor = { save_scope_as = ftr_court_backer }
					scope:actor.top_liege ?= {
						save_scope_as = ftr_court_monarch
					}
					scope:ftr_court_monarch ?= {
						trigger_event = {
							id = ftr_court_struggle.0041
							days = 1
						}
					}
				}
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = -20
		modifier = {
			add = 30
			scope:recipient = { has_trait = ambitious }
		}
		modifier = {
			add = 20
			opinion = { target = scope:recipient  value >= 50 }
		}
		modifier = {
			add = -30
			scope:actor = { has_trait = content }
		}
		# 自身是继承人时不轻易为他人公开背书（原 support_heir 语义）
		modifier = {
			add = -200
			scope:actor = { is_heir_of = top_liege }
		}
		# 与对象结盟者更愿支持
		modifier = {
			add = 50
			scope:actor = { is_allied_to = scope:recipient }
		}
		# 主公有共治者时，朝堂拉锯激烈，更倾向押注
		modifier = {
			add = 20
			scope:actor.top_liege = { has_active_diarchy = yes }
		}
		# 曾被君主警告公开背书者收敛（转入秘密或停手）
		modifier = {
			add = -40
			scope:actor = { has_character_flag = ftr_liege_warn_not_support_heir }
		}
		# 选项感知：多疑者畏公开树敌，但仍可秘密押注（按拟采取的方案核算）
		modifier = {
			add = -10
			scope:actor = { has_trait = paranoid }
			scope:ftr_backing_in_public = yes
		}
		# 走秘密支持：无树敌与警觉风险，谨慎者亦无顾虑
		modifier = {
			add = 15
			scope:ftr_backing_in_secret = yes
			scope:actor = {
				OR = {
					has_trait = paranoid
					has_trait = shy
					has_trait = craven
				}
			}
		}
		# 拟公开支持却曾被君主警告：大幅却步
		modifier = {
			add = -20
			scope:ftr_backing_in_public = yes
			scope:actor = { has_character_flag = ftr_liege_warn_not_support_heir }
		}
	}

	ai_accept = {
		# 获得一份支持或押注总体有益，但公开拥立也会招风树敌
		base = 20
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
		}
		# 支持对象是君主本人：忠君表态，天然欢迎
		modifier = {
			desc = ftr_court_accept_back_claimant_monarch
			add = 30
			scope:recipient = { ftr_court_is_struggle_monarch_trigger = yes }
		}
		# 朝堂存在共治角力时，每份支持都更有分量
		modifier = {
			desc = ftr_court_accept_back_claimant_diarchy
			add = 20
			scope:recipient = {
				ftr_court_is_struggle_monarch_trigger = yes
				has_active_diarchy = yes
			}
		}
		# 野心勃勃者渴求每一份助力
		modifier = {
			desc = ftr_court_accept_back_claimant_ambitious
			add = 15
			scope:recipient = { has_trait = ambitious }
		}
		# 自命不凡者乐于受拥戴声张声势
		modifier = {
			desc = ftr_court_accept_back_claimant_arrogant
			add = 15
			scope:recipient = { has_trait = arrogant }
		}
		# 勇毅者不惧公开站队引来嫉恨
		modifier = {
			desc = ftr_court_accept_back_claimant_brave
			add = 15
			scope:recipient = { has_trait = brave }
		}
		# 多疑者忌惮公开拥立使自己成为靶子
		modifier = {
			desc = ftr_court_accept_back_claimant_paranoid
			add = -25
			scope:recipient = { has_trait = paranoid }
		}
		# 怯懦者不愿卷入公开争斗
		modifier = {
			desc = ftr_court_accept_back_claimant_craven
			add = -15
			scope:recipient = { has_trait = craven }
		}
		# 知足者不喜节外生枝
		modifier = {
			desc = ftr_court_accept_back_claimant_content
			add = -10
			scope:recipient = { has_trait = content }
		}
		# 选项感知（对方按所持「秘密/公开」意图评估）：
		# 秘密支持免于朝堂侧目，受者更安心
		modifier = {
			desc = ftr_court_accept_back_claimant_secret
			add = 15
			scope:ftr_backing_in_secret = yes
		}
		# 公开拥立声势正合野心勃勃者的胃口
		modifier = {
			desc = ftr_court_accept_back_claimant_public
			add = 15
			scope:ftr_backing_in_public = yes
			scope:recipient = { has_trait = ambitious }
		}
	}
}

######### 撤回支持，重归中立 #########
ftr_court_withdraw_backing_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_withdraw_backing_interaction_desc
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = close_family
		max = 1
	}
	ai_frequency = 24
	is_shown = {
		scope:actor = {
			# 无地廷臣无法主动发起交互（站队走事件 0014，撤回暂无事件路径）
			is_landed = yes
			has_variable = ftr_court_backer_of
			var:ftr_court_backer_of ?= { this = scope:recipient }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			remove_variable = ftr_court_backer_of
			add_opinion = { target = scope:recipient  modifier = ftr_political_opponent_opinion }
		}
		# 从被支持者的个人支持者列表移除自己（新方案：支持者列表存于被支持者身上）
		scope:recipient = {
			ftr_court_remove_backer_effect = { BACKER = scope:actor }
			add_opinion = { target = scope:actor  modifier = ftr_political_opponent_opinion }
		}
	}
	on_decline = {}
	ai_will_do = {
		base = -30
		modifier = {
			add = 40
			opinion = { target = scope:recipient  value < 0 }
		}
		# 多疑者风声不对先抽身自保，不愿与人捆绑过深
		modifier = {
			add = 15
			scope:actor = { has_trait = paranoid }
		}
	}

	ai_accept = {
		# 强留一个离心的人弊大于利：总体放行，仅在惜才/急需实力时挽留
		base = 50
		# 被支持者越器重此人越惜才，越不愿放人（好感高 → 接受度降）
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = -0.5
		}
		# 竞逐正酣、局面胶着，缺不得爪牙
		modifier = {
			desc = ftr_court_accept_withdraw_struggle
			add = -25
			scope:recipient = {
				any_owned_story = {
					story_type = ftr_court_struggle
					has_variable = ftr_court_struggle_intensity
					var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value
				}
			}
		}
		# 野心勃勃者绝不轻易让麾下力量流失
		modifier = {
			desc = ftr_court_accept_withdraw_ambitious
			add = -20
			scope:recipient = { has_trait = ambitious }
		}
		# 刚愎自用者拉不下脸，宁可挽留也不愿见人转投
		modifier = {
			desc = ftr_court_accept_withdraw_arrogant
			add = -15
			scope:recipient = { has_trait = arrogant }
		}
		# 多疑者怕强留之人怀恨反噬，宁愿好聚好散
		modifier = {
			desc = ftr_court_accept_withdraw_paranoid
			add = 15
			scope:recipient = { has_trait = paranoid }
		}
		# 随遇而安者不强留人
		modifier = {
			desc = ftr_court_accept_withdraw_content
			add = 10
			scope:recipient = { has_trait = content }
		}
	}
}

######### 辅政：尽心王事，防止大权旁落 #########
ftr_court_assist_monarch_interaction = {
	category = interaction_category_vassal
	common_interaction = yes
	desc = ftr_court_assist_monarch_interaction_desc
	notification_text = ftr_court_assist_monarch_interaction_notification
	greeting = positive
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = liege
	}
	ai_frequency = 36
	is_shown = {
		scope:actor = {
			OR = {
				is_councillor_of = scope:recipient
				is_powerful_vassal_of = scope:recipient
			}
		}
		scope:recipient = {
			ftr_court_is_struggle_monarch_trigger = yes
			ftr_court_monarch_infirm_trigger = yes
			# 最高统治者必须存在共治者（辅政针对君主与共治者的权力拉锯）
			has_active_diarchy = yes
			# 发起者不能是共治者本人（共治者不向自己辅政）
			NOT = { diarch = scope:actor }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			NOT = { has_character_flag = ftr_court_assisted_this_year }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_assist_monarch_prestige_cost_value
				INFLUENCE = ftr_court_assist_monarch_influence_cost_value
			}
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_assist_monarch_prestige_cost_value
				INFLUENCE = ftr_court_assist_monarch_influence_cost_value
			}
			add_character_flag = { flag = ftr_court_assisted_this_year  years = 1 }
		}
		# 拉锯向君主一侧移动
		scope:recipient = {
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				save_scope_as = ftr_court_story
			}
			add_opinion = { target = scope:actor  modifier = ftr_court_loyalist_opinion }
			update_diarchy_swing_with_perspective_effect = {
				SWING = minor_sop_swing_liege_gain
				LIEGE = scope:recipient
			}
		}
		if = {
			limit = { exists = scope:ftr_court_story }
			scope:ftr_court_story = {
				set_variable = {
					name = ftr_court_swing_amount
					value = ftr_court_swing_support_monarch_value
				}
				ftr_court_shift_swing_effect = yes
				remove_variable = ftr_court_swing_amount
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = 0
		# 与最高统治者的关系（意见修正值）：君主越器重发起者，发起者越愿辅政
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		# 与共治者的关系（意见修正值）：共治者越器重发起者，发起者越不愿辅政打压共治者
		# 负权重稍大：并非每个臣子都愿冒风险做忠臣
		opinion_modifier = {
			who = scope:recipient.diarch
			opinion_target = scope:actor
			multiplier = -1.5
			desc = AI_OPINION_REASON
		}
		# 有忠信特质：天性忠贞，愿辅佐君主
		modifier = {
			add = 40
			scope:actor = { has_trait = loyal }
		}
		# 与共治者是仇敌：借辅政打压共治者
		modifier = {
			add = 40
			scope:actor = { has_relation_rival = scope:recipient.diarch }
		}

				if = {
					limit = { scope:actor = { has_character_flag = lobby_diarch } }
					# 劝进：游说对象被拉向共治阵营
					set_variable = {
						name = ftr_court_swing_amount
						value = ftr_court_swing_support_diarch_value
					}
				}
				else = {
					# 劝忠：游说对象被拉回君主阵营
					set_variable = {
						name = ftr_court_swing_amount
						value = ftr_court_swing_support_monarch_value
					}
				}
				ftr_court_shift_swing_effect = yes
				remove_variable = ftr_court_swing_amount
			}
		}

		# 人情记账
		if = {
			limit = { scope:actor = { has_character_flag = lobby_diarch } }
			scope:recipient = {
				add_opinion = { target = scope:actor  modifier = ftr_political_alliance_opinion }
			}
		}
		else = {
			scope:actor.top_liege ?= {
				save_scope_as = ftr_court_monarch
			}
			scope:recipient = {
				add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_court_loyalist_opinion }
			}
		}
	}
	on_decline = {}

		modifier = {

		modifier = {

		# 多疑者忌惮高调拉人引人侧目，亦难辨游说对象的真心
		modifier = {
			add = -15
			scope:actor = { has_trait = paranoid }
		}
		# —— 选项感知（按 AI 拟采用的劝进/劝忠方案）——
		# 拟劝进：共治者本人，或已拥立共治者者
		modifier = {
			add = 15
			scope:lobby_diarch = yes
			scope:actor = {
				OR = {
					this = scope:actor.top_liege.diarch
					AND = {
						has_variable = ftr_court_backer_of
						var:ftr_court_backer_of ?= { this = scope:actor.top_liege.diarch }
					}
				}
			}
		}
		# 拟劝忠：与君主相善者更愿为君张目
		modifier = {
			add = 10
			scope:lobby_monarch = yes
			scope:actor = { opinion = { target = scope:actor.top_liege  value > 0 } }
		}
	}
