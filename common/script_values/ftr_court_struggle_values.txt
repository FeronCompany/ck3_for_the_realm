########## 四、竞争力评分总分（实时综合判断） ##########
# 求值时的作用域：this = 候选人，scope:ftr_court_monarch = 在位君主
# 1.4 起不再依赖存储变量 ftr_claim_strength；排序/选头目处直接按本实时值评估，
# 每次求值均按候选人当下的头衔等级、兵力、威望等级、影响力等综合浮动。
ftr_court_claim_strength_value = {
	value = 0

	### 成年与年龄阶段
	if = {
		limit = { is_adult = yes }
		add = ftr_court_claim_adult_value
		if = {
			limit = { age >= 16  age <= 45 }
			add = ftr_court_claim_prime_age_value
		}
		else_if = {
			limit = { age >= ftr_court_monarch_elderly_age_value }
			add = ftr_court_claim_old_age_value
		}
	}

	### 个人能力
	add = ftr_court_claim_average_attribute_value

	### 特质
	if = { limit = { has_trait = ambitious }		add = ftr_court_claim_ambitious_value }
	if = { limit = { has_trait = brave }			add = ftr_court_claim_brave_value }
	if = { limit = { has_trait = intellect_good_3 }	add = ftr_court_claim_genius_value }
	if = { limit = { has_trait = content }			add = ftr_court_claim_content_value }
	if = { limit = { has_trait = craven }			add = ftr_court_claim_craven_value }
	if = { limit = { has_trait = humble }			add = ftr_court_claim_humble_value }
	if = { limit = { has_trait = incapable }		add = ftr_court_claim_incapable_value }
	if = { limit = { has_trait = lunatic }			add = ftr_court_claim_lunatic_value }
	if = { limit = { has_trait = inbred }			add = ftr_court_claim_inbred_value }
	if = { limit = { has_trait = drunkard }			add = ftr_court_claim_drunkard_value }
	if = { limit = { has_trait = reclusive }		add = ftr_court_claim_reclusive_value }
	if = { limit = { has_trait = shy }				add = ftr_court_claim_shy_value }

	### 实时硬实力：头衔等级（无地皇子不拿此档，仍可靠继承/亲缘/支持者得分）
	if = {
		limit = { highest_held_title_tier >= tier_empire }
		add = ftr_court_claim_emperor_tier_value
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		add = ftr_court_claim_kingdom_tier_value
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_duchy }
		add = ftr_court_claim_duchy_tier_value
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_county }
		add = ftr_court_claim_county_tier_value
	}

	### 实时硬实力：兵力（取最大军力，系数缩放避免淹没其它维度）
	add = {
		value = max_military_strength
		multiply = ftr_court_claim_military_factor_value
	}

	### 实时硬实力：威望等级
	add = {
		value = prestige_level
		multiply = ftr_court_claim_prestige_factor_value
	}

	### 实时硬实力：影响力（仅行政制角色具备）
	if = {
		limit = { government_allows = administrative }
		add = {
			value = influence
			multiply = ftr_court_claim_influence_factor_value
		}
	}

	### 身份地位
	if = { limit = { is_councillor_of = scope:ftr_court_monarch }		add = ftr_court_claim_councillor_value }
	if = { limit = { is_diarch_of_target = scope:ftr_court_monarch }	add = ftr_court_claim_diarch_value }
	if = { limit = { is_spouse_of = scope:ftr_court_monarch }			add = ftr_court_claim_spouse_value }
	if = { limit = { is_heir_of = scope:ftr_court_monarch }				add = ftr_court_claim_primary_heir_value }

	### 人望：君主好感（分段判定，避免在数值上下文直接引用 opinion）
	if = {
		limit = { opinion = { target = scope:ftr_court_monarch  value >= 50 } }
		add = ftr_court_claim_opinion_high_value
	}
	else_if = {
		limit = { opinion = { target = scope:ftr_court_monarch  value >= 0 } }
		add = ftr_court_claim_opinion_mid_value
	}
	else = {
		add = ftr_court_claim_opinion_low_value
	}

	### 支持者数量（支持者列表直接维护在权贵身上；variable list 数值计数引用 ftr_court_supporters_count_value）
	add = {
		value = ftr_court_claim_backer_value
		multiply = ftr_court_supporters_count_value
	}

	### 防负分，便于展示（用 min 而非 max：低于 0 抬回 0）
	min = 0
}

########## 四·B、列表人数（查询用） ##########
# 重要：以下 court 名单均为 add_to_variable_list 维护的 variable list。
# 取大小有两种合法姿势：
#   1) trigger/limit 里判断：variable_list_size = { name = 名单 value … }（原版 tests 权威）；
#   2) 需要数值（set_variable / multiply / tooltip 文本）：不能裸用 list_size:（那是 add_to_list 临时 list 的语法），
#      必须 has_variable_list 保护后用 every_in_list variable= 遍历累加（对齐原版 script value 计数范式）。
# 下面三个 count value 即按姿势 2 实现，供各调用点引用。

# 求值域：this = 某权贵（君主 / 共治者 / 参与角逐者）。数其 ftr_court_supporters 支持者。
# 调试决议 ftr_court_debug_report_claimants_decision 在 custom_tooltip 文本中引用本值，逐行显示各竞争者支持者数。
ftr_court_supporters_count_value = {
	value = 0
	if = {
		limit = { has_variable_list = ftr_court_supporters }
		every_in_list = {
			variable = ftr_court_supporters
			add = 1
		}
	}
}

# 求值域：this = 拟背书者（actor），scope:recipient = 被支持的权贵。
# AI 主动背书交互 ftr_court_back_claimant_interaction 的 ai_will_do 引用本值：
# 逐名遍历目标既有支持者名单，每名扣减一步长（首名豁免），线性压制"锦上添花"式背书。
ftr_court_backing_crowd_penalty_value = {
	value = 0
	scope:recipient = {
		if = {
			limit = { has_variable_list = ftr_court_supporters }
			every_in_list = {
				variable = ftr_court_supporters
				subtract = ftr_court_backing_crowd_penalty_step_value
			}
			# 豁免第一名：对象已有 1 名支持者时仍可正常押注
			add = ftr_court_backing_crowd_penalty_step_value
		}
	}
}
