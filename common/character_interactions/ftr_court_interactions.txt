######### 公开支持某位权贵（最高统治者 / 共治者 / 参与角逐的继任者） #########
# 若此前已支持他人，改投新主会严重破坏与旧主及其支持者的关系（原 defect 效果并入此处）。
ftr_court_back_claimant_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_back_claimant_interaction_desc
	cooldown_against_recipient = { years = 5 }
	ai_targets = {
		ai_recipients = close_family
		ai_recipients = courtiers
		max = 2
	}
	ai_frequency = 48
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		# 支持对象必须是权贵（君主 / 共治者 / 参与角逐的继任者）
		scope:recipient = { ftr_court_is_supported_person_trigger = yes }
		scope:actor = {
			# 无地廷臣无法主动发起交互，参与一律走事件（ftr_court_struggle.0014）
			is_landed = yes
			NOT = { ftr_court_is_struggle_monarch_trigger = yes }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_back_claimant_prestige_cost_value
				INFLUENCE = ftr_court_back_claimant_influence_cost_value
			}
		}
		scope:recipient = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		# ---- 若已支持他人，改投新主：与旧主及其支持者关系严重恶化 ----
		scope:actor = {
			if = {
				limit = {
					has_variable = ftr_court_backer_of
					NOT = { var:ftr_court_backer_of ?= { this = scope:recipient } }
				}
				save_scope_as = ftr_court_old_backed
				# 旧主记恨
				var:ftr_court_backer_of ?= {
					add_opinion = { target = scope:actor  modifier = ftr_court_betrayed_opinion }
				}
				# 旧主的支持者们（除了自己）也会敌视改投者
				scope:ftr_court_old_backed = {
					every_vassal = {
						limit = {
							has_variable = ftr_court_backer_of
							var:ftr_court_backer_of ?= { this = scope:ftr_court_old_backed }
							NOT = { this = scope:actor }
						}
						add_opinion = { target = scope:actor  modifier = ftr_court_rival_claimant_opinion }
					}
					every_courtier = {
						limit = {
							has_variable = ftr_court_backer_of
							var:ftr_court_backer_of ?= { this = scope:ftr_court_old_backed }
							NOT = { this = scope:actor }
						}
						add_opinion = { target = scope:actor  modifier = ftr_court_rival_claimant_opinion }
					}
				}
			}
			# 更新自己的被支持者变量
			set_variable = { name = ftr_court_backer_of  value = scope:recipient }
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_back_claimant_prestige_cost_value
				INFLUENCE = ftr_court_back_claimant_influence_cost_value
			}
			add_opinion = { target = scope:recipient  modifier = ftr_court_backer_opinion }
		}
		scope:recipient = {
			add_opinion = { target = scope:actor  modifier = ftr_political_alliance_opinion }
		}

		# 站队公开，其余竞争者同生芥蒂
		scope:actor.top_liege ?= {
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				save_scope_as = ftr_court_story
			}
		}
		if = {
			limit = { exists = scope:ftr_court_story }
			scope:ftr_court_story = {
				every_in_list = {
					list = ftr_court_participant_list
					limit = { NOT = { this = scope:recipient } }
					add_opinion = { target = scope:actor  modifier = ftr_court_rival_claimant_opinion }
				}
				ftr_court_refresh_backer_count_effect = yes
				# 朝堂通报：公开表态
				owner = {
					save_scope_as = ftr_court_monarch
					ftr_court_send_court_toast_effect = {
						TITLE = ftr_court_notice_backing
						ICON = scope:actor
					}
				}
			}
		}

		# ---- 公开支持有概率触发君主警觉事件，恶化君主与被支持者的关系 ----
		if = {
			limit = {
				NOT = { scope:recipient = scope:actor }
			}
			# 被支持者不是君主本人（支持君主不会让君主"警觉"）
			scope:recipient = {
				NOT = { ftr_court_is_struggle_monarch_trigger = yes }
			}
			# 概率触发（random + chance 作为概率门）
			random = {
				chance = ftr_court_monarch_alert_chance_value
				# 保存被支持者供警觉事件使用
				scope:recipient = {
					save_scope_as = ftr_court_supported
				}
				scope:actor.top_liege ?= {
					save_scope_as = ftr_court_monarch
				}
				scope:ftr_court_monarch ?= {
					trigger_event = {
						id = ftr_court_struggle.0041
						days = 1
					}
				}
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = 0
		modifier = {
			add = 30
			scope:recipient = { has_trait = ambitious }
		}
		modifier = {
			add = 20
			opinion = { target = scope:recipient  value >= 50 }
		}
		modifier = {
			add = -30
			scope:actor = { has_trait = content }
		}
	}
}

######### 辅政：尽心王事，防止大权旁落 #########
ftr_court_assist_monarch_interaction = {
	category = interaction_category_vassal
	common_interaction = yes
	desc = ftr_court_assist_monarch_interaction_desc
	notification_text = ftr_court_assist_monarch_interaction_notification
	greeting = positive
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = liege
	}
	ai_frequency = 36
	is_shown = {
		scope:actor = {
			OR = {
				is_councillor_of = scope:recipient
				is_powerful_vassal_of = scope:recipient
			}
		}
		scope:recipient = {
			ftr_court_is_struggle_monarch_trigger = yes
			ftr_court_monarch_infirm_trigger = yes
			# 最高统治者必须存在共治者（辅政针对君主与共治者的权力拉锯）
			has_active_diarchy = yes
			# 发起者不能是共治者本人（共治者不向自己辅政）
			NOT = { diarch = scope:actor }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			NOT = { has_character_flag = ftr_court_assisted_this_year }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_assist_monarch_prestige_cost_value
				INFLUENCE = ftr_court_assist_monarch_influence_cost_value
			}
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_assist_monarch_prestige_cost_value
				INFLUENCE = ftr_court_assist_monarch_influence_cost_value
			}
			add_character_flag = { flag = ftr_court_assisted_this_year  years = 1 }
		}
		# 拉锯向君主一侧移动
		scope:recipient = {
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				save_scope_as = ftr_court_story
			}
			add_opinion = { target = scope:actor  modifier = ftr_court_loyalist_opinion }
			update_diarchy_swing_with_perspective_effect = {
				SWING = minor_sop_swing_liege_gain
				LIEGE = scope:recipient
			}
		}
		if = {
			limit = { exists = scope:ftr_court_story }
			scope:ftr_court_story = {
				set_variable = {
					name = ftr_court_swing_amount
					value = ftr_court_swing_support_monarch_value
				}
				ftr_court_shift_swing_effect = yes
				remove_variable = ftr_court_swing_amount
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = 0
		# 与最高统治者的关系（意见修正值）：君主越器重发起者，发起者越愿辅政
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		# 与共治者的关系（意见修正值）：共治者越器重发起者，发起者越不愿辅政打压共治者
		# 负权重稍大：并非每个臣子都愿冒风险做忠臣
		opinion_modifier = {
			who = scope:recipient.diarch
			opinion_target = scope:actor
			multiplier = -1.5
			desc = AI_OPINION_REASON
		}
		# 有忠信特质：天性忠贞，愿辅佐君主
		modifier = {
			add = 40
			scope:actor = { has_trait = loyal }
		}
		# 与共治者是仇敌：借辅政打压共治者
		modifier = {
			add = 40
			scope:actor = { has_relation_rival = scope:recipient.diarch }
		}
	}
}

######### 许诺支持：许以内阁席位 / 许以宽免义务 #########
# 合并原两个 promise 交互，许诺内容做成 send_option 选项。
ftr_court_promise_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_promise_interaction_desc
	cooldown_against_recipient = { years = 3 }
	ai_targets = {
		ai_recipients = peer_vassals
		max = 2
	}
	ai_frequency = 72

	send_option = {
		flag = ftr_promise_council
		localization = ftr_court_promise_council_option
	}
	send_option = {
		flag = ftr_promise_contract
		localization = ftr_court_promise_contract_option
	}
	send_options_exclusive = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		# 只有竞争者（参与角逐）或共治者才许得出筹码
		scope:actor = {
			# 无地竞争者无法主动发起交互（拉拢走事件）
			is_landed = yes
			OR = {
				ftr_court_is_active_claimant_trigger = yes
				has_active_diarchy = yes
			}
		}
		# 契约放宽仅对有地臣属；内阁席位对朝堂成员（封臣/廷臣）均可
		scope:recipient = { ftr_court_is_court_member_trigger = yes }
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			# 两种许诺的支付门槛取较高者（统一用内阁席位成本判定，宽松）
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_promise_council_prestige_cost_value
				INFLUENCE = ftr_court_promise_council_influence_cost_value
			}
		}
		scope:recipient = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		# 依 send_option 选择许诺内容
		if = {
			limit = { scope:actor = { has_character_flag = ftr_promise_council } }
			scope:actor = {
				ftr_court_pay_cost_effect = {
					OTHER = scope:recipient
					PRESTIGE = ftr_court_promise_council_prestige_cost_value
					INFLUENCE = ftr_court_promise_council_influence_cost_value
				}
				# 欠账计数：继承后依此触发提醒事件
				if = {
					limit = { NOT = { has_variable = ftr_court_owes_council } }
					set_variable = { name = ftr_court_owes_council  value = 0 }
				}
				change_variable = { name = ftr_court_owes_council  add = 1 }
				remove_character_flag = ftr_promise_council
			}
			# 记在被承诺者身上，便于日后集中兑现
			scope:recipient = {
				set_variable = { name = ftr_court_promised_by  value = scope:actor }
				set_variable = { name = ftr_court_promise_council  value = 1 }
				add_opinion = { target = scope:actor  modifier = ftr_political_alliance_opinion }
			}
		}
		else = {
			# 契约放宽
			scope:actor = {
				ftr_court_pay_cost_effect = {
					OTHER = scope:recipient
					PRESTIGE = ftr_court_promise_contract_prestige_cost_value
					INFLUENCE = ftr_court_promise_contract_influence_cost_value
				}
				remove_character_flag = ftr_promise_contract
			}
			scope:recipient = {
				set_variable = { name = ftr_court_promised_by  value = scope:actor }
				set_variable = { name = ftr_court_promise_contract  value = 1 }
				add_opinion = { target = scope:actor  modifier = ftr_political_alliance_opinion }
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = 10
		modifier = {
			add = 40
			scope:actor = { opinion = { target = scope:recipient  value < 30 } }
		}
		modifier = {
			add = 25
			scope:recipient = { is_powerful_vassal_of = scope:actor.top_liege }
		}
		modifier = {
			add = 30
			scope:recipient = { has_trait = greedy }
		}
	}
}

######### 游说：拉拢同僚推动权力天平（议题 2 的主动博弈） #########
# 权臣阵营借此推高天平，保皇阵营借此巩固人心。
# 被游说者可以是封臣或无地廷臣——廷臣由此获得被动参与路径。
ftr_court_lobby_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_lobby_interaction_desc
	cooldown_against_recipient = { years = 1 }

	send_option = {
		flag = lobby_diarch
		localization = ftr_court_lobby_diarch
	}
	send_option = {
		flag = lobby_monarch
		localization = ftr_court_lobby_monarch
	}
	send_options_exclusive = yes

	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 2
	}
	ai_frequency = 48

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			# 无地廷臣不能主动发起交互
			is_landed = yes
			NOT = { ftr_court_is_struggle_monarch_trigger = yes }
			# 双方处于同一朝堂（故事挂在君主身上）
			OR = {
				top_liege ?= { any_owned_story = { type = ftr_court_struggle } }
				AND = {
					exists = employer
					employer ?= { any_owned_story = { type = ftr_court_struggle } }
				}
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_lobby_prestige_cost_value
				INFLUENCE = ftr_court_lobby_influence_cost_value
			}
		}
		scope:recipient = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_lobby_prestige_cost_value
				INFLUENCE = ftr_court_lobby_influence_cost_value
			}
		}

		# 定位所属朝堂
		scope:actor.top_liege ?= {
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				save_scope_as = ftr_court_story
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:ftr_court_story }
				exists = scope:actor.employer
			}
			scope:actor.employer = {
				every_owned_story = {
					limit = { story_type = ftr_court_struggle }
					save_scope_as = ftr_court_story
				}
			}
		}

		if = {
			limit = { exists = scope:ftr_court_story }
			scope:ftr_court_story = {
				if = {
					limit = { scope:actor = { has_character_flag = lobby_diarch } }
					# 劝进：游说对象被拉向共治阵营
					set_variable = {
						name = ftr_court_swing_amount
						value = ftr_court_swing_support_diarch_value
					}
				}
				else = {
					# 劝忠：游说对象被拉回君主阵营
					set_variable = {
						name = ftr_court_swing_amount
						value = ftr_court_swing_support_monarch_value
					}
				}
				ftr_court_shift_swing_effect = yes
				remove_variable = ftr_court_swing_amount
			}
		}

		# 人情记账
		if = {
			limit = { scope:actor = { has_character_flag = lobby_diarch } }
			scope:recipient = {
				add_opinion = { target = scope:actor  modifier = ftr_political_alliance_opinion }
			}
		}
		else = {
			scope:actor.top_liege ?= {
				save_scope_as = ftr_court_monarch
			}
			scope:recipient = {
				add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_court_loyalist_opinion }
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = 0
		modifier = {
			add = 25
			scope:actor = { has_trait = ambitious }
		}
		modifier = {
			add = 25
			scope:actor = { has_trait = gregarious }
		}
		modifier = {
			add = -30
			scope:actor = { has_trait = shy }
		}
	}
