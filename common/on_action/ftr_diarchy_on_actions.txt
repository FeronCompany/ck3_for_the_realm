########## 一、无地共治者授土提醒（§4.0） ##########

# 求值域：任意角色（年度脉冲）
ftr_diarchy_landless_reminder_on_action = {
	trigger = {
		ftr_diarchy_landless_reminder_trigger = yes
		# 每 3 年一次：提醒后打 3 年冷却 flag
		NOT = { has_character_flag = ftr_diarchy_land_reminder_cooldown }
		exists = liege
		liege = { ftr_diarchy_liege_has_land_to_give_trigger = yes }
	}
	effect = {
		add_character_flag = {
			flag = ftr_diarchy_land_reminder_cooldown
			years = ftr_diarchy_grant_land_reminder_years_value
		}
		trigger_event = { id = ftr_diarchy.0200  days = 3 }
	}
}

########## 二、共治者变更钩子（§4.3 拉升顺位的落地） ##########

# 原版 on_diarch_change：root = 主君；scope:old_diarch = 前任共治者；
# scope:reason = flag:death / depose / invalid / inheritance
# 用途：若主君曾登记「候任共治者」（§4.3），在新共治者选出后顶替之。
ftr_diarchy_diarch_change_on_action = {
	trigger = {
		has_variable = ftr_diarchy_boosted_candidate
		has_active_diarchy = yes
		# 候任者仍有效
		var:ftr_diarchy_boosted_candidate ?= {
			is_alive = yes
			is_adult = yes
			is_imprisoned = no
			is_diarch = no
		}
		NOT = { diarch ?= { this = var:ftr_diarchy_boosted_candidate } }
	}
	effect = {
		var:ftr_diarchy_boosted_candidate ?= { save_scope_as = ftr_diarchy_boost_target }
		set_diarch = scope:ftr_diarchy_boost_target
		remove_variable = ftr_diarchy_boosted_candidate
	}
}

########## 三、季度背书博弈（§7.1） ##########

# 求值域：季度脉冲下的可玩角色（root = 该角色）。
# 仅由**君主**一侧执行（排除共治者自身），避免同一共治关系被算两次。
# 注：季度脉冲只覆盖 playable 角色；AI 政权侧的 swing 仍由 0500 故事链与引擎 swing_balance 自然演化。
ftr_diarchy_quarterly_drift_on_action = {
	trigger = {
		is_diarch = no
		has_active_diarchy = yes
		exists = diarch
	}
	effect = {
		ftr_diarchy_quarterly_swing_drift_effect = yes
	}
}

########## 四、立储后异步询问（§2.2 2026-09 重构） ##########

# 求值域：君主（年度 everyone 脉冲）。立储只写继位，是否另立太子共治改由本异步询问决定：
#   触发 ftr_diarchy.0710（root = 君主）询问是否册立；同意即建、拒绝给被指定继承人好感惩罚且不再重问。
# 注：01 故事主时钟也会调用同一效果（ftr_diarchy_promote_adult_heir_effect），两路互为兜底；
#     效果自带 30 天冷却 flag 与「已拒」变量，避免重复派发 / 重复询问。
ftr_diarchy_crown_prince_ask_on_action = {
	trigger = {
		ftr_diarchy_can_start_crown_prince_trigger = yes
		exists = designated_heir
	}
	effect = {
		ftr_diarchy_promote_adult_heir_effect = yes
	}
}

########## 五、太子共治自动失效（§3.1 补） ##########

# 求值域：君主（root = 主君，年度 playable 脉冲）。
# 太子共治在下列任一条件下自动结束：
#   ① 君主与共治者不在同一政权 —— 共治者的 top_liege 已不是主君（被转封他国 / 独立）；
#   ② 共治者级别已等于或高于君主 —— 储君已然坐大，太子位名不副实。
# 注：主君侧的条件失效（失独立 / 换政体 / 等级跌破王国）由 diarchy_types 的 end 块自行处理，
#     本 on_action 只补「共治者侧」这两条。
ftr_diarchy_crown_prince_invalidate_on_action = {
	trigger = {
		has_diarchy_type = ftr_diarchy_crown_prince
		exists = diarch
		OR = {
			# ① 不在同一政权（顶层领主已非主君）
			NOT = { diarch.top_liege = root }
			# ② 共治者级别 ≥ 君主
			diarch = { highest_held_title_tier >= root.highest_held_title_tier }
			NOT = { is_sibling_of = diarch }
			NOT = { is_parent_of = diarch }
			NOT = { is_grandparent_of = diarch }
		}
	}
	effect = {
		end_diarchy = yes
	}
}

########## 六、太子留守宫廷（§3.1 补，2026-09） ##########

# 求值域：太子共治者本人（年度 everyone 脉冲）。
# 行政制下共治者一旦被外放为地方官，就不再是主君直属封臣、共治随之解除。
# 本 on_action 定期把跑偏的太子拉回宫廷：交出 landed 头衔 + 封锁后续任命。
# 注：立储（ftr_diarchy_try_start_crown_prince_effect）时就已即时执行一次，
#     本脉冲只是兜底——覆盖「立储之后才被任命」的情况。
ftr_diarchy_crown_prince_keep_in_court_on_action = {
	trigger = {
		ftr_diarchy_is_crown_prince_trigger = yes
	}
	effect = {
		ftr_diarchy_enforce_crown_prince_unlanded_effect = yes
	}
}

# 求值域：君主（on_left_diarchy：root = 君主，scope:old_diarch = 原共治者）。
# 共治结束（含君主驾崩、废储、篡位等）即解除对原太子的任命封锁，
# 避免其日后无法受任 / 无法被任命继位。
ftr_diarchy_crown_prince_release_on_left_on_action = {
	effect = {
		scope:old_diarch ?= { ftr_diarchy_release_appointment_block_effect = yes }
	}
}
