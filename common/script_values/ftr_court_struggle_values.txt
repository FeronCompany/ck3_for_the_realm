########## 四、竞争力评分总分 ##########
# 求值时的作用域：this = 候选人，scope:ftr_court_monarch = 在位君主
ftr_court_claim_strength_value = {
	value = 0

	### 成年与年龄阶段
	if = {
		limit = { is_adult = yes }
		add = ftr_court_claim_adult_value
		if = {
			limit = { age >= 16  age <= 45 }
			add = ftr_court_claim_prime_age_value
		}
		else_if = {
			limit = { age >= ftr_court_monarch_elderly_age_value }
			add = ftr_court_claim_old_age_value
		}
	}

	### 个人能力
	add = ftr_court_claim_average_attribute_value

	### 特质
	if = { limit = { has_trait = ambitious }		add = ftr_court_claim_ambitious_value }
	if = { limit = { has_trait = brave }			add = ftr_court_claim_brave_value }
	if = { limit = { has_trait = genius }			add = ftr_court_claim_genius_value }
	if = { limit = { has_trait = content }			add = ftr_court_claim_content_value }
	if = { limit = { has_trait = craven }			add = ftr_court_claim_craven_value }
	if = { limit = { has_trait = humble }			add = ftr_court_claim_humble_value }
	if = { limit = { has_trait = incapable }		add = ftr_court_claim_incapable_value }
	if = { limit = { has_trait = lunatic }			add = ftr_court_claim_lunatic_value }
	if = { limit = { has_trait = inbred }			add = ftr_court_claim_inbred_value }
	if = { limit = { has_trait = drunkard }			add = ftr_court_claim_drunkard_value }
	if = { limit = { has_trait = reclusive }		add = ftr_court_claim_reclusive_value }
	if = { limit = { has_trait = shy }				add = ftr_court_claim_shy_value }

	### 实力与身份
	if = { limit = { is_landed = yes }									add = ftr_court_claim_landed_value }
	if = { limit = { is_councillor_of = scope:ftr_court_monarch }		add = ftr_court_claim_councillor_value }
	if = { limit = { is_diarch_of_target = scope:ftr_court_monarch }	add = ftr_court_claim_diarch_value }
	if = { limit = { this = scope:ftr_court_monarch.spouse }			add = ftr_court_claim_spouse_value }
	if = { limit = { is_heir_of = scope:ftr_court_monarch }				add = ftr_court_claim_primary_heir_value }

	### 人望：君主好感（分段判定，避免在数值上下文直接引用 opinion）
	if = {
		limit = { opinion = { target = scope:ftr_court_monarch  value >= 50 } }
		add = ftr_court_claim_opinion_high_value
	}
	else_if = {
		limit = { opinion = { target = scope:ftr_court_monarch  value >= 0 } }
		add = ftr_court_claim_opinion_mid_value
	}
	else = {
		add = ftr_court_claim_opinion_low_value
	}

	### 支持者数量
	if = {
		limit = { has_variable = ftr_court_backer_count }
		add = {
			value = ftr_court_claim_backer_value
			multiply = var:ftr_court_backer_count
		}
	}

	### 永不出现负分，便于展示
	max = 0
}
