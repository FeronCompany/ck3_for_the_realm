##############################################################
### 朝堂权力角逐 COURT STRUGGLE — 条件层
### 议题1 继任之争 / 议题2 共治夺权
##############################################################

########## 一、准入：独立政权 + 等级 >= 王国 + 存在继承顺位 ##########

# 政权等级达标（王国 / 帝国 / 霸权）
ftr_court_realm_tier_valid_trigger = {
	highest_held_title_tier >= tier_kingdom
}

# 存在继承顺位：最高统治者的主头衔继承顺位列表不为空即触发
# 全面放宽：不要求候选人数、不要求继承人成年/有能力，只要主头衔有继承顺位即可。
ftr_court_has_succession_order_trigger = {
	primary_title = {
		any_title_heir = { always = yes }
	}
}

# 准入总判定：在最高统治者身上求值
ftr_court_struggle_valid_trigger = {
	is_ruler = yes
	is_independent_ruler = yes
	ftr_court_realm_tier_valid_trigger = yes
	ftr_court_has_succession_order_trigger = yes
	NOT = { is_imprisoned = yes }
}

# 收场核心判定：故事持有者是否仍是「最高领主」（求值域：story_owner）。
# 只有失去最高领主地位（不再独立，或等级跌至王国以下）才应当收场 end_story；
# 被囚禁、暂时无继承顺位、战争中 等临时失效一律不触发收场（保留故事等待恢复或真失位）。
# 注意：story owner 驾崩后 is_ruler / is_independent_ruler 在引擎语义下通常会失效，
# 死亡交接（有继承人 → transfer；无继承人 → end_story）已由 on_owner_death 独立处理，
# 此处判定不重复考虑死亡。
ftr_court_story_owner_is_supreme_liege_trigger = {
	is_independent_ruler = yes
	ftr_court_realm_tier_valid_trigger = yes
}

# 冷却期内不再激活（配合 add_character_flag 使用）
ftr_court_in_cooldown_trigger = {
	has_character_flag = ftr_court_cooldown
}

# 完整准入：满足基本条件 + 不在冷却 + 没有正在进行的战争（避免叠加）
ftr_court_can_start_struggle_trigger = {
	ftr_court_struggle_valid_trigger = yes
	NOT = { ftr_court_in_cooldown_trigger = yes }
	NOT = { is_at_war = yes }
}

########## 二、竞争者资格 ##########

# 求值域：this = 候选人；依赖 scope:ftr_court_monarch
ftr_court_claimant_valid_trigger = {
	exists = scope:ftr_court_monarch
	NOT = { this = scope:ftr_court_monarch }
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	# 必须同宗族，避免外戚/廷臣乱入
	dynasty ?= {
		this = scope:ftr_court_monarch.dynasty
	}
}

########## 二之二、参与角逐系统 ##########

# 求值域：this = 潜在继任者。是否可以被询问是否参与角逐。
# 血缘继承：子女 / 兄弟姐妹（按性别继承法过滤）；选举继承：主头衔继承顺位。
# 依赖 scope:ftr_court_monarch（君主）。
ftr_court_can_be_asked_participate_trigger = {
	exists = scope:ftr_court_monarch
	is_alive = yes
	is_adult = yes
	NOT = { this = scope:ftr_court_monarch }
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	# 已明确表示不参与者不再询问
	NOT = { has_character_flag = ftr_court_declined_contest }
	# 已参与角逐者不再询问
	NOT = { has_character_flag = ftr_court_participating }
}

# 求值域：this = 潜在继任者。按性别继承法过滤的血缘继承资格。
# 依赖 scope:ftr_court_monarch（君主）。
ftr_court_hereditary_claimant_candidate_trigger = {
	exists = scope:ftr_court_monarch
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	OR = {
		# 子女
		AND = {
			is_child_of = scope:ftr_court_monarch
			ftr_court_gender_succession_allows_trigger = yes
		}
		# 兄弟姐妹（仅当君主无任何成年子女时才询问手足，参考原 UI 血缘逻辑）
		AND = {
			is_sibling_of = scope:ftr_court_monarch
			NOT = {
				scope:ftr_court_monarch = {
					any_child = { is_adult = yes }
				}
			}
			dynasty ?= { this = scope:ftr_court_monarch.dynasty }
		}
	}
}

# 求值域：this = 潜在继任者。性别继承法是否允许其角逐（仅男性 / 仅女性 / 默认男性优先）。
ftr_court_gender_succession_allows_trigger = {
	OR = {
		# 仅男性继承：仅男性
		AND = {
			primary_title ?= { has_title_law = male_only_law }
			is_male = yes
		}
		# 仅女性继承：仅女性
		AND = {
			primary_title ?= { has_title_law = female_only_law }
			is_female = yes
		}
		# 其余（male_preference / equal）：男性优先——已有成年男性子女时不询问女性，否则男女皆可
		AND = {
			NOT = { primary_title ?= { has_title_law = male_only_law } }
			NOT = { primary_title ?= { has_title_law = female_only_law } }
			OR = {
				is_male = yes
				AND = {
					is_female = yes
					NOT = {
						scope:ftr_court_monarch = {
							any_child = { is_adult = yes  is_male = yes }
						}
					}
				}
			}
		}
	}
}

# 求值域：this = 潜在继任者（title heir 之一）。选举继承顺位资格（主头衔的 title heir）。
# 注意：本 trigger 仅在调用方对主头衔 every_title_heir 遍历的 limit 中使用，this 已确属
# title heir，因此无需在此重复进入主头衔嵌套比较（原写法以 this = prev 比较角色与领地
# 头衔两类作用域，触发 jomini 类型错误）。
ftr_court_elective_claimant_candidate_trigger = {
	exists = scope:ftr_court_monarch
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
}

# 求值域：this = 潜在继任者。是否仍符合参与角逐的资格（用于定期刷新移除）。
ftr_court_participant_remains_valid_trigger = {
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	# 被剥夺继承权者移除
	NOT = { has_trait = disinherited }
	# 仍然参与角逐
	has_character_flag = ftr_court_participating
}

########## 三、君主状态 ##########

# 君主因年老体衰或精神异常而不能正常履职
ftr_court_monarch_infirm_trigger = {
	OR = {
		has_trait = lunatic
		has_trait = possessed
		has_trait = depressed
		has_trait = blind
		age >= ftr_court_monarch_elderly_age_value
	}
}

# 君主权威衰弱（政变/内战的温床）
ftr_court_monarch_weak_trigger = {
	OR = {
		prestige < ftr_court_monarch_weak_prestige_value
		piety < ftr_court_monarch_weak_piety_value
		ftr_court_monarch_infirm_trigger = yes
	}
}

########## 四、阶段判定 ##########

# 候选人数量达标（数量由参与事件 0230 写入 ftr_court_candidate_count）
ftr_court_has_enough_candidates_trigger = {
	has_variable = ftr_court_candidate_count
	var:ftr_court_candidate_count >= ftr_court_min_candidates_value
}

# 竞争者众多（三人成势）
ftr_court_has_many_candidates_trigger = {
	has_variable = ftr_court_candidate_count
	var:ftr_court_candidate_count >= 3
}

# 君主域实时判定：潜在继任竞争者是否充足。
# 参考 UI 继任者收集逻辑（ftr_court_sync_courtiers_effect）——在君主域按血缘/选举分支
# 实时收集，不依赖 story 域变量 ftr_court_candidate_count（修复 on_action 准入时 story
# 尚未建立、读不到该变量导致准入恒失败的缺陷）。
# 血缘继承：有继承权的子女（含未成年）或（无成年子女时）同宗族成年手足；
# 选举继承：主头衔顺位继承者。
# 议题1 是否应当激活（在君主身上求值）
# 全面放宽：只要最高统治者的主头衔继承顺位列表非空即触发，
# 不再要求候选人数达标或君主衰弱/竞争者众多。
ftr_court_should_activate_succession_trigger = {
	ftr_court_struggle_valid_trigger = yes
	NOT = { ftr_court_in_cooldown_trigger = yes }
}

# ---- 烈度判定（替代原 ftr_court_phase 0-3 离散阶段）----
# 各 trigger 求值域：story（root = 故事本体）。烈度 = var:ftr_court_struggle_intensity（0-200）。

# 参与者数量是否已进入角逐下限（有至少 ftr_court_min_candidates_value 名参与者才谈得上对抗）
ftr_court_has_active_struggle_trigger = {
	has_variable = ftr_court_candidate_count
	var:ftr_court_candidate_count >= ftr_court_min_candidates_value
}

# 烈度是否进入「对立」区（>=90）：两派公开倾轧、对立脉冲 / 0200 站队等在此区触发。
ftr_court_is_contested_trigger = {
	ftr_court_has_active_struggle_trigger = yes
	has_variable = ftr_court_struggle_intensity
	var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value
}

# 烈度是否进入「濒危/危机」区（>=140）：政变/内战出口开启、危机脉冲在此区触发。
# 原逻辑含「君主衰弱」要求，烈度化后该偏置已通过 ftr_court_intensity_weak_bias_value 计入烈度，
# 故此处不再单独要求君主弱（君主弱会推高烈度，间接进入此窗口）。
ftr_court_is_crisis_trigger = {
	ftr_court_has_active_struggle_trigger = yes
	has_variable = ftr_court_struggle_intensity
	var:ftr_court_struggle_intensity >= ftr_court_intensity_high_bound_value
}

# 烈度是否进入「临界」区（>=170）：废黜昏君（0150）等最强手段在此区开放。
ftr_court_intensity_critical_trigger = {
	has_variable = ftr_court_struggle_intensity
	var:ftr_court_struggle_intensity >= ftr_court_intensity_critical_bound_value
}

# 缓和决议的可用前提：烈度高于门槛才值得烧资源缓和（§12.5 / §12.11）
ftr_court_ease_struggle_valid_trigger = {
	has_variable = ftr_court_struggle_intensity
	var:ftr_court_struggle_intensity >= ftr_court_intensity_high_bound_value
	NOT = { has_character_flag = ftr_court_eased_struggle }
}

# 立场不同判定（M2 立场辩论选角用）。
# 求值域：this = 候选人；参数 OTHER = 对照者。成立：两人封臣立场不相同（七种立场逐一比对）。
ftr_court_has_different_vassal_stance_trigger = {
	OR = {
		AND = { $OTHER$ = { has_vassal_stance = courtly }  NOT = { has_vassal_stance = courtly } }
		AND = { $OTHER$ = { has_vassal_stance = glory_hound }  NOT = { has_vassal_stance = glory_hound } }
		AND = { $OTHER$ = { has_vassal_stance = parochial }  NOT = { has_vassal_stance = parochial } }
		AND = { $OTHER$ = { has_vassal_stance = zealot }  NOT = { has_vassal_stance = zealot } }
		AND = { $OTHER$ = { has_vassal_stance = minority }  NOT = { has_vassal_stance = minority } }
		AND = { $OTHER$ = { has_vassal_stance = barons_and_minor_landholders }  NOT = { has_vassal_stance = barons_and_minor_landholders } }
		AND = { $OTHER$ = { has_vassal_stance = belligerent }  NOT = { has_vassal_stance = belligerent } }
	}
}

# 求值域：this = 某内阁成员。判定其是否「不够称职」（M1 弹劾内阁的目标筛选）。
# 口径：按其所任内阁职位对应的属性低于称职线（decent_skill_rating）即算不称职；
# 五种常规职位（掌玺/军事/财政/间谍/宫廷司铎）逐一对应，其他职位（如游牧）不参与。
ftr_court_is_incompetent_councillor_trigger = {
	is_councillor = yes
	OR = {
		AND = { has_council_position = councillor_chancellor      diplomacy < decent_skill_rating }
		AND = { has_council_position = councillor_marshal         martial < decent_skill_rating }
		AND = { has_council_position = councillor_steward         stewardship < decent_skill_rating }
		AND = { has_council_position = councillor_spymaster       intrigue < decent_skill_rating }
		AND = { has_council_position = councillor_court_chaplain  learning < decent_skill_rating }
	}
}

# 求值域：this = 任意角色；参数 THRESHOLD = 烈度门槛（script value 名或数字）。
# 判定其所在政权的朝堂故事烈度是否 ≥ THRESHOLD：依次尝试自身持有的 story / 最高领主 / 雇主。
# 供内斗事件选项的 ai_chance 使用（事件 root 未必持有 story，故三路兜底）。
ftr_court_struggle_intensity_of_realm_at_least_trigger = {
	OR = {
		any_owned_story = {
			story_type = ftr_court_struggle
			var:ftr_court_struggle_intensity >= $THRESHOLD$
		}
		top_liege ?= {
			any_owned_story = {
				story_type = ftr_court_struggle
				var:ftr_court_struggle_intensity >= $THRESHOLD$
			}
		}
		employer ?= {
			any_owned_story = {
				story_type = ftr_court_struggle
				var:ftr_court_struggle_intensity >= $THRESHOLD$
			}
		}
	}
}

########## 五、议题2：共治夺权 ##########

# 议题2 是否应当激活：君主不能正常履职，或已处于摄政
ftr_court_should_activate_regency_trigger = {
	ftr_court_struggle_valid_trigger = yes
	NOT = { ftr_court_in_cooldown_trigger = yes }
	OR = {
		ftr_court_monarch_infirm_trigger = yes
		has_active_diarchy = yes
	}
}

# 拉锯足以发起共治请愿（求值域 = 朝堂 story，story_owner 为君主）。
# 裁定 R1：请愿仅在「朝中尚无共治者」时开放（`is_shown` 已作独占限制），而天平是 diarchy 的
# 属性、无共治时恒为 0 ⇒ 唯一门槛 = 「朝堂已入对立区」
# （复用 ftr_court_is_contested_trigger：参与者达标 + 烈度 ≥ 90）。原 swing ≥ 70 分支已删除。
ftr_court_can_petition_coregency_trigger = {
	story_owner ?= { NOT = { has_active_diarchy = yes } }
	ftr_court_is_contested_trigger = yes
}

# 拉锯足以尝试夺位（官方 swing + 君主失能）
ftr_court_can_attempt_usurp_trigger = {
	story_owner ?= {
		diarchy_swing >= ftr_court_swing_usurp_threshold_value
		ftr_court_monarch_infirm_trigger = yes
	}
}

# 辅政成功、大权稳固（官方 swing 偏向君主）
ftr_court_loyalists_hold_trigger = {
	story_owner ?= {
		diarchy_swing <= ftr_court_swing_loyal_threshold_value
	}
}

########## 六、行动资格 ##########

# 竞争者能否发动政变（走 coup 计谋）
# 求值域：候选人自身（支持者列表 ftr_court_supporters 直接维护在角色上）
ftr_court_can_launch_coup_trigger = {
	ftr_court_own_struggle_in_crisis_trigger = yes
	ftr_court_supporters_adequate_trigger = yes
}

# 竞争者能否起兵内战
# 求值域：候选人自身（支持者列表 ftr_court_supporters 直接维护在角色上）
ftr_court_can_declare_succession_war_trigger = {
	ftr_court_own_struggle_in_crisis_trigger = yes
	NOT = { is_at_war = yes }
	ftr_court_supporters_adequate_trigger = yes
	# 无地者也能起兵：CB 会为其创建临时公国
}

# 求值域：候选人自身。判断其 ftr_court_supporters 支持者列表是否达到起事下限。
# ftr_court_supporters 是 add_to_variable_list 维护的 variable list：先 has_variable_list 保护，
# 大小用 variable_list_size trigger 判断（不用临时 list 的 list_size: 前缀）。
ftr_court_supporters_adequate_trigger = {
	has_variable_list = ftr_court_supporters
	variable_list_size = {
		name = ftr_court_supporters
		value >= ftr_court_min_backers_value
	}
}

# 从自身出发，找到所参与的朝堂争夺并判断其是否已进入危机
# 求值域：任意朝堂成员（封臣走 top_liege；无地廷臣走 employer）
ftr_court_own_struggle_in_crisis_trigger = {
	OR = {
		top_liege ?= {
			any_owned_story = {
				type = ftr_court_struggle
				ftr_court_is_crisis_trigger = yes
			}
		}
		employer ?= {
			any_owned_story = {
				type = ftr_court_struggle
				ftr_court_is_crisis_trigger = yes
			}
		}
	}
}

# 无地竞争者的起兵资格（用于事件路径，不要求宣战 UI 可达）
ftr_court_landless_rebel_valid_trigger = {
	is_landed = no
	NOT = { is_imprisoned = yes }
	NOT = { is_at_war = yes }
	ftr_court_can_declare_succession_war_trigger = yes
}

# 无地竞争者（廷臣皇子/王子），需要创建临时公国才能参战
ftr_court_needs_temp_duchy_trigger = {
	is_landed = no
	NOT = { is_imprisoned = yes }
}

# 2.2 派间谍调查（0400）触发者资格（§9.2，root = 潜在触发者）
# 三种合法来源，任满足其一：
#   ① 最高统治者（持有朝堂 story 的独立统治者）
#   ② 有头衔 / 领地的参与者（is_landed 即持有头衔与领地，且其 top_liege / employer 持有 story）
#   ③ 参与者的有头衔 / 领地支持者（backer_of 指向某参与者且自己有领地）
# 注：无地廷臣不在池内（调查需自有班底与头衔地位）。
ftr_court_can_order_investigation_trigger = {
	OR = {
		# ① 最高统治者
		AND = {
			is_independent_ruler = yes
			any_owned_story = { type = ftr_court_struggle }
		}
		# ② 有领地的参与者（须在朝堂中：所属 top_liege / employer 持有 story）
		AND = {
			is_landed = yes
			has_character_flag = ftr_court_participating
			OR = {
				top_liege ?= { any_owned_story = { type = ftr_court_struggle } }
				employer ?= { any_owned_story = { type = ftr_court_struggle } }
			}
		}
		# ③ 参与者的有领地支持者
		AND = {
			is_landed = yes
			has_variable = ftr_court_backer_of
			var:ftr_court_backer_of ?= { has_character_flag = ftr_court_participating }
		}
	}
}

########## 七、身份判定（用于交互/决议的可见性） ##########

# 玩家是当前朝堂的最高统治者
ftr_court_is_struggle_monarch_trigger = {
	is_ruler = yes
	is_independent_ruler = yes
	ftr_court_realm_tier_valid_trigger = yes
}

# 玩家是竞争者之一（参与角逐的继任者）
ftr_court_is_active_claimant_trigger = {
	has_character_flag = ftr_court_participating
	NOT = { ftr_court_is_struggle_monarch_trigger = yes }
}

# 可被臣属支持的对象（权贵）：最高统治者 / 共治者 / 参与角逐的继任者
ftr_court_is_supported_person_trigger = {
	OR = {
		# 最高统治者本人
		ftr_court_is_struggle_monarch_trigger = yes
		# 共治者
		is_diarch = yes
		# 参与角逐的继任者
		has_character_flag = ftr_court_participating
	}
}

# 玩家有站队对象（已卷入）
ftr_court_has_backed_someone_trigger = {
	has_variable = ftr_court_backer_of
}

# 已站队且支持对象仍是权贵（用于统计支持者数量时过滤失效站队）
ftr_court_has_backing_trigger = {

# 党羽同阵营判定（供手段族 AI 的党羽立场过滤复用，§7.3）。
# 求值域：this = 待判定者；参数 OTHER = 对照者（角色 scope 表达式）。
# 成立：我已站队，且 OTHER 属我方党羽——① OTHER 在我支持党首的 ftr_court_supporters 名单内，
# 或 ② OTHER 正是我支持的党首本人（党首亦属本阵营）。未站队者无阵营 → 恒为假。
ftr_court_same_camp_trigger = {
	has_variable = ftr_court_backer_of
	var:ftr_court_backer_of ?= {
		OR = {
			is_target_in_variable_list = { name = ftr_court_supporters  target = $OTHER$ }
			this = $OTHER$
		}
	}
}

########## 七之二、AI 背书立场偏好（事件 0200 站队用） ##########
# 供 0200 五个站队选项复用：把"封臣立场 ↔ 候选特质"是否契合做成参数化 trigger，
# 契合加分 / 违背减分，使 AI 不只看名次而按自身立场挑选对象（对齐交互 ai_will_do 的 stance 语义）。
# 参数：CANDIDATE = 被评估的候选；ACTOR = 朝臣本人（求值域当前 this 即朝臣）。
ftr_court_backing_stance_fit_trigger = {
	OR = {
		AND = {
			has_vassal_stance = courtly
			$CANDIDATE$ = {
				OR = {
					has_trait = gregarious
					has_trait = generous
					diplomacy >= 6
				}
			}
		}
		AND = {
			has_vassal_stance = glory_hound
			$CANDIDATE$ = {
				OR = {
					has_trait = brave
					has_trait = arrogant
					martial >= 6
				}
			}
		}
		AND = {
			has_vassal_stance = parochial
			$CANDIDATE$ = {
				OR = {
					has_trait = content
					has_trait = stubborn
					stewardship >= 6
				}
			}
		}
		AND = {
			has_vassal_stance = zealot
			$CANDIDATE$ = {
				NOT = { has_trait = cynical }
				OR = {
					has_trait = zealous
					has_trait = humble
					learning >= 6
				}
			}
		}
		AND = {
			has_vassal_stance = belligerent
			$CANDIDATE$ = {
				OR = {
					has_trait = wrathful
					has_trait = ambitious
					prowess >= 6
				}
			}
		}
		AND = {
			has_vassal_stance = minority
			$CANDIDATE$ = {
				OR = {
					faith = $ACTOR$.faith
					culture = $ACTOR$.culture
				}
			}
		}
	}
}

########## 八、资源支付（双轨制） ##########

# Mod 自带的 ftr_political_influence 已废弃。
# 支付规则：双方均为行政制 -> 消耗官方影响力；任一方非行政制 -> 消耗威望。
# 求值域：支付方。参数：OTHER = 对方角色；PRESTIGE / INFLUENCE = 正数成本常量。
ftr_court_can_pay_cost_trigger = {
	OR = {
		AND = {
			government_allows = administrative
			$OTHER$ = { government_allows = administrative }
			influence >= $INFLUENCE$
		}
		AND = {
			OR = {
				NOT = { government_allows = administrative }
				$OTHER$ = { NOT = { government_allows = administrative } }
			}
			prestige >= $PRESTIGE$
		}
	}
}

########## 九、继承类型判定（按政体/继承法区分继任竞争者获取方式） ##########

# 血缘继承（分割继承 / 长子幼子单继承人）：
# 竞争者优先为血亲——有继承权的子女 + 同宗族亲兄弟姐妹。
# 求值域：君主。
ftr_court_hereditary_succession_trigger = {
	OR = {

	}
}

# 选举 / 继承不稳定（选侯、部分头衔选举、欢呼继承）：
# 直接按头衔继承顺位（every_heir）获取。
# 求值域：君主。
ftr_court_elective_succession_trigger = {
	OR = {
		# 任一公爵级及以上直辖头衔采用选举继承法（CK3 无 any_title，用 any_held_title）
		any_held_title = {
			tier >= tier_duchy
			OR = {
				has_title_law = feudal_elective_succession_law
				has_title_law = princely_elective_succession_law
				has_title_law = saxon_elective_succession_law
				has_title_law = scandinavian_elective_succession_law
				has_title_law = gaelic_elective_succession_law
				has_title_law = tribal_elective_succession_law
				has_title_law = confederation_elective_succession_law
			}
		}
		# 拜占庭欢呼继承
		has_realm_law = acclamation_succession_law
	}
}

########## 十、党羽逼宫（Backer Demand） ##########

# 求值域：this = 权贵（竞争者 / 共治者）。其党羽规模是否已达逼宫门槛。
# 规模 = 支持者人数 + 强力封臣党羽加权（ftr_court_backer_demand_scale_value）。
ftr_court_backer_demand_scale_reached_trigger = {

	}
	ftr_court_backer_demand_scale_value >= ftr_court_backer_demand_threshold_value
}

########## 十一、参与询问刷新短路 ##########

# 求值域：最高领主。是否存在「尚未被询问参与」的潜在继任者。
# 供 story 的参与询问 effect_group（30–60 天）作前置短路：无候选时整轮跳过，避免全量重扫。
ftr_court_has_unasked_candidate_trigger = {

		any_child = { ftr_court_can_be_asked_participate_relative_trigger = yes }
		any_sibling = { ftr_court_can_be_asked_participate_relative_trigger = yes }
		primary_title = { any_title_heir = { ftr_court_can_be_asked_participate_relative_trigger = yes } }
	}
}
