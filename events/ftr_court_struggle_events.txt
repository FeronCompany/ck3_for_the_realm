			modifier = { add = 20  has_trait = ambitious }
			modifier = { add = -20  has_trait = craven }

		# 立即压制参与角逐者，把权力天平拉回君权一侧；同时压一档烈度

				limit = {
					has_variable = ftr_court_struggle_intensity
					var:ftr_court_struggle_intensity > ftr_court_intensity_min_value
				}
				change_variable = {
					name = ftr_court_struggle_intensity
					subtract = ftr_court_monarch_restraint_drop_value
				}
			}
			if = {
				limit = {
					has_variable = ftr_court_struggle_intensity
					var:ftr_court_struggle_intensity < ftr_court_intensity_min_value
				}
				set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_min_value }

######### 0011 其余竞争者的命运（root = 新君） #########
ftr_court_struggle.0011 = {
	type = character_event
	title = ftr_court_struggle.0011.t
	desc = ftr_court_struggle.0011.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		save_scope_as = ftr_court_new_monarch
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			save_scope_as = ftr_court_story
		}
		# 找出威胁最大的竞争者（有地、有兵、未被囚禁且非新君）——供逮捕/起兵事件使用
		scope:ftr_court_story ?= {
			ordered_in_list = {
				variable = ftr_court_participant_list
				order_by = max_military_strength
				limit = {
					NOT = { this = scope:ftr_court_new_monarch }
					is_landed = yes
					max_military_strength > 0
					NOT = { is_imprisoned = yes }
				}
				save_scope_as = ftr_court_dissenter
			}
		}
	}

	option = { # Pardon them and honour my promises
		name = ftr_court_struggle.0011.a
		ai_chance = {
			base = 40
			modifier = {
				add = 40
				has_trait = just
			}
		}
		ftr_court_fulfill_promises_effect = yes
		add_prestige = ftr_court_prestige_minor_value
	}
	option = { # Strike the loudest voice first
		name = ftr_court_struggle.0011.b
		trigger = { exists = scope:ftr_court_dissenter }
		ai_chance = {
			base = 40
			modifier = { add = 35  has_trait = ambitious }
			modifier = { add = 25  has_trait = callous }
			modifier = { add = 20  has_trait = paranoid }
			modifier = { add = -30  has_trait = content }
			modifier = { add = -25  has_trait = just }
			modifier = { add = -25  has_trait = forgiving }
		}
		# 枪打出头鸟：逮捕威胁最大的对手（有地且有兵者）。
		# 成功 → 囚禁并废其竞争地位、免其生前承诺；失败 → 对方不得不举兵造反。
		random = {
			chance = ftr_court_arrest_success_base_value
			scope:ftr_court_dissenter ?= {
				ftr_court_clear_claimant_status_effect = yes
				imprison = { type = dungeon  target = scope:ftr_court_dissenter }
			}
			add_dread = ftr_court_dread_medium_value
			# 成功标记：供失败兜底判定（互斥）
			set_variable = { name = ftr_court_arrest_succeeded  value = 1 }
			# 被捕者生前许下的承诺一并作废
			scope:ftr_court_new_monarch = {
				every_vassal = {
					limit = {
						var:ftr_court_promised_by ?= { this = scope:ftr_court_dissenter }
					}
					remove_variable = ftr_court_promise_council
					remove_variable = ftr_court_promise_contract
					remove_variable = ftr_court_promised_by
				}
				every_courtier = {
					limit = {
						var:ftr_court_promised_by ?= { this = scope:ftr_court_dissenter }
					}
					remove_variable = ftr_court_promise_council
					remove_variable = ftr_court_promise_contract
					remove_variable = ftr_court_promised_by
				}
			}
		}
		if = {
			limit = { NOT = { has_variable = ftr_court_arrest_succeeded } }
			# 逮捕失手：对方再无退路，被迫举兵
			scope:ftr_court_dissenter ?= {
				ftr_court_prepare_rebellion_effect = { TARGET = scope:ftr_court_new_monarch }
			}
			# 处置失败令新君威望受损，其余竞争者亦人人自危
			add_prestige = { value = 0  subtract = ftr_court_prestige_minor_value }
			scope:ftr_court_story ?= {
				every_in_list = {
					variable = ftr_court_participant_list
					limit = { NOT = { this = scope:ftr_court_dissenter } }
					add_opinion = { target = scope:ftr_court_new_monarch  modifier = ftr_political_opponent_opinion }
				}
			}
		}
		remove_variable = ftr_court_arrest_succeeded
	}
	option = { # Let them keep their titles, but watch them
		name = ftr_court_struggle.0011.c
		trigger = {
			exists = scope:ftr_court_dissenter
		}
		ai_chance = { base = 25 }
		scope:ftr_court_dissenter = {
			trigger_event = {
				id = ftr_court_struggle.0012
				days = 5
			}
		}
	}
}

######### 0016 首次提醒（root = 新君） #########
# 可立刻任命（按才干分派官方五席），也可推迟一次（180-360 天后终裁）。
ftr_court_struggle.0016 = {
	type = character_event
	title = ftr_court_struggle.0016.t
	desc = ftr_court_struggle.0016.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:ftr_court_promised_vassal
		animation = happiness
	}

	# 无欠账则事件静默取消
	trigger = {
		has_variable = ftr_court_owes_council
		var:ftr_court_owes_council > 0
		OR = {
			any_vassal = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
			any_courtier = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
		}
	}

	immediate = {
		# 取一名被承诺者（先封臣后廷臣）
		random_vassal = {
			limit = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
			save_scope_as = ftr_court_promised_vassal
		}
		if = {
			limit = { NOT = { exists = scope:ftr_court_promised_vassal } }
			random_courtier = {
				limit = {
					has_variable = ftr_court_promise_council
					var:ftr_court_promised_by ?= { this = prev }
				}
				save_scope_as = ftr_court_promised_vassal
			}
		}
	}

	option = { # Grant them the promised seat
		name = ftr_court_struggle.0016.a
		ai_chance = {
			base = 60
			modifier = { add = 30  has_trait = just }
			modifier = { add = -40  has_trait = deceitful }
		}
		trigger = { exists = scope:ftr_court_promised_vassal }
		ftr_court_appoint_promised_councillor_effect = { TARGET = scope:ftr_court_promised_vassal }
		# 还有别的欠账则串行下一个提醒
		ftr_court_queue_next_council_reminder_effect = yes
	}
	option = { # Postpone it — they will not wait forever
		name = ftr_court_struggle.0016.b
		ai_chance = {
			base = 40
			modifier = { add = 40  has_trait = deceitful }
			modifier = { add = 20  has_trait = impatient }
			modifier = { add = -30  has_trait = just }
		}
		trigger = { exists = scope:ftr_court_promised_vassal }
		# 唯一一次推迟：终裁事件只给任命或拒绝
		trigger_event = {
			id = ftr_court_struggle.0017
			days = { ftr_court_council_final_delay_min_value  ftr_court_council_final_delay_max_value }
		}
	}
