##############################################################
### 朝堂权力角逐 COURT STRUGGLE
### 挂在独立王国/帝国/霸权的最高统治者（story owner）身上
###
### 议题1 继任之争：竞争者以「竞争力评分」排名，取前五名进入朝堂名单
### 议题2 共治夺权：以官方 diarchy swing（diarchy_swing）为拉锯轴
###
### 烈度（var:ftr_court_struggle_intensity，0-200）替代原离散 ftr_court_phase：
###   0-40   平静/潜伏     40-90  酝酿
###   90-140 对立（0014 站队 / 对立脉冲）
###   140-170 濒危（危机脉冲 / 政变·内战出口开启，story 图标亮剑）
###   170-200 临界（0018 废黜昏君等最强手段）
###
### 作用域约定：root / scope:story = 故事本体；owner = 在位君主
##############################################################

ftr_court_struggle = {
	visible = yes

	################## 生命周期 ##################

	on_setup = {
		ftr_court_setup_effect = yes
		# 朝堂权力角逐开始：通知最高统治者。
		story_owner = {
			add_character_flag = { flag = ftr_court_notified }
			trigger_event = { id = ftr_court_struggle.0019 }
		}
	}

	on_owner_death = {
		# 君主驾崩：若存在主要继承人则把故事移交给他（角逐随新朝延续）；
		# 若主要继承人本人正是角逐参与者，继承大位后自动从竞争者名单中除名。
		# 无主要继承人时才收场（清理统一交给 on_end 处理，此处不重复清理）。
		ftr_court_owner_death_effect = yes
	}

	on_end = {
		# 落幕通知必须赶在 cleanup 之前：cleanup 会清空参与者名单与 story 变量，
		# 通知 effect 依赖名单仍在（遍历在世竞争者逐一发 0021）。
		ftr_court_notify_struggle_end_effect = yes
		ftr_court_cleanup_effect = yes
	}

	################## 外观 ##################

	# 烈度进入濒危/危机区（>=140）改用剑图标，其余沿用宫廷政治图标
	icon = {
		trigger = {
			has_variable = ftr_court_struggle_intensity
			var:ftr_court_struggle_intensity >= ftr_court_intensity_high_bound_value
		}
		reference = "gfx/interface/icons/story_cycles/story_icon_conqueror.dds"
	}
	icon = {
		reference = "gfx/interface/icons/story_cycles/story_icon_palace_politics.dds"
	}

	background = {
		reference = "gfx/interface/illustrations/event_scenes/corridor_day.dds"
	}

	################## 面板可视化 ##################

	visualization = {
		# 继任竞争者名单（最多五人）
		character_list = {
			variable_name = "ftr_court_participant_list"
			label = "FTR_COURT_CLAIMANTS"
		}

		# 朝堂烈度进度条（0 平静 ↔ 200 临界；随烈度升高逼近政变/内战）
		basic_counter = {
			variable_name = "ftr_court_struggle_intensity"
			min = 0
			max = 200
			label = "FTR_COURT_INTENSITY"
			min_label = "ftr_court_ui_intensity_quiet"
			max_label = "ftr_court_ui_intensity_critical"
		}

		custom_string_key = "FTR_COURT_STRINGS"

		# 缓和内部矛盾决议入口：由官方 story 面板渲染为决策按钮，
		# 君主（story owner）可直接点击办理，引擎自动处理可点/灰显与执行。
		decisions = {
			ftr_court_ease_struggle_decision
		}
	}

	################## 年度维护：拉锯回落、烈度 ##################

	effect_group = {
		years = 1
		triggered_effect = {
			trigger = {
				story_owner ?= { ftr_court_struggle_valid_trigger = yes }
			}
			effect = {
				# 议题2 权力天平交由官方 diarchy swing 的 swing_balance 机制自然维护
				# 烈度年度维护（自然回落 + 君主状态偏置；替代原 phase 定档）
				ftr_court_refresh_intensity_effect = yes
			}
		}
		# 收场收紧：仅当持有者不再是最高领主（失去独立 / 等级跌至王国以下）时才 end_story；
		# 被囚禁、战争、暂时无继承顺位等临时失效不在此收场（由其他 effect_group 各自 fallback 同判）
		fallback = {
			ftr_court_end_story_if_owner_deposed_effect = yes
		}
	}

	################## 参与者询问刷新（30-60 天随机） ##################

	# 定期重新询问：给之前未参与且未明确表示不参与的潜在继承人一次选择机会。
	effect_group = {
		days = { ftr_court_refresh_interval_min_value  ftr_court_refresh_interval_max_value }
		triggered_effect = {
			trigger = {
				story_owner ?= { ftr_court_struggle_valid_trigger = yes }
			}
			effect = {
				ftr_court_launch_participation_asks_effect = yes
			}
		}
		fallback = {
			ftr_court_end_story_if_owner_deposed_effect = yes
		}
	}

	################## 内斗主时钟（60-120 天，§12.4.1） ##################

	# 内斗主时钟（60-120 天，§12.4.1）：只留心跳——守卫通过后把"这一拍选哪一族出手"交给
	# on_action 随机池 ftr_court_struggle_main_pulse（参考原版 pet_dog 机制：心跳 +
	# story_owner = { trigger_event = { on_action = X } } + random_events 加权表；
	# 池内事件可直接访问 scope:story）。各族入口（互殴手段族 / 强出口族 / 无地拉锯 /
	# 共治反击）作为 hidden 族调度事件挂在池上，由引擎按权重每拍随机选一族执行，
	# 新增事件链只需往 on_action 池登记即可接入。缓和决议休战期内不触发心跳。
	effect_group = {
		days = { ftr_court_strife_interval_min_value  ftr_court_strife_interval_max_value }
		triggered_effect = {
			trigger = {
				story_owner ?= { ftr_court_struggle_valid_trigger = yes }
				ftr_court_has_active_struggle_trigger = yes
				has_variable = ftr_court_struggle_intensity
				var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value
				story_owner ?= {
					NOT = { has_character_flag = ftr_court_eased_struggle }
				}
			}
			effect = {
				# 心跳拍：直接把"选族"交给 on_action 随机池（参考原版 pet_dog：
				# story_owner = { trigger_event = { on_action = X } }，池内事件可直接访问 scope:story）
				story_owner = {
					trigger_event = { on_action = ftr_court_struggle_main_pulse }
				}
			}
		}
		fallback = {
			ftr_court_end_story_if_owner_deposed_effect = yes
		}
	}

	################## 潜伏点缀（0001，静默期氛围） ##################

	# 低烈度（<90）时偶尔以 0001 点缀朝堂氛围；由内斗主时钟的触发校验天然保证
	# 互殴/强出口只在烈度 ≥90 后出现，此处独立小轮负责安静期（0-90）的点缀。
	effect_group = {
		days = { ftr_court_strife_interval_min_value  ftr_court_strife_interval_max_value }
		random_valid = {
			triggered_effect = {
				trigger = {
					NOT = { ftr_court_is_contested_trigger = yes }
					story_owner ?= { ftr_court_has_succession_order_trigger = yes }
				}
				effect = {
					random = { chance = 8  story_owner = { trigger_event = { id = ftr_court_struggle.0001 } } }
				}
			}
		}
		fallback = {
			ftr_court_end_story_if_owner_deposed_effect = yes
		}
	}
}
