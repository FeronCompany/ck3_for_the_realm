######### 撤回支持，重归中立 #########
ftr_court_withdraw_backing_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_withdraw_backing_interaction_desc
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = close_family
		max = 1
	}
	ai_frequency = 24
	is_shown = {
		scope:actor = {
			# 无地廷臣无法主动发起交互（站队走事件 0014，撤回暂无事件路径）
			is_landed = yes
			has_variable = ftr_court_backer_of
			var:ftr_court_backer_of ?= { this = scope:recipient }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			remove_variable = ftr_court_backer_of
			# 主动撤回支持 = 主动结清拥立账（功勋好感随之清零）
			ftr_personal_merit_backing_clear_effect = yes
			add_opinion = { target = scope:recipient  modifier = ftr_political_opponent_opinion }
		}
		# 从被支持者的个人支持者列表移除自己（新方案：支持者列表存于被支持者身上）
		scope:recipient = {
			ftr_court_remove_backer_effect = { BACKER = scope:actor }
			add_opinion = { target = scope:actor  modifier = ftr_political_opponent_opinion }
		}
	}
	on_decline = {}
	ai_will_do = {
		base = -30
		modifier = {
			add = 40
			opinion = { target = scope:recipient  value < 0 }
		}
		# 多疑者风声不对先抽身自保，不愿与人捆绑过深
		modifier = {
			add = 15
			scope:actor = { has_trait = paranoid }
		}
	}

	ai_accept = {
		# 强留一个离心的人弊大于利：总体放行，仅在惜才/急需实力时挽留
		base = 50
		# 被支持者越器重此人越惜才，越不愿放人（好感高 → 接受度降）
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = -0.5
		}
		# 竞逐正酣、局面胶着，缺不得爪牙
		modifier = {
			desc = ftr_court_accept_withdraw_struggle
			add = -25
			scope:recipient = {
				any_owned_story = {
					story_type = ftr_court_struggle
					has_variable = ftr_court_struggle_intensity
					var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value
				}
			}
		}
		# 野心勃勃者绝不轻易让麾下力量流失
		modifier = {
			desc = ftr_court_accept_withdraw_ambitious
			add = -20
			scope:recipient = { has_trait = ambitious }
		}
		# 刚愎自用者拉不下脸，宁可挽留也不愿见人转投
		modifier = {
			desc = ftr_court_accept_withdraw_arrogant
			add = -15
			scope:recipient = { has_trait = arrogant }
		}
		# 多疑者怕强留之人怀恨反噬，宁愿好聚好散
		modifier = {
			desc = ftr_court_accept_withdraw_paranoid
			add = 15
			scope:recipient = { has_trait = paranoid }
		}
		# 随遇而安者不强留人
		modifier = {
			desc = ftr_court_accept_withdraw_content
			add = 10
			scope:recipient = { has_trait = content }
		}
	}
}

######### 辅政：尽心王事，防止大权旁落 #########
ftr_court_assist_monarch_interaction = {
	category = interaction_category_vassal
	common_interaction = yes
	desc = ftr_court_assist_monarch_interaction_desc
	notification_text = ftr_court_assist_monarch_interaction_notification
	greeting = positive
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = liege
	}
	ai_frequency = 36
	is_shown = {
		scope:actor = {
			OR = {
				is_councillor_of = scope:recipient
				is_powerful_vassal_of = scope:recipient
			}
		}
		scope:recipient = {
			ftr_court_is_struggle_monarch_trigger = yes
			ftr_court_monarch_infirm_trigger = yes
			# 最高统治者必须存在共治者（辅政针对君主与共治者的权力拉锯）
			has_active_diarchy = yes
			# 发起者不能是共治者本人（共治者不向自己辅政）
			NOT = { diarch = scope:actor }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			NOT = { has_character_flag = ftr_court_assisted_this_year }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_assist_monarch_prestige_cost_value
				INFLUENCE = ftr_court_assist_monarch_influence_cost_value
			}
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_assist_monarch_prestige_cost_value
				INFLUENCE = ftr_court_assist_monarch_influence_cost_value
			}
			add_character_flag = { flag = ftr_court_assisted_this_year  years = 1 }
		}
		# 拉锯向君主一侧移动（官方 diarchy swing：负值 = liege 得势）
		scope:recipient = {
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				save_scope_as = ftr_court_story
			}
			add_opinion = { target = scope:actor  modifier = ftr_court_loyalist_opinion }
			update_diarchy_swing_with_perspective_effect = {
				SWING = minor_sop_swing_liege_gain
				LIEGE = scope:recipient
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = 0
		# 与最高统治者的关系（意见修正值）：君主越器重发起者，发起者越愿辅政
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		# 与共治者的关系（意见修正值）：共治者越器重发起者，发起者越不愿辅政打压共治者
		# 负权重稍大：并非每个臣子都愿冒风险做忠臣
		opinion_modifier = {
			who = scope:recipient.diarch
			opinion_target = scope:actor
			multiplier = -1.5
			desc = AI_OPINION_REASON
		}
		# 有忠信特质：天性忠贞，愿辅佐君主
		modifier = {
			add = 40
			scope:actor = { has_trait = loyal }
		}
		# 与共治者是仇敌：借辅政打压共治者
		modifier = {
			add = 40
			scope:actor = { has_relation_rival = scope:recipient.diarch }
		}
		# 多疑者恐出头勤王反遭共治者及其党羽记恨
		modifier = {
			add = -15
			scope:actor = { has_trait = paranoid }
		}
	}

	ai_accept = {
		# 有人勤王助己压制共治者，君主通常乐见，但需掂量来人是否可信
		base = 30
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.75
		}
		# 素来忠贞的臣下值得倚重
		modifier = {
			desc = ftr_court_accept_assist_loyal
			add = 20
			scope:actor = { has_trait = loyal }
		}
		# 关系紧密者更可信
		modifier = {
			desc = ftr_court_accept_assist_allied
			add = 15
			scope:actor = { is_allied_to = scope:recipient }
		}
		# 曾公开支持他人者「勤王」或另有盘算
		modifier = {
			desc = ftr_court_accept_assist_backer
			add = -40
			scope:actor = { has_variable = ftr_court_backer_of }
		}
		# 曾因背立被君主警告过者不可轻信
		modifier = {
			desc = ftr_court_accept_assist_warned
			add = -30
			scope:actor = { has_character_flag = ftr_liege_warn_not_support_heir }
		}
		# 花言巧语者未必真心
		modifier = {
			desc = ftr_court_accept_assist_deceitful
			add = -15
			scope:actor = { has_trait = deceitful }
		}
		# 野心过盛者借辅政近身揽权，须防
		modifier = {
			desc = ftr_court_accept_assist_ambitious
			add = -15
			scope:actor = { has_trait = ambitious }
		}
		# 多疑的君主对来投者格外审慎
		modifier = {
			desc = ftr_court_accept_assist_paranoid
			add = -10
			scope:recipient = { has_trait = paranoid }
		}
	}
}

	cooldown_against_recipient = { years = 3 }
	ai_targets = {

		scope:recipient = {
			ftr_court_is_struggle_monarch_trigger = yes

		}

			OR = {
				is_councillor_of = scope:recipient

			is_capable_adult = yes

				OTHER = scope:recipient
				PRESTIGE = ftr_court_request_coregency_prestige_cost_value

		}
		# 需已积累足够的朝堂倾向
		scope:recipient = {
			any_owned_story = {

		}
	}
	can_send = {}

			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient

		base = -60
		modifier = {

			scope:actor = { has_trait = ambitious }
		}

			add = 25

		}

			add = -20
			scope:actor = { has_trait = paranoid }
		}
	}

		# 分权大位非同小可：仅对信得过、且实在需要其分忧的人松口
		base = -40
		opinion_modifier = {
			who = scope:recipient

		}

######### 游说：拉拢同僚推动权力天平（议题 2 的主动博弈） #########
# 权臣阵营借此推高天平，保皇阵营借此巩固人心。
# 被游说者可以是封臣或无地廷臣——廷臣由此获得被动参与路径。
ftr_court_lobby_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_lobby_interaction_desc
	cooldown_against_recipient = { years = 1 }

	send_option = {
		flag = lobby_diarch
		localization = ftr_court_lobby_diarch
	}
	send_option = {
		flag = lobby_monarch
		localization = ftr_court_lobby_monarch
	}
	send_options_exclusive = yes

	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 2
	}
	ai_frequency = 12

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			# 无地廷臣不能主动发起交互
			is_landed = yes
			NOT = { ftr_court_is_struggle_monarch_trigger = yes }
			# 游说是围绕"共治权力天平"拉拢人心：领主（朝堂故事 owner）须有共治者才谈得上推拉
			# 双方处于同一朝堂（故事挂在君主身上），且该领主处于共治状态
			OR = {
				AND = {
					exists = top_liege
					top_liege = {
						any_owned_story = { type = ftr_court_struggle }
						has_active_diarchy = yes
					}
				}
				AND = {
					exists = employer
					employer = {
						any_owned_story = { type = ftr_court_struggle }
						has_active_diarchy = yes
					}
				}
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_lobby_prestige_cost_value
				INFLUENCE = ftr_court_lobby_influence_cost_value
			}
		}
		scope:recipient = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_lobby_prestige_cost_value
				INFLUENCE = ftr_court_lobby_influence_cost_value
			}
		}

		# 定位所属朝堂
		scope:actor.top_liege ?= {
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				save_scope_as = ftr_court_story
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:ftr_court_story }
				exists = scope:actor.employer
			}
			scope:actor.employer = {
				every_owned_story = {
					limit = { story_type = ftr_court_struggle }
					save_scope_as = ftr_court_story
				}
			}
		}

		if = {
			limit = { exists = scope:ftr_court_story }
			scope:ftr_court_story = {
				story_owner = { save_scope_as = ftr_court_lobby_monarch }
			}
		}
		# 官方 diarchy swing：劝进 +5（diarch 侧）、劝忠 −5（liege 侧）
		scope:ftr_court_lobby_monarch ?= {
			if = {
				limit = { scope:actor = { has_character_flag = lobby_diarch } }
				# 劝进：游说对象被拉向共治阵营
				update_diarchy_swing_with_perspective_effect = {
					SWING = minor_sop_swing_diarch_gain
					LIEGE = scope:ftr_court_lobby_monarch
				}
			}
			else = {
				# 劝忠：游说对象被拉回君主阵营
				update_diarchy_swing_with_perspective_effect = {
					SWING = minor_sop_swing_liege_gain
					LIEGE = scope:ftr_court_lobby_monarch
				}
			}
		}

		# 人情记账
		if = {
			limit = { scope:actor = { has_character_flag = lobby_diarch } }
			scope:recipient = {
				add_opinion = { target = scope:actor  modifier = ftr_political_alliance_opinion }
			}
		}
		else = {
			scope:actor.top_liege ?= {
				save_scope_as = ftr_court_monarch
			}
			scope:recipient = {
				add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_court_loyalist_opinion }
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = 0
		modifier = {
			add = 25
			scope:actor = { has_trait = ambitious }
		}
		modifier = {
			add = 25
			scope:actor = { has_trait = gregarious }
		}
		modifier = {
			add = -30
			scope:actor = { has_trait = shy }
		}
		# 多疑者忌惮高调拉人引人侧目，亦难辨游说对象的真心
		modifier = {
			add = -15
			scope:actor = { has_trait = paranoid }
		}
		# —— 选项感知（按 AI 拟采用的劝进/劝忠方案）——
		# 拟劝进：共治者本人，或已拥立共治者者
		modifier = {
			add = 15
			scope:lobby_diarch = yes
			scope:actor = {
				OR = {
					this = scope:actor.top_liege.diarch
					AND = {
						has_variable = ftr_court_backer_of
						var:ftr_court_backer_of ?= { this = scope:actor.top_liege.diarch }
					}
				}
			}
		}
		# 拟劝忠：与君主相善者更愿为君张目
		modifier = {
			add = 10
			scope:lobby_monarch = yes
			scope:actor = { opinion = { target = scope:actor.top_liege  value > 0 } }
		}
	}

	ai_accept = {
		# 被游说改变朝堂立场是件大事：没想法的人看交情，有立场的人看利益

##############################################################
### 朝堂权力角逐 — 劝人背书（02 §3.2 / §3.3）
###
### actor = 已站队的支持者；向另一名未站队的朝堂成员推销自己的主君（权贵 C）。
### 被劝者同意背书 C → 按"被劝背书 +5"计功（apply_backing 默认 +10 后再回调 5）；
### 劝说者本人同记 +5 拥立功（并入其既有拥立账）。
##############################################################
ftr_court_persuade_backing_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_persuade_backing_interaction_desc
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 3
	}
	ai_frequency = 12
	is_shown = {
		scope:actor = {
			is_capable_adult = yes
			is_landed = yes
			NOT = { is_imprisoned = yes }
			has_variable = ftr_court_backer_of
			# 同一朝堂角逐进行中
			any_owned_story = {
				story_type = ftr_court_struggle
			}
		}
		scope:recipient = {
			is_alive = yes
			is_capable_adult = yes
			NOT = { this = scope:actor }
			NOT = { has_variable = ftr_court_backer_of }
			OR = {
				is_vassal_of = scope:actor.top_liege
				NOT = { is_landed = yes }
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
		scope:recipient = { NOT = { is_imprisoned = yes } }
	}
	can_send = {}

	on_accept = {
		# 劝说者所支持的权贵 C
		scope:actor = {
			var:ftr_court_backer_of ?= {
				save_scope_as = ftr_pm_persuaded_claimant
			}
		}
		if = {
			limit = { exists = scope:ftr_pm_persuaded_claimant }
			# 被劝者背书 C（apply_backing 全套记账），功勋回调为"被劝 +5"
			scope:recipient = {
				ftr_court_apply_backing_effect = { CLAIMANT = scope:ftr_pm_persuaded_claimant }
				ftr_personal_merit_backing_deduct_effect = { AMOUNT = ftr_personal_merit_backing_active_premium_value }
			}
			# 劝说者：对同一权贵 +5 拥立功（并入既有账）
			scope:actor = {
				ftr_personal_merit_backing_add_effect = {
					CLAIMANT = scope:ftr_pm_persuaded_claimant
					AMOUNT = ftr_personal_merit_persuading_value
				}
			}
		}
	}

	on_decline = {}

	ai_will_do = {
		base = 10
		modifier = {
			add = 20
			scope:actor = { has_trait = ambitious }
		}
		modifier = {
			add = -20
			scope:actor = { has_trait = content }
		}
	}
	ai_accept = {
		base = 0
		modifier = {
			add = 30
			opinion = { target = scope:actor  value >= 25 }
		}
		modifier = {
			add = 20
			scope:recipient = { has_trait = ambitious }
		}
		modifier = {
			add = -25
			scope:recipient = { has_trait = craven }
		}
	}
