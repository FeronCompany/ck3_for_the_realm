##############################################################
### 朝堂权力角逐 COURT STRUGGLE — 调度层
###
### 设计原则：
###   1) 故事一旦建立，其年度维护完全交给 story 的 effect_group，
###      不在 on_action 里重复做，避免双重开销。
###   2) on_action 只负责三件事：准入创建、候选池增量刷新、收场。
###   3) 君主驾崩后新君是否重启角逐，交由下一年的年度检查处理——
###      刻意避开继承结算的时序风险。
##############################################################

########## 一、准入：为符合条件的新君建立朝堂角逐 ##########

ftr_court_try_start_on_action = {
	trigger = {
		# 廉价前置过滤（AND 短路求值，绝大多数角色在此即退出，避免进入昂贵判定）
		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
		# 已有角逐则跳过
		NOT = { any_owned_story = { type = ftr_court_struggle } }
		# 完整准入 + 议题1触发（内部已含 valid / 非冷却 / 非战争 / 候选数）
		ftr_court_can_start_struggle_trigger = yes
		ftr_court_should_activate_succession_trigger = yes
	}
	effect = {
		create_story = ftr_court_struggle
	}
}

########## 二、收场 ##########

# 收紧后的兜底收场：仅当故事持有者不再是最高领主（失去独立 / 等级跌至王国以下）才收场；
# 被囚禁、战争、暂时无继承顺位等临时失效一概保留（由 story 的 effect_group fallback 同判）。
# story owner 驾崩无继承的收场由 on_owner_death → end_story 独立处理，不在此重复。
ftr_court_validate_on_action = {
	trigger = {
		# 身上确实挂着朝堂故事，且持有者已非最高领主才需要兜底收场
		any_owned_story = { type = ftr_court_struggle }
		is_alive = yes
		NOT = { ftr_court_story_owner_is_supreme_liege_trigger = yes }
	}
	effect = {
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			end_story = yes
		}
	}
}

######################################################
### 挂载到原版 on_action
######################################################

# 每年一次的全局脉冲：负责准入创建与失效收场
random_yearly_everyone_pulse = {
	on_actions = {
		ftr_court_try_start_on_action
		ftr_court_validate_on_action
	}
}


# 角色死亡时的朝堂名单清理（root = 死者）。
# 任何角色死亡都可能涉入朝堂（作为支持者或参与者），用变量前置过滤，
# 仅当死者身上确有朝堂关系（支持者登记 / 所属政权 backlink）才进入 effect。
ftr_court_death_cleanup_on_action = {
	trigger = {
		OR = {
			has_variable = ftr_court_backer_of
			has_variable = ftr_court_participant_of
		}
	}
	effect = {
		ftr_court_death_cleanup_effect = yes
	}
}

on_death = {
	on_actions = {
		ftr_court_death_cleanup_on_action
	}
}

######################################################
### 三、内斗主时钟选族（由 story 心跳每 60–120 天调用）
### story effect_group → story_owner = { trigger_event = { on_action = X } }
### → on_action 把求值域落到 story，用 random_list **按烈度分档**选族：
###   外层 4 档（低 / 对立 / 濒危 / 临界）由互补 trigger 互斥，档内一行为一个族，
###   权重即抽签权重，「是否已有共治」等行内条件写在该行自己的 trigger 里；
###   条件不成立的行不参与本轮抽签（候选有效性）。
###   各族内部再按既有 effect 选角 / 概率派发；新增入口只需在对应档加一行（自带 trigger）。
### 注：random_list 权重槽只接受数值字面量，故分档权重写在下方表内（不放 script_values）。
######################################################

ftr_court_struggle_main_pulse = {
	effect = {
		# root = story_owner：经其 every_owned_story 定位故事（不依赖 scope:story）
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			save_scope_as = ftr_court_pulse_story
		}
		if = {
			limit = { exists = scope:ftr_court_pulse_story }
			# 求值域：story（var:ftr_court_struggle_intensity 与共治状态在此层直接可读）
			# 权重口径（2026-09 起改为**按烈度分档**，对齐 01 §12）：
			#   外层 4 个 1 权重行由互补 trigger 互斥，每拍恒中恰一档；档内为相对权重（无需归一）。
			#   · 低烈度（<90）  ：文明 100 + 空过 100（+ 无共治时请立共治 50）——文明族为唯一常驻入口
			#   · 对立（90–139）：互殴 55 / 强出口 50 / 逼宫 50 / 拉锯 50 / 反扑 50 / 文明 50 / 空过 90
			#   · 濒危（140–169）：互殴 85 / 强出口 75 / 逼宫 50 / 拉锯 50 / 反扑 50 / 文明 30 / 空过 55
			#   · 临界（≥170）  ：互殴 120 / 强出口 95 / 逼宫 60 / 拉锯 50 / 反扑 50 / 文明 20 / 空过 25
			#   调"平静度"只动空过权重（高烈度下调空过 = 事端更密）；调某族存在感只动该行权重。
			#   注：random_list 的权重槽只接受数值字面量，故分档权重写在此处（不放 script_values）。
			scope:ftr_court_pulse_story = {
				random_list = {
					# ===== 档 ①：低烈度（<90）——文明族为唯一常驻入口 =====
					1 = {
						trigger = { var:ftr_court_struggle_intensity < ftr_court_intensity_stir_bound_value }
						random_list = {
							100 = {
								ftr_court_civil_strife_effect = yes
							}
							# 尚无共治者时开放请立共治（不受烈度封锁）
							50 = {
								trigger = { NOT = { story_owner ?= { has_active_diarchy = yes } } }
								ftr_court_backer_demand_pulse_effect = yes
							}
							# 无事发生（朝堂风平浪静的一拍；仅写 debug.log）
							100 = {
								debug_log = "FTR_PULSE[low]: quiet pulse (nothing happens)"
								debug_log_scopes = yes
							}
						}
					}
					# ===== 档 ②：对立（90–139）=====
					1 = {
						trigger = {
							var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value
							var:ftr_court_struggle_intensity < ftr_court_intensity_high_bound_value
						}
						random_list = {
							# 互殴手段族（0300 / 0410 档）：需 ≥2 名参与者才有对手可打，否则剔除、权重让给其余族
							55 = {
								trigger = { ftr_court_has_two_participants_trigger = yes }
								ftr_court_strife_slander_effect = yes
							}
							# 强出口族（0003 通知 + 0200 站队）
							50 = {
								ftr_court_strife_exit_pulse_effect = yes
							}
							# 党羽逼宫 / 请立共治（effect 内部按有无共治分流）
							50 = {
								ftr_court_backer_demand_pulse_effect = yes
							}
							# 无地拉锯（effect 内部要求天平 ≥ 中性且有共治者）
							50 = {
								ftr_court_landless_swing_effect = yes
							}
							# 共治者游说反扑入口（effect 内部要求有共治者 + 冷却允许）
							50 = {
								ftr_court_diarch_backlash_effect = yes
							}
							# 文明争斗族（M1–M6，仍保留"就事论事"的出口）：朝中需至少一名已入营党羽
							50 = {
								trigger = { ftr_court_realm_has_camp_backer_trigger = yes }
								ftr_court_civil_strife_effect = yes
							}
							# 无事发生
							90 = {
								debug_log = "FTR_PULSE[mid]: quiet pulse (nothing happens)"
								debug_log_scopes = yes
							}
						}
					}
					# ===== 档 ③：濒危（140–169）=====
					1 = {
						trigger = {
							var:ftr_court_struggle_intensity >= ftr_court_intensity_high_bound_value
							var:ftr_court_struggle_intensity < ftr_court_intensity_critical_bound_value
						}
						random_list = {
							# 互殴手段族（0310 / 0430 / 0420 档）：需 ≥2 名参与者
							85 = {
								trigger = { ftr_court_has_two_participants_trigger = yes }
								ftr_court_strife_slander_effect = yes
							}
							# 强出口族（0004 通知 + 0200 + 0140）
							75 = {
								ftr_court_strife_exit_pulse_effect = yes
							}
							50 = {
								ftr_court_backer_demand_pulse_effect = yes
							}
							50 = {
								ftr_court_landless_swing_effect = yes
							}
							50 = {
								ftr_court_diarch_backlash_effect = yes
							}
							30 = {
								trigger = { ftr_court_realm_has_camp_backer_trigger = yes }
								ftr_court_civil_strife_effect = yes
							}
							55 = {
								debug_log = "FTR_PULSE[high]: quiet pulse (nothing happens)"
								debug_log_scopes = yes
							}
						}
					}
					# ===== 档 ④：临界（≥170）——权重压向脏手段与强出口 =====
					1 = {
						trigger = { var:ftr_court_struggle_intensity >= ftr_court_intensity_critical_bound_value }
						random_list = {
							# 互殴手段族（0320 下毒 / 0430 谋杀 / 0420 绑架档）：需 ≥2 名参与者
							120 = {
								trigger = { ftr_court_has_two_participants_trigger = yes }
								ftr_court_strife_slander_effect = yes
							}
							# 强出口族（0004 + 0200 + 0140 + 0150 废黜）
							95 = {
								ftr_court_strife_exit_pulse_effect = yes
							}
							60 = {
								ftr_court_backer_demand_pulse_effect = yes
							}
							50 = {
								ftr_court_landless_swing_effect = yes
							}
							50 = {
								ftr_court_diarch_backlash_effect = yes
							}
							20 = {
								trigger = { ftr_court_realm_has_camp_backer_trigger = yes }
								ftr_court_civil_strife_effect = yes
							}
							25 = {
								debug_log = "FTR_PULSE[crit]: quiet pulse (nothing happens)"
								debug_log_scopes = yes
							}
						}
					}
				}
			}
		}
	}
}
