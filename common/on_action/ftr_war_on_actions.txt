
# 指挥官战场成长：独立命名子 on_action，仅通过原版 on_combat_end_winner 追加引用。
# 不要在 on_combat_end_winner 上直接写 effect（原版已定义，重复 effect 会顶掉原版结算并报错）。
ftr_combat_end_commander_rank_up_on_action = {
	effect = {
		# 敌我参战规模由原版 combat_side 直接暴露（side_max_army_size / side_army_size / side_strength），
		# 无需自定义 trigger：仅在敌方规模达阈值时计经验，规模越大经验越多（上下限见 script_values）。
		if = {
			limit = {
				enemy_side.side_max_army_size >= ftr_battle_commander_min_enemy_size_value
			}
			save_scope_value_as = {
				name = ftr_battle_commander_xp
				value = ftr_battle_commander_xp_value
			}
			side_commander = {
				ftr_battle_commander_rank_up_effect = { XP = scope:ftr_battle_commander_xp }
			}
		}
	}
}

# 战后补给（ftr_war_effects.txt）
ftr_combat_end_supply_on_action = {
	effect = {
		ftr_combat_end_supply_effect = yes
	}
}

on_combat_end_winner = {
	on_actions = {
		ftr_combat_end_commander_rank_up_on_action
		ftr_combat_end_supply_on_action
	}
}
