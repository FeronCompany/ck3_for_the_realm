
# ════════════════════════════════════════════════════════════════════════════
# 行政军制法（admin_army_law）—— 本 Mod 原创法律
#
# 【1.20 格式迁移】
#   法律组定义移至 common/law_groups/ftr_law_groups.txt；
#   法律本体改为扁平顶层条目，携带强制字段
#     · law_group_type = admin_army_law
#     · index
#
# 【1.20 触发器迁移】
#   government_allows = administrative 已失效（该值已从 government_rules 枚举移除），
#   改用 government_has_mechanic = administrative。
#   同时以 should_synchronize_liege_law 取代旧写法 top_liege != this / top_liege = this。
#
# 【待定（D4）】
#   本组暂不设 required_government_flag（法律组级快速过滤），
#   准入仍由各档 can_keep 的 government_has_mechanic = administrative 把关。
#   若后续希望与官方行政法组保持一致，可改为
#   required_government_flag = { government_uses_imperial_bureaucracy }。
# ════════════════════════════════════════════════════════════════════════════

@admin_army_law_cooldown_value = 5


admin_army_law_0 = {
	law_group_type = admin_army_law
	index = 0

	modifier = {
		levy_size = 0.1
	}

	can_keep = {
		government_has_mechanic = administrative
		trigger_if = {
			limit = { should_synchronize_liege_law = yes }
			liege = { has_realm_law = admin_army_law_0 }
		}
	}

	can_pass = {
		trigger_if = {
			limit = { should_synchronize_liege_law = no }
			custom_tooltip = {
				text = admin_law_cooldown_desc
				NOT = { has_variable = admin_law_cooldown }
			}
			is_at_war = no
		}
		trigger_if = {
			limit = {
				should_synchronize_liege_law = yes
				liege = { has_realm_law = admin_army_law_0 }
			}
			custom_tooltip = {
				text = imperial_bureaucracy_pending_law_change
				always = no
			}
		}
	}

	on_pass = {
		## Set cooldown
		set_variable = {
			name = admin_law_cooldown
			years = @admin_army_law_cooldown_value
		}
	}

	ai_will_do = {
		value = 0
	}
}

			}

		}
	}
