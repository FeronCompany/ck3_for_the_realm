##############################################################
### 朝堂权力角逐 COURT STRUGGLE — 条件层
### 议题1 继任之争 / 议题2 共治夺权
##############################################################

########## 一、准入：独立政权 + 等级 >= 王国 + 存在继承顺位 ##########

# 政权等级达标（王国 / 帝国 / 霸权）
ftr_court_realm_tier_valid_trigger = {
	highest_held_title_tier >= tier_kingdom
}

# 存在继承顺位：最高统治者的主头衔继承顺位列表不为空即触发
# 全面放宽：不要求候选人数、不要求继承人成年/有能力，只要主头衔有继承顺位即可。
ftr_court_has_succession_order_trigger = {
	primary_title = {
		any_title_heir = { always = yes }
	}
}

# 准入总判定：在最高统治者身上求值
ftr_court_struggle_valid_trigger = {
	is_ruler = yes
	is_independent_ruler = yes
	ftr_court_realm_tier_valid_trigger = yes
	ftr_court_has_succession_order_trigger = yes
	NOT = { is_imprisoned = yes }
}

# 冷却期内不再激活（配合 add_character_flag 使用）
ftr_court_in_cooldown_trigger = {
	has_character_flag = ftr_court_cooldown
}

# 完整准入：满足基本条件 + 不在冷却 + 没有正在进行的战争（避免叠加）
ftr_court_can_start_struggle_trigger = {
	ftr_court_struggle_valid_trigger = yes
	NOT = { ftr_court_in_cooldown_trigger = yes }
	NOT = { is_at_war = yes }
}

########## 二、竞争者资格 ##########

# 求值域：this = 候选人；依赖 scope:ftr_court_monarch
ftr_court_claimant_valid_trigger = {
	exists = scope:ftr_court_monarch
	NOT = { this = scope:ftr_court_monarch }
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	# 必须同宗族，避免外戚/廷臣乱入
	dynasty ?= {
		this = scope:ftr_court_monarch.dynasty
	}
}

########## 二之二、参与角逐系统 ##########

# 求值域：this = 潜在继任者。是否可以被询问是否参与角逐。
# 血缘继承：子女 / 兄弟姐妹（按性别继承法过滤）；选举继承：主头衔继承顺位。
# 依赖 scope:ftr_court_monarch（君主）。
ftr_court_can_be_asked_participate_trigger = {
	exists = scope:ftr_court_monarch
	is_alive = yes
	is_adult = yes
	NOT = { this = scope:ftr_court_monarch }
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	# 已明确表示不参与者不再询问
	NOT = { has_character_flag = ftr_court_declined_contest }
	# 已参与角逐者不再询问
	NOT = { has_character_flag = ftr_court_participating }
}

# 求值域：this = 潜在继任者。按性别继承法过滤的血缘继承资格。
# 依赖 scope:ftr_court_monarch（君主）。
ftr_court_hereditary_claimant_candidate_trigger = {
	exists = scope:ftr_court_monarch
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	OR = {
		# 子女
		AND = {
			is_child_of = scope:ftr_court_monarch
			ftr_court_gender_succession_allows_trigger = yes
		}
		# 兄弟姐妹（仅当君主无任何成年子女时才询问手足，参考原 UI 血缘逻辑）
		AND = {
			is_sibling_of = scope:ftr_court_monarch
			NOT = {
				scope:ftr_court_monarch = {
					any_child = { is_adult = yes }
				}
			}
			dynasty ?= { this = scope:ftr_court_monarch.dynasty }
		}
	}
}

# 求值域：this = 潜在继任者。性别继承法是否允许其角逐（仅男性 / 仅女性 / 默认男性优先）。
ftr_court_gender_succession_allows_trigger = {
	OR = {
		# 仅男性继承：仅男性
		AND = {
			primary_title ?= { has_title_law = male_only_law }
			is_male = yes
		}
		# 仅女性继承：仅女性
		AND = {
			primary_title ?= { has_title_law = female_only_law }
			is_female = yes
		}
		# 其余（male_preference / equal）：男性优先——已有成年男性子女时不询问女性，否则男女皆可
		AND = {
			NOT = { primary_title ?= { has_title_law = male_only_law } }
			NOT = { primary_title ?= { has_title_law = female_only_law } }
			OR = {
				is_male = yes
				AND = {
					is_female = yes
					NOT = {
						scope:ftr_court_monarch = {
							any_child = { is_adult = yes  is_male = yes }
						}
					}
				}
			}
		}
	}
}

# 求值域：this = 潜在继任者。选举继承顺位资格（主头衔的 title heir）。
ftr_court_elective_claimant_candidate_trigger = {
	exists = scope:ftr_court_monarch
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	scope:ftr_court_monarch = {
		primary_title = {
			any_title_heir = {
				this = prev
			}
		}
	}
}

# 求值域：this = 潜在继任者。是否仍符合参与角逐的资格（用于定期刷新移除）。
ftr_court_participant_remains_valid_trigger = {
	is_alive = yes
	is_adult = yes
	NOT = { is_imprisoned = yes }
	NOT = { has_trait = incapable }
	# 被剥夺继承权者移除
	NOT = { has_trait = disinherited }
	# 仍然参与角逐
	has_character_flag = ftr_court_participating
}

########## 三、君主状态 ##########

# 君主因年老体衰或精神异常而不能正常履职
ftr_court_monarch_infirm_trigger = {
	OR = {
		has_trait = incapable
		has_trait = lunatic
		has_trait = possessed
		has_trait = depressed
		age >= ftr_court_monarch_elderly_age_value
	}
}

# 君主权威衰弱（政变/内战的温床）
ftr_court_monarch_weak_trigger = {
	OR = {
		prestige < ftr_court_monarch_weak_prestige_value
		piety < ftr_court_monarch_weak_piety_value
		ftr_court_monarch_infirm_trigger = yes
	}
}

########## 四、阶段判定 ##########

# 候选人数量达标（数量由参与事件 0040 写入 ftr_court_candidate_count）
ftr_court_has_enough_candidates_trigger = {
	has_variable = ftr_court_candidate_count
	var:ftr_court_candidate_count >= ftr_court_min_candidates_value
}

# 竞争者众多（三人成势）
ftr_court_has_many_candidates_trigger = {
	has_variable = ftr_court_candidate_count
	var:ftr_court_candidate_count >= 3
}

# 君主域实时判定：潜在继任竞争者是否充足。
# 参考 UI 继任者收集逻辑（ftr_court_sync_courtiers_effect）——在君主域按血缘/选举分支
# 实时收集，不依赖 story 域变量 ftr_court_candidate_count（修复 on_action 准入时 story
# 尚未建立、读不到该变量导致准入恒失败的缺陷）。
# 血缘继承：有继承权的子女（含未成年）或（无成年子女时）同宗族成年手足；
# 选举继承：主头衔顺位继承者。
# 议题1 是否应当激活（在君主身上求值）
# 全面放宽：只要最高统治者的主头衔继承顺位列表非空即触发，
# 不再要求候选人数达标或君主衰弱/竞争者众多。
ftr_court_should_activate_succession_trigger = {
	ftr_court_struggle_valid_trigger = yes
	NOT = { ftr_court_in_cooldown_trigger = yes }
}

# 前两名评分接近 -> contested
ftr_court_is_contested_trigger = {
	has_variable = ftr_court_top_gap
	var:ftr_court_top_gap <= ftr_court_contested_gap_value
}

# 前两名评分极其接近 -> crisis（政变/内战窗口开启）
# 求值域：story（root = 故事本体）
ftr_court_is_crisis_trigger = {
	ftr_court_is_contested_trigger = yes
	var:ftr_court_top_gap <= ftr_court_crisis_gap_value
	story_owner ?= {
		ftr_court_monarch_weak_trigger = yes
		NOT = { ftr_court_in_cooldown_trigger = yes }
	}
}

########## 五、议题2：共治夺权 ##########

# 议题2 是否应当激活：君主不能正常履职，或已处于摄政
ftr_court_should_activate_regency_trigger = {
	ftr_court_struggle_valid_trigger = yes
	NOT = { ftr_court_in_cooldown_trigger = yes }
	OR = {
		ftr_court_monarch_infirm_trigger = yes
		has_active_diarchy = yes
	}
}

# 拉锯足以发起共治请愿
ftr_court_can_petition_coregency_trigger = {
	has_variable = ftr_court_regency_swing
	var:ftr_court_regency_swing >= ftr_court_swing_regency_threshold_value
}

# 拉锯足以尝试夺位
ftr_court_can_attempt_usurp_trigger = {
	has_variable = ftr_court_regency_swing
	var:ftr_court_regency_swing >= ftr_court_swing_usurp_threshold_value
	ftr_court_monarch_infirm_trigger = yes
}

# 辅政成功、大权稳固
ftr_court_loyalists_hold_trigger = {
	has_variable = ftr_court_regency_swing
	var:ftr_court_regency_swing <= ftr_court_swing_loyal_threshold_value
}

########## 六、行动资格 ##########

# 竞争者能否发动政变（走 coup 计谋）
# 求值域：候选人自身
ftr_court_can_launch_coup_trigger = {
	has_variable = ftr_court_support_count
	var:ftr_court_support_count >= ftr_court_min_backers_value
	ftr_court_own_struggle_in_crisis_trigger = yes
}

# 竞争者能否起兵内战
# 求值域：候选人自身
ftr_court_can_declare_succession_war_trigger = {
	has_variable = ftr_court_support_count
	var:ftr_court_support_count >= ftr_court_min_backers_value
	ftr_court_own_struggle_in_crisis_trigger = yes
	NOT = { is_at_war = yes }
	# 无地者也能起兵：CB 会为其创建临时公国
}

# 从自身出发，找到所参与的朝堂争夺并判断其是否已进入危机
# 求值域：任意朝堂成员（封臣走 top_liege；无地廷臣走 employer）
ftr_court_own_struggle_in_crisis_trigger = {
	OR = {
		top_liege ?= {
			any_owned_story = {
				type = ftr_court_struggle
				ftr_court_is_crisis_trigger = yes
			}
		}
		employer ?= {
			any_owned_story = {
				type = ftr_court_struggle
				ftr_court_is_crisis_trigger = yes
			}
		}
	}
}

# 无地竞争者的起兵资格（用于事件路径，不要求宣战 UI 可达）
ftr_court_landless_rebel_valid_trigger = {
	is_landed = no
	NOT = { is_imprisoned = yes }
	NOT = { is_at_war = yes }
	ftr_court_can_declare_succession_war_trigger = yes
}

# 无地竞争者（廷臣皇子/王子），需要创建临时公国才能参战
ftr_court_needs_temp_duchy_trigger = {
	is_landed = no
	NOT = { is_imprisoned = yes }
}

########## 七、身份判定（用于交互/决议的可见性） ##########

# 玩家是当前朝堂的最高统治者
ftr_court_is_struggle_monarch_trigger = {
	is_ruler = yes
	is_independent_ruler = yes
	ftr_court_realm_tier_valid_trigger = yes
}

# 玩家是竞争者之一（参与角逐的继任者）
ftr_court_is_active_claimant_trigger = {
	has_character_flag = ftr_court_participating
	NOT = { ftr_court_is_struggle_monarch_trigger = yes }
}

# 可被臣属支持的对象（权贵）：最高统治者 / 共治者 / 参与角逐的继任者
ftr_court_is_supported_person_trigger = {
	OR = {
		# 最高统治者本人
		ftr_court_is_struggle_monarch_trigger = yes
		# 共治者
		is_diarch = yes
		# 参与角逐的继任者
		has_character_flag = ftr_court_participating
	}
}

# 玩家有站队对象（已卷入）
ftr_court_has_backed_someone_trigger = {
	has_variable = ftr_court_backer_of
}

# 已站队且支持对象仍是权贵（用于统计支持者数量时过滤失效站队）
ftr_court_has_backing_trigger = {
	has_variable = ftr_court_backer_of
	var:ftr_court_backer_of ?= {
		ftr_court_is_supported_person_trigger = yes
	}
}

########## 八、资源支付（双轨制） ##########

# Mod 自带的 ftr_political_influence 已废弃。
# 支付规则：双方均为行政制 -> 消耗官方影响力；任一方非行政制 -> 消耗威望。
# 求值域：支付方。参数：OTHER = 对方角色；PRESTIGE / INFLUENCE = 正数成本常量。
ftr_court_can_pay_cost_trigger = {
	OR = {
		AND = {
			government_allows = administrative
			$OTHER$ = { government_allows = administrative }
			influence >= $INFLUENCE$
		}
		AND = {
			OR = {
				NOT = { government_allows = administrative }
				$OTHER$ = { NOT = { government_allows = administrative } }
			}
			prestige >= $PRESTIGE$
		}
	}
}

# 朝堂成员：君主的封臣、廷臣或宗族成员
ftr_court_is_court_member_trigger = {
	OR = {
		exists = liege
		exists = employer
	}
}

########## 九、继承类型判定（按政体/继承法区分继任竞争者获取方式） ##########

# 血缘继承（分割继承 / 长子幼子单继承人）：
# 竞争者优先为血亲——有继承权的子女 + 同宗族亲兄弟姐妹。
# 求值域：君主。
ftr_court_hereditary_succession_trigger = {
	OR = {
		# 分割继承（封臣分割 / 氏族系列）
		has_partition_succession_realm_law_trigger = yes
		# 单继承人（长子/幼子等，按习俗）
		has_realm_law = single_heir_succession_law
	}
}

# 选举 / 继承不稳定（选侯、部分头衔选举、欢呼继承）：
# 直接按头衔继承顺位（every_heir）获取。
# 求值域：君主。
ftr_court_elective_succession_trigger = {
	OR = {
		# 任一最高头衔采用选举继承法
		any_title = {
			OR = {
				has_title_law = feudal_elective_succession_law
				has_title_law = princely_elective_succession_law
				has_title_law = saxon_elective_succession_law
				has_title_law = scandinavian_elective_succession_law
				has_title_law = gaelic_elective_succession_law
				has_title_law = tribal_elective_succession_law
				has_title_law = confederation_elective_succession_law
			}
		}
		# 拜占庭欢呼继承
		has_realm_law = acclamation_succession_law
	}
}

# 是否已有至少一名男性继承人（成年子女），用于男性优先时的女性排除。
# 求值域：君主。
ftr_court_has_male_heir_trigger = {
	any_child = {
		is_male = yes
		is_adult = yes
	}
}

# 是否已有至少一名女性继承人（成年子女），用于女性优先时的男性排除。
# 求值域：君主。
ftr_court_has_female_heir_trigger = {
	any_child = {
		is_female = yes
		is_adult = yes
	}
}
