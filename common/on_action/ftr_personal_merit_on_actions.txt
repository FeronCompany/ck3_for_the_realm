########## 一、战斗功勋回调 ##########

	effect = {

	}
}

ftr_personal_merit_combat_loser_on_action = {
	effect = {
		ftr_personal_merit_combat_loser_effect = yes
	}
}

########## 二、年度论功回调（每年每领主 10%） ##########

ftr_personal_merit_annual_on_action = {
	trigger = {
		is_landed = yes
		is_ruler = yes
		is_capable_adult = yes
		NOT = { is_imprisoned = yes }
		NOT = { has_character_flag = ftr_personal_merit_annual_skip }
	}
	effect = {
		# 存在有功勋的直属廷臣才掷年度概率
		if = {
			limit = {
				any_courtier = {
					OR = {
						AND = {
							has_variable = ftr_personal_merit_crown_to
							var:ftr_personal_merit_crown_to ?= { this = root }
							has_variable = ftr_personal_merit_crown
							var:ftr_personal_merit_crown >= ftr_personal_merit_annual_min_value
						}
						AND = {
							has_variable = ftr_personal_merit_backing_to
							var:ftr_personal_merit_backing_to ?= { this = root }
							has_variable = ftr_personal_merit_backing
							var:ftr_personal_merit_backing >= ftr_personal_merit_annual_min_value
						}
					}
				}
			}
			random = {
				chance = ftr_personal_merit_annual_chance_value
				trigger_event = ftr_personal_merit.0001
			}
		}
	}
}
