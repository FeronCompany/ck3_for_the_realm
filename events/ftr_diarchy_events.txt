ftr_diarchy.0400 = {
	type = character_event
	title = ftr_diarchy.0400.t
	desc = ftr_diarchy.0400.desc
	theme = intrigue

	# 派发后 2 天才落地：若党羽在这两日内亡故，事件静默作废（不留 error）
	trigger = {
		is_alive = yes
	}

	left_portrait = {
		character = var:ftr_diarchy_abolished_prince
	}
	right_portrait = {
		character = var:ftr_diarchy_abolished_liege
	}

	immediate = {
		# 变量指针由废储效果写入（days = 60 自过期，此处不再手动清除——
		# 立绘 datamodel 在 immediate 之后仍会读 var:，提前清变量会让头像空白）
		var:ftr_diarchy_abolished_prince ?= { save_scope_as = ftr_diarchy_fallen_prince }
		var:ftr_diarchy_abolished_liege ?= { save_scope_as = ftr_diarchy_abolishing_liege }
	}

	option = { # 切割：退出太子党
		name = ftr_diarchy.0400.a
		custom_tooltip = { text = ftr_diarchy_0400_exit_tt }
		# 事态严重：AI 退党概率偏高（基准 65），仅在「与太子交情极深 / 天性忠直」时才倾向留守
		ai_chance = {
			base = 65
			modifier = {
				add = 15
				has_trait = craven
			}
			modifier = {
				add = 10
				has_trait = content
			}
			modifier = {
				add = -25
				has_trait = loyal
			}
			modifier = {
				add = -20
				has_trait = just
			}
			modifier = {
				add = -20
				opinion = {
					target = scope:ftr_diarchy_fallen_prince
					value >= 60
				}
			}
			modifier = {
				add = -15
				has_trait = ambitious
			}
		}
		# ① 与被废太子切割：解除背书（清 backer_of + 结清拥立账 + 从太子名单除名）
		scope:ftr_diarchy_fallen_prince ?= {
			ftr_court_remove_backer_effect = { BACKER = root }
		}
		remove_variable = ftr_court_backer_of
		ftr_personal_merit_backing_clear_effect = yes
		# ② 被废太子记恨这个转身而去的人
		scope:ftr_diarchy_fallen_prince ?= {
			add_opinion = {
				target = root
				modifier = ftr_court_betrayed_opinion
			}
			ftr_court_rival_grudge_effect = { CHARACTER = root }
		}
		# ③ 主君乐见其成：肯与废太子切割者，不在清算之列
		scope:ftr_diarchy_abolishing_liege ?= {
			add_opinion = {
				target = root
				modifier = ftr_political_alliance_opinion
			}
		}
	}

	option = { # 守着废太子不走
		name = ftr_diarchy.0400.b
		custom_tooltip = { text = ftr_diarchy_0400_stay_tt }
		ai_chance = {
			base = 35
			modifier = {
				add = 30
				opinion = {
					target = scope:ftr_diarchy_fallen_prince
					value >= 60
				}
			}
			modifier = {
				add = 15
				has_trait = loyal
			}
			modifier = {
				add = -20
				has_trait = craven
			}
		}
		# 名分上仍是他的党羽（太子已废，此背书不再推动共治天平）
		scope:ftr_diarchy_fallen_prince ?= {
			add_opinion = {
				target = root
				modifier = ftr_court_backer_opinion
			}
		}
		# 代价：主君记住了替废太子撑场的人
		scope:ftr_diarchy_abolishing_liege ?= {
			add_opinion = {
				target = root
				modifier = ftr_political_opponent_opinion
			}
			ftr_court_rival_grudge_effect = { CHARACTER = root }
		}
	}
}

	}

			base = 60

				add = 20
				has_trait = ambitious
			}
			modifier = {
				add = -20
				has_trait = content
			}
		}
	}

	option = { # 【swing < 60】委婉奏请：请地表递到君上案前，由主君裁断是否赐地

			base = 65
			modifier = {
				add = 15
				has_trait = ambitious
			}
			modifier = {
				add = -20
				has_trait = content
			}
		}
	}

	option = { # 暂不考虑（后续还会再刷）
		name = ftr_diarchy.0200.b
		ai_chance = { base = 30 }
	}

	option = { # 忠心事主（此后不再刷）
		name = ftr_diarchy.0200.c
		add_character_flag = ftr_diarchy_loyal_to_liege

ftr_diarchy.0210 = {
	type = character_event
	title = ftr_diarchy.0210.t
	desc = ftr_diarchy.0210.desc
	theme = diplomacy

	# 派发后 2 天落地：请地者亡故 / 变量过期 / 已无可划之地 / 请地者已持有头衔 → 整件事静默作废
	trigger = {
		is_alive = yes
		exists = var:ftr_diarchy_land_requester
		var:ftr_diarchy_land_requester ?= { is_alive = yes }
		# 此二者即「允其所请」的成立条件（与 ftr_diarchy_landless_reminder_trigger / 授土效果同口径）：
		# 主君无地可划、或请地者已持有头衔（含无地虚衔，划了也会被收回）→ 连「推托」都不必出现。
		ftr_diarchy_liege_has_land_to_give_trigger = yes
		var:ftr_diarchy_land_requester ?= { NOT = { any_held_title = { always = yes } } }
	}

	left_portrait = {
		character = var:ftr_diarchy_land_requester
	}
	right_portrait = {
		character = root
	}

	immediate = {
		var:ftr_diarchy_land_requester ?= { save_scope_as = ftr_diarchy_petitioning_diarch }
	}

				target = root

		}

				add = 15

			}
			modifier = {
				add = 15
				opinion = {
					target = scope:ftr_diarchy_petitioning_diarch
					value >= 40
				}
			}
			modifier = {
				add = -20
				OR = {
					has_trait = greedy
					has_trait = paranoid
					has_trait = arbitrary
				}
			}
			modifier = {
				add = -15
				opinion = {
					target = scope:ftr_diarchy_petitioning_diarch
					value <= -20
				}
			}
		}
	}

### 0710 询问主君：是否为指定继承人册立太子共治（§2.2 2026-09 重构）
### 由 ftr_diarchy_promote_adult_heir_effect（异步：01 故事主时钟 / 年度 on_action）触发，root = 主君。
### 同意 → 建立太子共治（并触发 0700 回执）；拒绝 → 被指定继承人对主君不满 + 打记账（不再重问）。
######################################################

ftr_diarchy.0710 = {
	type = character_event
	title = ftr_diarchy.0710.t
	desc = ftr_diarchy.0710.desc
	theme = diplomacy

	# 事件级过滤：确有待册立的指定继承人
	trigger = {
		is_alive = yes
		exists = designated_heir
	}

	left_portrait = { character = root }
	right_portrait = { character = designated_heir }

	immediate = {
		designated_heir ?= { save_scope_as = ftr_diarchy_corule_ask_heir }
	}

	option = { # 册立为太子共治
		name = ftr_diarchy.0710.a
		trigger = { exists = scope:ftr_diarchy_corule_ask_heir }
		remove_character_flag = ftr_diarchy_corule_ask_pending
		remove_variable = ftr_diarchy_corule_ask_declined
		ftr_diarchy_try_start_crown_prince_effect = { PRINCE = scope:ftr_diarchy_corule_ask_heir }
		# 回执：0700（告知主君 + 消息回执新太子）
		if = {
			limit = {
				has_active_diarchy = yes
				exists = diarch
			}
			trigger_event = { id = ftr_diarchy.0700  days = 1 }
		}
		ai_chance = {
			base = 70
			modifier = { add = 20  has_trait = ambitious }
			modifier = { add = -30  has_trait = content }
		}
	}

	option = { # 暂不册立
		name = ftr_diarchy.0710.b
		remove_character_flag = ftr_diarchy_corule_ask_pending
		# 被指定继承人感到被冷落（对主君不满）
		scope:ftr_diarchy_corule_ask_heir ?= {
			add_opinion = {
				target = root
				modifier = ftr_diarchy_corule_refused_opinion
			}
			# 君主拒绝册立 → 解除该继承人的任命封锁，允许其重新出仕（找工作）
			ftr_diarchy_release_appointment_block_effect = yes
		}
		# 记账：同一继承人不再重问（改储时由交互侧清除该记账）
		if = {
			limit = { exists = scope:ftr_diarchy_corule_ask_heir }
			set_variable = {
				name = ftr_diarchy_corule_ask_declined
				value = scope:ftr_diarchy_corule_ask_heir
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 30  has_trait = content }
			modifier = { add = 20  has_trait = paranoid }
		}
	}
}
