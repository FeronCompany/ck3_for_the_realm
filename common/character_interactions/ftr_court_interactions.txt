######### 撤回支持，重归中立 #########
ftr_court_withdraw_backing_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_withdraw_backing_interaction_desc
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		ai_recipients = family
		max = 1
	}
	ai_frequency = 24
	is_shown = {
		scope:actor = {
			# 无地廷臣无法主动发起交互（站队走事件 0200，撤回暂无事件路径）
			is_landed = yes
			has_variable = ftr_court_backer_of
			var:ftr_court_backer_of ?= { this = scope:recipient }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			remove_variable = ftr_court_backer_of
			# 主动撤回支持 = 主动结清拥立账（功勋好感随之清零）
			ftr_personal_merit_backing_clear_effect = yes
			add_opinion = { target = scope:recipient  modifier = ftr_political_opponent_opinion }
		}
		# 从被支持者的个人支持者列表移除自己（新方案：支持者列表存于被支持者身上）
		scope:recipient = {
			ftr_court_remove_backer_effect = { BACKER = scope:actor }
			add_opinion = { target = scope:actor  modifier = ftr_political_opponent_opinion }
		}
	}
	on_decline = {}
	# 撤回支持是单方行为，被支持者无从拒绝——直接受理，不再走 ai_accept
	auto_accept = yes
	ai_will_do = {
		base = -30
		modifier = {
			add = 40
			opinion = { target = scope:recipient  value < 20 }
		}
		# 与所支持者交情深厚者，不忍抽身而去（朋友 / 挚友 / 姻亲 → 更不易撤销）
		modifier = {
			add = -30
			has_relation_friend = scope:recipient
		}
		modifier = {
			add = -50
			has_relation_best_friend = scope:recipient
		}
		# 姻亲：配偶与对方是近亲，或本人近亲与对方结为配偶
		modifier = {
			add = -20
			OR = {
				any_spouse = { is_close_family_of = scope:recipient }
				any_close_family_member = {
					any_spouse = { this = scope:recipient }
				}
			}
		}
		# 已成仇雠者，乐见其势衰（敌人 / 死敌 / 怨恨者 → 更容易撤销）
		modifier = {
			add = 30
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = 50
			has_relation_nemesis = scope:recipient
		}
		modifier = {
			add = 25
			has_relation_grudge = scope:recipient
		}
	}
}

######### 敲打党羽：最高领主施压结党权贵 #########
# 敲打对象 = 党羽的背书对象（结党的权贵本人，非君主）：其党羽成员依性格顺从度
# （懦弱 / 忠信 / 公正 / 安于现状 / 谦卑等）与对君主的好感，各自小概率宣布退出党羽；
# 被敲打的权贵本人记恨。党首是君主的子女 / 继承人时，敲打即「惊君处置」——
# 其党羽全体被打 3 年警告 flag（原 0041 语义承接）。
ftr_court_discipline_backer_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_discipline_backer_interaction_desc
	cooldown = { years = 5 }
	ai_targets = {
		ai_recipients = vassals
		ai_recipients = courtiers
		max = 2
	}
	ai_frequency = 24
	is_shown = {
		scope:actor = {
			ftr_court_is_struggle_monarch_trigger = yes
		}
		scope:recipient = {
			top_liege = scope:actor
			NOT = { this = scope:actor }
			# 党羽的背书对象 = 持有非空支持者名单的权贵（共治者 / 参与者；名单由记账与 cleanup 维护）
			has_variable_list = ftr_court_supporters
			variable_list_size = {
				name = ftr_court_supporters
				value >= 1
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
		scope:recipient = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	on_accept = {
		# 敲打本身即羞辱：被敲打的权贵对君主心生怨怼
		scope:recipient = {
			add_opinion = { target = scope:actor  modifier = ftr_court_disciplined_opinion }
		}
		# 惊君处置：党首是君主的子女 / 继承人 → 其党羽全体被警告（3 年，抑制继续公开背书）
		scope:recipient = {
			if = {
				limit = {
					OR = {
						is_child_of = scope:actor
						is_heir_of = scope:actor
					}
				}
				every_in_list = {
					variable = ftr_court_supporters
					add_character_flag = {
						flag = ftr_liege_warn_not_support_heir
						years = 3
					}
				}
			}
		}
		# 党羽去留结算：整体包 custom_tooltip，玩家只看到一行文案（内部为标记 + 排干实现）
		custom_tooltip = {
			text = ftr_court_discipline_backer_effect_text
			# 党羽逐一权衡去留：顺从者小概率屈服（只打标记，不在遍历中修改名单）
			scope:recipient = {
				every_in_list = {
					variable = ftr_court_supporters
					random = {
						chance = ftr_court_discipline_exit_chance_value
						add_character_flag = {
							flag = ftr_court_discipline_yielded
							days = 2
						}
					}
				}
			}
			# 屈服者解除背书（清 backer_of 与拥立账），并从党首名单除名
			# （while 逐个排干：任何遍历中都不修改 ftr_court_supporters 本身）
			scope:recipient = {
				while = {
					count = 100
					limit = {
						any_in_list = {
							variable = ftr_court_supporters
							has_character_flag = ftr_court_discipline_yielded
						}
					}
					random_in_list = {
						variable = ftr_court_supporters
						limit = { has_character_flag = ftr_court_discipline_yielded }
						remove_variable = ftr_court_backer_of
						ftr_personal_merit_backing_clear_effect = yes
						save_scope_as = ftr_court_yielded_backer
					}
					ftr_court_remove_backer_effect = { BACKER = scope:ftr_court_yielded_backer }
				}
			}
			# 有人屈服才通报君主
			scope:actor = {
				if = {
					limit = { exists = scope:ftr_court_yielded_backer }
					send_interface_message = {
						title = ftr_court_discipline_success_message
						left_icon = scope:recipient
						right_icon = scope:ftr_court_yielded_backer
					}
				}
			}
		}
	}
	on_decline = {}
	auto_accept = yes
	ai_will_do = {
		base = 0
		# 朝堂竞逐正酣（烈度 ≥ 酝酿档）时更需敲打立威
		modifier = {
			add = 30
			any_owned_story = {
				story_type = ftr_court_struggle
				var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value
			}
		}
		# 多疑 / 记仇者爱敲打
		modifier = {
			add = 20
			OR = {
				has_trait = paranoid
				has_trait = vengeful
			}
		}
		# 公正 / 安于现状者不爱威压臣下
		modifier = {
			add = -25
			OR = {
				has_trait = just
				has_trait = content
			}
		}
		# 与该权贵交恶 → 更想敲打
		modifier = {
			add = 20
			opinion = { target = scope:recipient  value < 0 }
		}
	}
}

######### 请愿共治：权臣试探大位 #########
ftr_court_petition_coregency_interaction = {
	category = interaction_category_vassal
	common_interaction = yes
	desc = ftr_court_petition_coregency_interaction_desc
	cooldown_against_recipient = { years = 10 }
	ai_targets = {
		ai_recipients = liege
	}
	ai_frequency = 36
	is_shown = {
		scope:recipient = {
			ftr_court_is_struggle_monarch_trigger = yes
			ftr_court_monarch_infirm_trigger = yes
			NOT = { has_active_diarchy = yes }
		}
		scope:actor = {
			OR = {
				is_councillor_of = scope:recipient
				is_vassal_of = scope:recipient
			}
		}
	}
	is_valid_showing_failures_only = {
		# 放宽口径（2026-09）：只要「存在朝堂角逐故事 + 君主有合法继承人」即可发动。
		# 保留三项底线：能正常行动（成年可用 / 未囚）与**出得起代价**——
		# on_accept 恒走 ftr_court_pay_cost_effect，去掉支付检查会让请愿者空手取利。
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_request_coregency_prestige_cost_value
				INFLUENCE = ftr_court_request_coregency_influence_cost_value
			}
		}
		scope:recipient = {
			any_owned_story = { type = ftr_court_struggle }
			ftr_court_has_legitimate_heir_trigger = yes
			# 接收方冷却（2026-09）：同一君主 2 年内只受理一份共治请愿
			NOT = { ftr_court_petition_in_cooldown_trigger = yes }
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_request_coregency_prestige_cost_value
				INFLUENCE = ftr_court_request_coregency_influence_cost_value
			}
		}
		# 请愿结果交由事件裁决，君主可允可拒
		scope:recipient = {
			save_scope_as = ftr_court_monarch
			# 接收方冷却（2026-09）：2 年内不再受理新的共治请愿（flag 到期自动失效）
			add_character_flag = {
				flag = ftr_court_petition_cooldown
				years = ftr_court_petition_cooldown_years_value
			}
			# 朝堂通报：共治请愿已呈递
			ftr_court_send_court_toast_effect = {
				TITLE = ftr_court_notice_petition
				ICON = scope:actor
			}
			trigger_event = {
				id = ftr_court_struggle.0220
				days = 1
			}
		}
	}
	on_decline = {}
	# 君主的选择在 0220 事件内完成（可允可拒），交互侧不设 ai_accept：直接受理并进入事件
	auto_accept = yes
	ai_will_do = {
		base = -30
		# ① 首选：请愿者本人即角逐中的竞争者——请愿可直接把自己（或其背书对象）推上共治位
		modifier = {
			add = 80
			scope:actor = { ftr_court_is_active_claimant_trigger = yes }
		}
		# ② 次之：请愿者是某竞争者的党羽——为其党首请愿，落地时授权对象 = 其背书对象
		modifier = {
			add = 40
			scope:actor = {
				has_variable = ftr_court_backer_of
				var:ftr_court_backer_of ?= { ftr_court_is_active_claimant_trigger = yes }
			}
		}
		# ③ 既非竞争者、也非竞争者党羽（含保皇党羽 / 未站队者）→ 意愿稍降
		modifier = {
			add = -20
			scope:actor = {
				ftr_court_is_active_claimant_trigger = no
				NOT = {
					has_variable = ftr_court_backer_of
					var:ftr_court_backer_of ?= { ftr_court_is_active_claimant_trigger = yes }
				}
			}
		}
		# 君主身体状态不佳 → 请愿意愿**大幅上升**（2026-09；趁其无力弹压）
		modifier = {
			add = 20
			scope:recipient = { ftr_court_monarch_infirm_trigger = yes }
		}
		# 病重 / 失能：再加一档
		modifier = {
			add = 40
			scope:recipient = {
				OR = {
					is_incapable = yes
					health <= poor_health
				}
			}
		}
	}
}

##############################################################
### 朝堂权力角逐 — 劝人背书（02 §3.2 / §3.3）
###
### actor = 已站队的支持者；向另一名未站队的朝堂成员推销自己的主君（权贵 C）。
### 被劝者同意背书 C → 按"被劝背书 +5"计功（apply_backing 默认 +10 后再回调 5）；
### 劝说者本人同记 +5 拥立功（并入其既有拥立账）。
##############################################################
ftr_court_persuade_backing_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_persuade_backing_interaction_desc
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 3
	}
	ai_frequency = 12
	is_shown = {
		scope:actor = {
			is_capable_adult = yes
			is_landed = yes
			NOT = { is_imprisoned = yes }
			has_variable = ftr_court_backer_of
			# 同一朝堂角逐进行中
			any_owned_story = {
				story_type = ftr_court_struggle
			}
		}
		scope:recipient = {
			is_alive = yes
			is_capable_adult = yes
			NOT = { this = scope:actor }
			NOT = { has_variable = ftr_court_backer_of }
			OR = {
				is_vassal_of = scope:actor.top_liege
				NOT = { is_landed = yes }
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
		scope:recipient = { NOT = { is_imprisoned = yes } }
	}
	can_send = {}

	on_accept = {
		# 劝说者所支持的权贵 C
		scope:actor = {
			var:ftr_court_backer_of ?= {
				save_scope_as = ftr_pm_persuaded_claimant
			}
		}
		if = {
			limit = { exists = scope:ftr_pm_persuaded_claimant }
			# 被劝者背书 C（apply_backing 全套记账），功勋回调为"被劝 +5"
			scope:recipient = {
				ftr_court_apply_backing_effect = { CLAIMANT = scope:ftr_pm_persuaded_claimant }
				ftr_personal_merit_backing_deduct_effect = { AMOUNT = ftr_personal_merit_backing_active_premium_value }
			}
			# 劝说者：对同一权贵 +5 拥立功（并入既有账）
			scope:actor = {
				ftr_personal_merit_backing_add_effect = {
					CLAIMANT = scope:ftr_pm_persuaded_claimant
					AMOUNT = ftr_personal_merit_persuading_value
				}
			}
		}
	}

	on_decline = {}

	ai_will_do = {
		base = 10
		modifier = {
			add = 20
			scope:actor = { has_trait = ambitious }
		}
		modifier = {
			add = -20
			scope:actor = { has_trait = content }
		}
	}
	ai_accept = {
		base = 0
		modifier = {
			add = 30
			opinion = { target = scope:actor  value >= 25 }
		}
		modifier = {
			add = 20
			scope:recipient = { has_trait = ambitious }
		}
		modifier = {
			add = -25
			scope:recipient = { has_trait = craven }
		}
	}
}
