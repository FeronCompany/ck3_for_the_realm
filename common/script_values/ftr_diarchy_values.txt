# —— 背书盘点（求值域 = 君主）——
# 背书共治者的人数（封臣 + 内阁）
ftr_diarchy_diarch_backer_count_value = {
	value = 0
	if = {
		limit = { exists = diarch }
		every_vassal = {
			limit = {
				has_variable = ftr_court_backer_of
				var:ftr_court_backer_of ?= { this = prev.diarch }
			}
			add = 1
		}
		every_councillor = {
			limit = {
				has_variable = ftr_court_backer_of
				var:ftr_court_backer_of ?= { this = prev.diarch }
			}
			add = 1
		}
	}
}

### 六、拉封臣入战（§4.5）货币成本（AI 同意加权内联在交互 ai_accept）
ftr_diarchy_levy_vassal_gold_cost_value = 200
# 影响力单价（offer influence 选项的实际扣量）
ftr_diarchy_levy_vassal_influence_cost_value = 100
# 必拒者过滤：目标对共治者的好感低于此值则不出现在候选列表
ftr_diarchy_levy_vassal_min_opinion_value = -25
# 拉人代价：每激活一项 send_option（黄金 / 人情 / 影响力）的减分
ftr_diarchy_levy_vassal_per_option_penalty_value = -15
# 拉人代价：同时动用两项以上代价时的额外递增减分
ftr_diarchy_levy_vassal_multi_option_penalty_value = -15
# 拉封臣入战（§4.5）AI 意愿：send_option 激活越多越不愿出手。
# 求值域 = 共治者（交互 actor）；send_option 的 flag 在此可见（同项目既有写法）。
ftr_diarchy_levy_vassal_option_cost_value = {
	value = 0
	if = {
		limit = { scope:ftr_diarchy_levy_gold = yes }
		add = ftr_diarchy_levy_vassal_per_option_penalty_value
	}
	if = {
		limit = { scope:ftr_diarchy_levy_hook = yes }
		add = ftr_diarchy_levy_vassal_per_option_penalty_value
	}
	if = {
		limit = { scope:ftr_diarchy_levy_influence = yes }
		add = ftr_diarchy_levy_vassal_per_option_penalty_value
	}
	# 两种以上：额外递增
	if = {
		limit = {
			OR = {
				AND = {
					scope:ftr_diarchy_levy_gold = yes
					scope:ftr_diarchy_levy_hook = yes
				}
				AND = {
					scope:ftr_diarchy_levy_gold = yes
					scope:ftr_diarchy_levy_influence = yes
				}
				AND = {
					scope:ftr_diarchy_levy_hook = yes
					scope:ftr_diarchy_levy_influence = yes
				}
			}
		}
		add = ftr_diarchy_levy_vassal_multi_option_penalty_value
	}
	# 三种齐上：再加一层
	if = {
		limit = {
			scope:ftr_diarchy_levy_gold = yes
			scope:ftr_diarchy_levy_hook = yes
			scope:ftr_diarchy_levy_influence = yes
		}
		add = ftr_diarchy_levy_vassal_multi_option_penalty_value
	}
}

### 九、战争实力对比（用于「拉人意愿」判断）

# 阈值：敌我比 ≥ 2 → 我方被压制，需要拉人；≤ 0.5 → 我方压倒，不必拉人
ftr_diarchy_war_ratio_threshold_high_value = 2
ftr_diarchy_war_ratio_threshold_low_value = 0.5

# 求值域：war。攻方 / 守方总军力（原版同款写法：every_war_attacker / every_war_defender）
ftr_diarchy_war_attacker_strength_value = {
	value = 0
	every_war_attacker = { add = current_military_strength }
}
ftr_diarchy_war_defender_strength_value = {
	value = 0
	every_war_defender = { add = current_military_strength }
}

# 求值域：war；参照 = scope:actor（我方）。
# 敌我比 = 敌方军力 ÷ 我方军力（先按我在哪一侧定向，再取对方 / 我方）
ftr_diarchy_war_enemy_to_our_ratio_value = {
	value = 0
	if = {
		limit = { is_attacker = scope:actor }
		# 我方 = 攻方，敌方 = 守方
		value = ftr_diarchy_war_defender_strength_value
		divide = ftr_diarchy_war_attacker_strength_value
	}
	else_if = {
		limit = { is_defender = scope:actor }
		# 我方 = 守方，敌方 = 攻方
		value = ftr_diarchy_war_attacker_strength_value
		divide = ftr_diarchy_war_defender_strength_value
	}
}
