ftr_court_struggle.0011 = {
	type = character_event
	title = ftr_court_struggle.0011.t
	desc = ftr_court_struggle.0011.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		save_scope_as = ftr_court_new_monarch
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			save_scope_as = ftr_court_story
		}
		# 找出威胁最大的竞争者（有地、有兵、未被囚禁且非新君）——供逮捕/起兵事件使用
		scope:ftr_court_story ?= {
			ordered_in_list = {
				variable = ftr_court_participant_list
				order_by = max_military_strength
				limit = {
					NOT = { this = scope:ftr_court_new_monarch }
					is_landed = yes
					max_military_strength > 0
					NOT = { is_imprisoned = yes }
				}
				save_scope_as = ftr_court_dissenter
			}
		}
	}

	option = { # Pardon the remaining competitors
		name = ftr_court_struggle.0011.a
		ai_chance = {
			base = 40
			modifier = {
				add = 40
				has_trait = just
			}
		}
		add_prestige = ftr_court_prestige_minor_value
	}
	option = { # Strike the loudest voice first
		name = ftr_court_struggle.0011.b
		trigger = { exists = scope:ftr_court_dissenter }
		ai_chance = {
			base = 40
			modifier = { add = 35  has_trait = ambitious }
			modifier = { add = 25  has_trait = callous }
			modifier = { add = 20  has_trait = paranoid }
			modifier = { add = -30  has_trait = content }
			modifier = { add = -25  has_trait = just }
			modifier = { add = -25  has_trait = forgiving }
		}
		# 枪打出头鸟：逮捕威胁最大的对手（有地且有兵者）。
		# 成功 → 囚禁并废其竞争地位；失败 → 对方不得不举兵造反。
		random = {
			chance = ftr_court_arrest_success_base_value
			scope:ftr_court_dissenter ?= {
				ftr_court_clear_claimant_status_effect = yes
				imprison = { type = dungeon  target = scope:ftr_court_dissenter }
			}
			add_dread = ftr_court_dread_medium_value
			# 成功标记：供失败兜底判定（互斥）
			set_variable = { name = ftr_court_arrest_succeeded  value = 1 }
		}
		if = {
			limit = { NOT = { has_variable = ftr_court_arrest_succeeded } }
			# 逮捕失手：对方再无退路，被迫举兵
			scope:ftr_court_dissenter ?= {
				ftr_court_prepare_rebellion_effect = { TARGET = scope:ftr_court_new_monarch }
			}
			# 处置失败令新君威望受损，其余竞争者亦人人自危
			add_prestige = { value = 0  subtract = ftr_court_prestige_minor_value }
			scope:ftr_court_story ?= {
				every_in_list = {
					variable = ftr_court_participant_list
					limit = { NOT = { this = scope:ftr_court_dissenter } }
					add_opinion = { target = scope:ftr_court_new_monarch  modifier = ftr_political_opponent_opinion }
				}
			}
		}
		remove_variable = ftr_court_arrest_succeeded
	}
	option = { # Let them keep their titles, but watch them
		name = ftr_court_struggle.0011.c
		trigger = {
			exists = scope:ftr_court_dissenter
		}
		ai_chance = { base = 25 }
		scope:ftr_court_dissenter = {
			trigger_event = {
				id = ftr_court_struggle.0012
				days = 5
			}
		}
	}
}

######### 0014 朝堂站队（root = 被询问/主动表态的朝堂成员，可为无地廷臣） #########
# 触发源：
#   1) story 的对立/危机脉冲推送（随机询问一名未站队封臣/廷臣）；
#   2) 决策 ftr_court_declare_backing_decision（玩家/AI 主动表态，原支持交互已废除）。
# 五个支持选项不再直接建立背书：先把支持者身份暂存到被支持者身上并触发 0022，
# 由被支持者决定是否接受这份公开拥立（接受才走 ftr_court_apply_backing_effect 全套记账）。
ftr_court_struggle.0014 = {
	type = character_event
	title = ftr_court_struggle.0014.t
	desc = ftr_court_struggle.0014.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		save_scope_as = ftr_court_asked
		# 定位君主（封臣走 top_liege，廷臣走 employer）与其名下的角逐
		if = {
			limit = { exists = top_liege }
			top_liege = {
				save_scope_as = ftr_court_monarch
				every_owned_story = {
					limit = { story_type = ftr_court_struggle }
					save_scope_as = ftr_court_story
				}
			}
		}
		else_if = {
			limit = { exists = employer }
			employer = {
				save_scope_as = ftr_court_monarch
				every_owned_story = {
					limit = { story_type = ftr_court_struggle }
					save_scope_as = ftr_court_story
				}
			}
		}
		scope:ftr_court_story ?= {
			ftr_court_save_ranked_claimants_effect = yes
		}
		# 共治者作为独立可选目标（背书目标扩展）：
		#   仅当存在共治者、其不是本人、且未占前五候选位（共治者可能同时是竞争者）时设立选项。
		if = {
			limit = {
				exists = scope:ftr_court_monarch
				scope:ftr_court_monarch = { exists = diarch }
				NOT = { scope:ftr_court_monarch.diarch ?= { this = root } }
				NOT = {
					ftr_court_claimant_slots_contain_trigger = { WHO = scope:ftr_court_monarch.diarch }
				}
			}
			scope:ftr_court_monarch = {
				diarch ?= { save_scope_as = ftr_court_diarch }
			}
		}
	}

	option = { # Back the strongest
		name = ftr_court_struggle.0014.a

					save_scope_as = ftr_court_story

# ---- 手段 2.6 反馈链：共治者反击（0084，root = 共治者 diarch）----
# 由 ftr_court_diarch_backlash_effect 在主时钟派发（§9.6）：swing ≤30 时共治者阵营暗中游说拉回。
ftr_court_struggle.0084 = {
	type = character_event
	title = ftr_court_struggle.0084.t
	desc = ftr_court_struggle.0084.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = {
		character = scope:ftr_court_story_owner
		animation = personality_rational
	}

		# 定位所在朝堂 story 与君主（共治者走 top_liege / employer 兜底）
		if = {
			limit = { any_owned_story = { type = ftr_court_struggle } }
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				save_scope_as = ftr_court_story
			}
		}
		else_if = {
			limit = { exists = top_liege }
			top_liege = {
				every_owned_story = {
					limit = { story_type = ftr_court_struggle }
					save_scope_as = ftr_court_story
				}
			}
		}
		else_if = {
			limit = { exists = employer }
			employer = {
				every_owned_story = {
					limit = { story_type = ftr_court_struggle }
					save_scope_as = ftr_court_story
				}
			}
		}
		scope:ftr_court_story ?= {
			story_owner = { save_scope_as = ftr_court_story_owner }
		}
		# 泛化（03 §7.1）：无朝堂角逐 story 但存在共治关系时，用共治君主补全作用域
		# （root = 共治者本人，liege 即其君主），使本事件成为共治期的共治者反击入口。
		if = {
			limit = { NOT = { exists = scope:ftr_court_story_owner } }
			if = {
				limit = { exists = liege }
				liege = { save_scope_as = ftr_court_story_owner }
			}
			else_if = {
				limit = { exists = top_liege }
				top_liege = { save_scope_as = ftr_court_story_owner }
			}
		}
	}

		name = ftr_court_struggle.0084.a
		ai_chance = {
			base = 60
			modifier = { add = 20  has_trait = ambitious }
			modifier = { add = -20  has_trait = content }
		}
		# 密令亲信游说，把天平向自己（共治者）拉回（官方 swing 向 diarch 侧）
		scope:ftr_court_story_owner ?= {
			if = {
				limit = {
					has_active_diarchy = yes
					exists = diarch
				}
				update_diarchy_swing_with_perspective_effect = {
					SWING = minor_sop_swing_diarch_gain
					LIEGE = scope:ftr_court_story_owner
				}
			}
		}
	}
	option = { # Endure for now
		name = ftr_court_struggle.0084.b
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = patient }
			modifier = { add = -20  has_trait = ambitious }
		}
	}
}

###### 0085 党羽逼宫：请立共治（root = 最高领主） ######
# 某竞争者的党羽规模已达门槛、且朝中尚无共治者时，党羽集体要求君主接纳其拥立对象共治。
ftr_court_struggle.0085 = {
	type = character_event
	title = ftr_court_struggle.0085.t
	desc = ftr_court_struggle.0085.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = var:ftr_court_demand_champion
		animation = personality_bold
	}

	trigger = {
		exists = var:ftr_court_demand_champion
		var:ftr_court_demand_champion ?= { is_alive = yes }
		NOT = { has_active_diarchy = yes }
	}

	immediate = {
		save_scope_as = ftr_court_demand_monarch
		var:ftr_court_demand_champion ?= { save_scope_as = ftr_court_demand_champion }
		var:ftr_court_demand_agitator ?= { save_scope_as = ftr_court_demand_agitator }
	}

	option = { # 允准分权：接纳其为共治者
		name = ftr_court_struggle.0085.a
		ai_chance = {
			base = 25
			modifier = { add = 40  ftr_court_monarch_infirm_trigger = yes }
			modifier = { add = -50  has_trait = arrogant }
			modifier = { add = -30  has_trait = ambitious }
		}
		# 按政体分流建立共治（东罗马共治皇帝 / 行政制太子 / 其他共治君主 / 缺省固化摄政）
		ftr_court_establish_champion_coregency_effect = { CHAMPION = scope:ftr_court_demand_champion }
		# 党羽如愿：全体党羽与拥立对象对君主好感提升
		scope:ftr_court_demand_champion ?= {
			every_in_list = {
				variable = ftr_court_supporters
				add_opinion = { target = scope:ftr_court_demand_monarch  modifier = ftr_political_alliance_opinion }
			}
			add_opinion = { target = scope:ftr_court_demand_monarch  modifier = ftr_political_alliance_opinion }
		}
		# 请立共治抬升朝堂烈度（夹紧上限）
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			if = {
				limit = { has_variable = ftr_court_struggle_intensity }
				change_variable = { name = ftr_court_struggle_intensity  add = ftr_court_backer_demand_intensity_value }
				if = {
					limit = { var:ftr_court_struggle_intensity > ftr_court_intensity_max_value }
					set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_max_value }
				}
			}
		}
	}

	option = { # 断然拒绝：独揽大权
		name = ftr_court_struggle.0085.b
		ai_chance = {
			base = 60
			modifier = { add = 40  has_trait = arrogant }
			modifier = { add = 20  has_trait = ambitious }
		}
		# 党羽受挫：全体党羽与拥立对象对君主好感下降
		scope:ftr_court_demand_champion ?= {
			every_in_list = {
				variable = ftr_court_supporters
				add_opinion = { target = scope:ftr_court_demand_monarch  modifier = ftr_political_opponent_opinion }
			}
			add_opinion = { target = scope:ftr_court_demand_monarch  modifier = ftr_political_opponent_opinion }
		}
		# 拒绝激化对立：烈度上升（夹紧上限）
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			if = {
				limit = { has_variable = ftr_court_struggle_intensity }
				change_variable = { name = ftr_court_struggle_intensity  add = ftr_court_backer_demand_intensity_value }
				if = {
					limit = { var:ftr_court_struggle_intensity > ftr_court_intensity_max_value }
					set_variable = { name = ftr_court_struggle_intensity  value = ftr_court_intensity_max_value }
				}
			}
		}
	}
}
