namespace = ftr_personal_merit

##############################################################
### 个人功勋 PERSONAL MERIT — 年度论功（02 §4.4）
###
### root = 现任最高领主；immediate 挑选一名有功勋(≥10)的直属廷臣。
### 选项：a 赏钱 / b 授以随机非首都伯爵领 / c 授以内阁职位 / d 从简。
##############################################################

ftr_personal_merit.0001 = {
	type = character_event
	title = ftr_personal_merit.0001.t
	desc = ftr_personal_merit.0001.desc
	theme = realm

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		save_scope_as = ftr_personal_merit_monarch
		# 随机挑一名对 root 持有可兑功勋(≥10)的直属廷臣
		random_courtier = {
			limit = {
				OR = {
					AND = {
						has_variable = ftr_personal_merit_crown_to
						var:ftr_personal_merit_crown_to ?= { this = root }
						has_variable = ftr_personal_merit_crown
						var:ftr_personal_merit_crown >= ftr_personal_merit_annual_min_value
					}
					AND = {
						has_variable = ftr_personal_merit_backing_to
						var:ftr_personal_merit_backing_to ?= { this = root }
						has_variable = ftr_personal_merit_backing
						var:ftr_personal_merit_backing >= ftr_personal_merit_annual_min_value
					}
				}
			}
			save_scope_as = ftr_pm_subject
		}
	}

	option = { # Bestow a purse of gold
		name = ftr_personal_merit.0001.a
		trigger = { exists = scope:ftr_pm_subject }
		ai_chance = {
			base = 45
			modifier = {
				add = 20
				gold >= 600
			}
		}
		# 中档赏钱（400 金 / 耗 30 功勋）
		add_gold = {
			value = 0
			subtract = ftr_personal_merit_annual_gold_value
		}
		scope:ftr_pm_subject = {
			add_gold = ftr_personal_merit_annual_gold_value
			ftr_personal_merit_consume_effect = { AMOUNT = ftr_personal_merit_annual_gold_cost_value }
			add_opinion = { target = scope:ftr_personal_merit_monarch  modifier = grateful_opinion  opinion = 20 }
		}
	}

	option = { # Grant them a council post
		name = ftr_personal_merit.0001.c
		trigger = { exists = scope:ftr_pm_subject }
		ai_chance = {
			base = 25
			modifier = {
				add = 25
				scope:ftr_pm_subject = { has_trait = ambitious }
			}
		}
		ftr_personal_merit_grant_council_position_effect = { TARGET = scope:ftr_pm_subject }
		scope:ftr_pm_subject = {
			ftr_personal_merit_consume_effect = { AMOUNT = ftr_personal_merit_council_cost_value }
			add_character_flag = {
				flag = ftr_personal_merit_council_tenure
				years = ftr_personal_merit_council_tenure_years_value
			}
		}
	}

	option = { # Grant them a non-capital county
		name = ftr_personal_merit.0001.b
		trigger = { exists = scope:ftr_pm_subject }
		ai_chance = {
			base = 20
			modifier = {
				add = 20
				scope:ftr_pm_subject = { has_trait = ambitious }
			}
		}
		# 随机挑一枚本人直辖的非首都伯爵领，转移所有权给该廷臣并扣 county 档功勋
		ftr_personal_merit_grant_random_county_effect = { TARGET = scope:ftr_pm_subject }
	}

	option = { # Leave it for another day
		name = ftr_personal_merit.0001.d
		trigger = { exists = scope:ftr_pm_subject }
		ai_chance = {
			base = 30
			modifier = {
				add = -20
				gold < 300
			}
		}
		# 从简不发：一年内不再触发
		add_character_flag = {
			flag = ftr_personal_merit_annual_skip
			years = 1
		}
	}
}
