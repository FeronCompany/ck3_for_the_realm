######### 0001 潜伏：暗流涌动 #########
ftr_court_struggle.0001 = {
	type = character_event
	title = ftr_court_struggle.0001.t
	desc = ftr_court_struggle.0001.desc
	theme = intrigue

	left_portrait = {
		character = root
	}

	immediate = {
		save_scope_as = ftr_court_monarch
		any_owned_story = {
			type = ftr_court_struggle
			save_scope_as = ftr_court_story
		}
		scope:ftr_court_story ?= {
			random_in_list = {
				list = ftr_court_claimant_list
				save_scope_as = ftr_court_candidate
			}
		}
	}

	option = { # Ignore it
		name = ftr_court_struggle.0001.a
		ai_chance = { base = 50 }
	}
	option = { # Admonish them
		name = ftr_court_struggle.0001.b
		ai_chance = { base = 30 }
		scope:ftr_court_candidate ?= {
			add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_political_opponent_opinion }
		}
	}
	option = { # Show favour instead
		name = ftr_court_struggle.0001.c
		ai_chance = { base = 20 }
		scope:ftr_court_candidate ?= {
			add_opinion = { target = scope:ftr_court_monarch  modifier = ftr_political_alliance_opinion }
		}
		# 恩宠双轨：双方均行政制 -> 影响力；否则 -> 威望
		ftr_court_grant_favor_effect = { TARGET = scope:ftr_court_candidate }
	}
}

######### 0011 其余竞争者的命运（root = 新君） #########
ftr_court_struggle.0011 = {
	type = character_event
	title = ftr_court_struggle.0011.t
	desc = ftr_court_struggle.0011.desc
	theme = intrigue

	left_portrait = {
		character = root
	}

	immediate = {
		save_scope_as = ftr_court_new_monarch
		any_owned_story = {
			type = ftr_court_struggle
			save_scope_as = ftr_court_story
		}
		# 找出仍有实力反抗的竞争者（有地且有兵），供后续起兵事件使用
		scope:ftr_court_story ?= {
			every_in_list = {
				list = ftr_court_claimant_list
				limit = {
					NOT = { this = scope:ftr_court_new_monarch }
					is_landed = yes
					max_military_strength > 0
					NOT = { is_imprisoned = yes }
				}
				save_scope_as = ftr_court_dissenter
			}
		}
	}

	option = { # Pardon them and honour my promises
		name = ftr_court_struggle.0011.a
		ai_chance = {
			base = 40
			modifier = {
				add = 40
				has_trait = just
			}
		}
		ftr_court_fulfill_promises_effect = yes
		add_prestige = ftr_court_prestige_minor_value
	}
	option = { # Silence them, one and all
		name = ftr_court_struggle.0011.b
		ai_chance = {
			base = 35
			modifier = {
				add = 40
				has_trait = callous
			}
		}
		# 标记被清算者
		scope:ftr_court_story ?= {
			every_in_list = {
				list = ftr_court_claimant_list
				limit = { NOT = { this = scope:ftr_court_new_monarch } }
				add_character_flag = ftr_court_purged
				ftr_court_clear_claimant_status_effect = yes
				add_opinion = { target = scope:ftr_court_new_monarch  modifier = ftr_court_rival_claimant_opinion }
			}
		}
		# 清算即免债：被清算者生前许下的承诺一并作废，新君只继承存活者的承诺
		scope:ftr_court_new_monarch = {
			every_vassal = {
				limit = {
					var:ftr_court_promised_by ?= { has_character_flag = ftr_court_purged }
				}
				remove_variable = ftr_court_promise_council
				remove_variable = ftr_court_promise_contract
				remove_variable = ftr_court_promised_by
			}
			every_courtier = {
				limit = {
					var:ftr_court_promised_by ?= { has_character_flag = ftr_court_purged }
				}
				remove_variable = ftr_court_promise_council
				remove_variable = ftr_court_promise_contract
				remove_variable = ftr_court_promised_by
			}
		}
		# 清除被清算者的欠账计数与临时标记
		scope:ftr_court_story ?= {
			every_in_list = {
				list = ftr_court_claimant_list
				limit = { has_character_flag = ftr_court_purged }
				remove_variable = ftr_court_owes_council
				remove_character_flag = ftr_court_purged
			}
		}
		add_dread = ftr_court_dread_medium_value
	}
	option = { # Let them keep their titles, but watch them
		name = ftr_court_struggle.0011.c
		trigger = {
			exists = scope:ftr_court_dissenter
		}
		ai_chance = { base = 25 }
		scope:ftr_court_dissenter = {
			trigger_event = {
				id = ftr_court_struggle.0012
				days = 5
			}
		}
	}
}

######### 0016 首次提醒（root = 新君） #########
# 可立刻任命（按才干分派官方五席），也可推迟一次（180-360 天后终裁）。
ftr_court_struggle.0016 = {
	type = character_event
	title = ftr_court_struggle.0016.t
	desc = ftr_court_struggle.0016.desc
	theme = stewardship

	left_portrait = {
		character = root
	}
	right_portrait = {
		character = scope:ftr_court_promised_vassal
	}

	# 无欠账则事件静默取消
	trigger = {
		has_variable = ftr_court_owes_council
		var:ftr_court_owes_council > 0
		OR = {
			any_vassal = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
			any_courtier = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
		}
	}

	immediate = {
		# 取一名被承诺者（先封臣后廷臣）
		random_vassal = {
			limit = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
			save_scope_as = ftr_court_promised_vassal
		}
		if = {
			limit = { NOT = { exists = scope:ftr_court_promised_vassal } }
			random_courtier = {
				limit = {
					has_variable = ftr_court_promise_council
					var:ftr_court_promised_by ?= { this = prev }
				}
				save_scope_as = ftr_court_promised_vassal
			}
		}
	}

	option = { # Grant them the promised seat
		name = ftr_court_struggle.0016.a
		ai_chance = {
			base = 60
			modifier = { add = 30  has_trait = just }
			modifier = { add = -40  has_trait = deceitful }
		}
		trigger = { exists = scope:ftr_court_promised_vassal }
		ftr_court_appoint_promised_councillor_effect = { TARGET = scope:ftr_court_promised_vassal }
		# 还有别的欠账则串行下一个提醒
		ftr_court_queue_next_council_reminder_effect = yes
	}
	option = { # Postpone it — they will not wait forever
		name = ftr_court_struggle.0016.b
		ai_chance = {
			base = 40
			modifier = { add = 40  has_trait = deceitful }
			modifier = { add = 20  has_trait = impatient }
			modifier = { add = -30  has_trait = just }
		}
		trigger = { exists = scope:ftr_court_promised_vassal }
		# 唯一一次推迟：终裁事件只给任命或拒绝
		trigger_event = {
			id = ftr_court_struggle.0017
			days = { ftr_court_council_final_delay_min_value  ftr_court_council_final_delay_max_value }
		}
	}
}

######### 0014 朝堂站队（root = 被询问的朝堂成员，可为无地廷臣） #########
# 由 story 的对立/危机脉冲推送。
# 存在意义：无地廷臣不能主动发起角色交互，其参与一律依赖事件触发。
ftr_court_struggle.0014 = {
	type = character_event
	title = ftr_court_struggle.0014.t
	desc = ftr_court_struggle.0014.desc
	theme = intrigue

	left_portrait = {
		character = root
	}

	immediate = {
		save_scope_as = ftr_court_asked
		# 定位君主（封臣走 top_liege，廷臣走 employer）与其名下的角逐
		if = {
			limit = { exists = top_liege }
			top_liege = {
				save_scope_as = ftr_court_monarch
				any_owned_story = {
					type = ftr_court_struggle
					save_scope_as = ftr_court_story
				}
			}
		}
		else_if = {
			limit = { exists = employer }
			employer = {
				save_scope_as = ftr_court_monarch
				any_owned_story = {
					type = ftr_court_struggle
					save_scope_as = ftr_court_story
				}
			}
		}
		scope:ftr_court_story ?= {
			ftr_court_save_ranked_claimants_effect = yes
		}
	}

	option = { # Back the strongest
		name = ftr_court_struggle.0014.a
		trigger = { exists = scope:ftr_court_claimant_1 }
		ai_chance = {
			base = 40
			modifier = { add = 30  has_trait = ambitious }
		}
