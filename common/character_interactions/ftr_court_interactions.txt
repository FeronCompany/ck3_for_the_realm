### 议题 1：继任之争
############################################################################

######### 撤回支持，重归中立 #########
ftr_court_withdraw_backing_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_withdraw_backing_interaction_desc
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = close_family
		max = 1
	}
	ai_frequency = 24
	is_shown = {
		scope:actor = {
			# 无地廷臣无法主动发起交互（站队走事件 0014，撤回暂无事件路径）
			is_landed = yes
			has_variable = ftr_court_backer_of
			var:ftr_court_backer_of ?= { this = scope:recipient }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			remove_variable = ftr_court_backer_of
			# 主动撤回支持 = 主动结清拥立账（功勋好感随之清零）
			ftr_personal_merit_backing_clear_effect = yes
			add_opinion = { target = scope:recipient  modifier = ftr_political_opponent_opinion }
		}
		# 从被支持者的个人支持者列表移除自己（新方案：支持者列表存于被支持者身上）
		scope:recipient = {
			ftr_court_remove_backer_effect = { BACKER = scope:actor }
			add_opinion = { target = scope:actor  modifier = ftr_political_opponent_opinion }
		}
	}
	on_decline = {}
	ai_will_do = {
		base = -30
		modifier = {
			add = 40
			opinion = { target = scope:recipient  value < 0 }
		}
		# 多疑者风声不对先抽身自保，不愿与人捆绑过深
		modifier = {
			add = 15
			scope:actor = { has_trait = paranoid }
		}
	}

	ai_accept = {
		# 强留一个离心的人弊大于利：总体放行，仅在惜才/急需实力时挽留
		base = 50
		# 被支持者越器重此人越惜才，越不愿放人（好感高 → 接受度降）
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = -0.5
		}
		# 竞逐正酣、局面胶着，缺不得爪牙
		modifier = {
			desc = ftr_court_accept_withdraw_struggle
			add = -25
			scope:recipient = {
				any_owned_story = {
					story_type = ftr_court_struggle
					has_variable = ftr_court_struggle_intensity
					var:ftr_court_struggle_intensity >= ftr_court_intensity_stir_bound_value
				}
			}
		}
		# 野心勃勃者绝不轻易让麾下力量流失
		modifier = {
			desc = ftr_court_accept_withdraw_ambitious
			add = -20
			scope:recipient = { has_trait = ambitious }
		}
		# 刚愎自用者拉不下脸，宁可挽留也不愿见人转投
		modifier = {
			desc = ftr_court_accept_withdraw_arrogant
			add = -15
			scope:recipient = { has_trait = arrogant }
		}
		# 多疑者怕强留之人怀恨反噬，宁愿好聚好散
		modifier = {
			desc = ftr_court_accept_withdraw_paranoid
			add = 15
			scope:recipient = { has_trait = paranoid }
		}
		# 随遇而安者不强留人
		modifier = {
			desc = ftr_court_accept_withdraw_content
			add = 10
			scope:recipient = { has_trait = content }
		}
	}
}

######### 请愿共治：权臣试探大位 #########
ftr_court_petition_coregency_interaction = {
	category = interaction_category_vassal
	common_interaction = yes
	desc = ftr_court_petition_coregency_interaction_desc
	cooldown_against_recipient = { years = 3 }
	ai_targets = {
		ai_recipients = liege
	}
	ai_frequency = 6
	is_shown = {
		scope:recipient = {
			ftr_court_is_struggle_monarch_trigger = yes
			ftr_court_monarch_infirm_trigger = yes
			NOT = { has_active_diarchy = yes }
		}
		scope:actor = {
			OR = {
				is_councillor_of = scope:recipient
				is_powerful_vassal_of = scope:recipient
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
			ftr_court_can_pay_cost_trigger = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_request_coregency_prestige_cost_value
				INFLUENCE = ftr_court_request_coregency_influence_cost_value
			}
		}
		# 需已积累足够的朝堂倾向
		scope:recipient = {
			any_owned_story = {
				type = ftr_court_struggle
				ftr_court_can_petition_coregency_trigger = yes
			}
		}
	}
	can_send = {}
	on_accept = {
		scope:actor = {
			ftr_court_pay_cost_effect = {
				OTHER = scope:recipient
				PRESTIGE = ftr_court_request_coregency_prestige_cost_value
				INFLUENCE = ftr_court_request_coregency_influence_cost_value
			}
		}
		# 请愿结果交由事件裁决，君主可允可拒
		scope:recipient = {
			save_scope_as = ftr_court_monarch
			# 朝堂通报：共治请愿已呈递
			ftr_court_send_court_toast_effect = {
				TITLE = ftr_court_notice_petition
				ICON = scope:actor
			}
			trigger_event = {
				id = ftr_court_struggle.0020
				days = 1
			}
		}
	}
	on_decline = {}
	ai_will_do = {
		base = -60
		modifier = {
			add = 50
			scope:actor = { has_trait = ambitious }
		}
		modifier = {
			add = 25
			scope:actor = { is_powerful_vassal_of = scope:recipient }
		}
		# 公然请立共治近乎逼宫，多疑者绝不轻蹈险地
		modifier = {
			add = -20
			scope:actor = { has_trait = paranoid }
		}
		# 「太子优先」生效时（君主已有可用指定继承人且非本人）→ 请愿多半替他人作嫁衣，意愿大减
		modifier = {
			add = ftr_court_petition_heir_priority_disincentive_value
			scope:recipient = {
				ftr_diarchy_can_start_crown_prince_trigger = yes
				exists = designated_heir
				designated_heir ?= {
					is_alive = yes
					is_adult = yes
					is_imprisoned = no
					NOT = { this = scope:actor }
				}
			}
		}
	}

	ai_accept = {
		# 分权大位非同小可：仅对信得过、且实在需要其分忧的人松口
		base = -40
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
		}
		# 素来忠贞的臣下，分权亦不虞反噬
		modifier = {
			desc = ftr_court_accept_petition_loyal
			add = 35
			scope:actor = { has_trait = loyal }
		}
		# 能力卓著者能真正分忧（声望高者治政名声在外）
		modifier = {
			desc = ftr_court_accept_petition_prestige
			add = 15
			scope:actor = { prestige_level >= 3 }
		}
		# 病弱君主渴望强援分担朝政（本交互已要求君主失能/衰弱）
		modifier = {
			desc = ftr_court_accept_petition_need
			add = 30
			scope:recipient = { has_trait = craven }
		}
		# 慷慨、轻信者更愿托付
		modifier = {
			desc = ftr_court_accept_petition_generous
			add = 20
			scope:recipient = { has_trait = generous }
		}
		modifier = {
			desc = ftr_court_accept_petition_trusting
			add = 15
			scope:recipient = { has_trait = trusting }
		}
		# 权欲旺盛的君主绝不分权
		modifier = {
			desc = ftr_court_accept_petition_monarch_ambitious
			add = -40
			scope:recipient = { has_trait = ambitious }
		}
		# 刚愎自用者自视独力可支
		modifier = {
			desc = ftr_court_accept_petition_arrogant
			add = -25
			scope:recipient = { has_trait = arrogant }
		}
		# 守成者不愿轻改朝局
		modifier = {
			desc = ftr_court_accept_petition_content
			add = -20
			scope:recipient = { has_trait = content }
		}
		# 多疑的君主恐请立实为染指大宝
		modifier = {
			desc = ftr_court_accept_petition_paranoid
			add = -15
			scope:recipient = { has_trait = paranoid }
		}
		# 来者是头号强藩时，名为辅政实为逼宫
		modifier = {
			desc = ftr_court_accept_petition_powerful
			add = -20
			scope:actor = { is_powerful_vassal_of = scope:recipient }
		}
		# 来者野心昭彰者不可托付
		modifier = {
			desc = ftr_court_accept_petition_actor_ambitious
			add = -30
			scope:actor = { has_trait = ambitious }
		}
		# 巧言令色者恐揽权谋私
		modifier = {
			desc = ftr_court_accept_petition_deceitful
			add = -20
			scope:actor = { has_trait = deceitful }
		}
	}
}

##############################################################
### 朝堂权力角逐 — 劝人背书（02 §3.2 / §3.3）
###
### actor = 已站队的支持者；向另一名未站队的朝堂成员推销自己的主君（权贵 C）。
### 被劝者同意背书 C → 按"被劝背书 +5"计功（apply_backing 默认 +10 后再回调 5）；
### 劝说者本人同记 +5 拥立功（并入其既有拥立账）。
##############################################################
ftr_court_persuade_backing_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = ftr_court_persuade_backing_interaction_desc
	cooldown_against_recipient = { years = 2 }
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 3
	}
	ai_frequency = 12
	is_shown = {
		scope:actor = {
			is_capable_adult = yes
			is_landed = yes
			NOT = { is_imprisoned = yes }
			has_variable = ftr_court_backer_of
			# 同一朝堂角逐进行中
			any_owned_story = {
				story_type = ftr_court_struggle
			}
		}
		scope:recipient = {
			is_alive = yes
			is_capable_adult = yes
			NOT = { this = scope:actor }
			NOT = { has_variable = ftr_court_backer_of }
			OR = {
				is_vassal_of = scope:actor.top_liege
				NOT = { is_landed = yes }
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_capable_adult = yes
			NOT = { is_imprisoned = yes }
		}
		scope:recipient = { NOT = { is_imprisoned = yes } }
	}
	can_send = {}

	on_accept = {
		# 劝说者所支持的权贵 C
		scope:actor = {
			var:ftr_court_backer_of ?= {
				save_scope_as = ftr_pm_persuaded_claimant
			}
		}
		if = {
			limit = { exists = scope:ftr_pm_persuaded_claimant }
			# 被劝者背书 C（apply_backing 全套记账），功勋回调为"被劝 +5"
			scope:recipient = {
				ftr_court_apply_backing_effect = { CLAIMANT = scope:ftr_pm_persuaded_claimant }
				ftr_personal_merit_backing_deduct_effect = { AMOUNT = ftr_personal_merit_backing_active_premium_value }
			}
			# 劝说者：对同一权贵 +5 拥立功（并入既有账）
			scope:actor = {
				ftr_personal_merit_backing_add_effect = {
					CLAIMANT = scope:ftr_pm_persuaded_claimant
					AMOUNT = ftr_personal_merit_persuading_value
				}
			}
		}
	}

	on_decline = {}

	ai_will_do = {
		base = 10
		modifier = {
			add = 20
			scope:actor = { has_trait = ambitious }
		}
		modifier = {
			add = -20
			scope:actor = { has_trait = content }
		}
	}
	ai_accept = {
		base = 0
		modifier = {
			add = 30
			opinion = { target = scope:actor  value >= 25 }
		}
		modifier = {
			add = 20
			scope:recipient = { has_trait = ambitious }
		}
		modifier = {
			add = -25
			scope:recipient = { has_trait = craven }
		}
	}
}
