		name = ftr_court_struggle.0011.a

			}
		}
		ftr_court_fulfill_promises_effect = yes
		add_prestige = ftr_court_prestige_minor_value
	}
	option = { # Silence them, one and all
		name = ftr_court_struggle.0011.b
		ai_chance = {
			base = 35
			modifier = {
				add = 40
				has_trait = callous
			}
		}
		# 标记被清算者
		scope:ftr_court_story ?= {
			every_in_list = {
				list = ftr_court_participant_list
				limit = { NOT = { this = scope:ftr_court_new_monarch } }
				add_character_flag = ftr_court_purged
				ftr_court_clear_claimant_status_effect = yes
				add_opinion = { target = scope:ftr_court_new_monarch  modifier = ftr_court_rival_claimant_opinion }
			}
		}
		# 清算即免债：被清算者生前许下的承诺一并作废，新君只继承存活者的承诺
		scope:ftr_court_new_monarch = {
			every_vassal = {
				limit = {
					var:ftr_court_promised_by ?= { has_character_flag = ftr_court_purged }
				}
				remove_variable = ftr_court_promise_council
				remove_variable = ftr_court_promise_contract
				remove_variable = ftr_court_promised_by
			}
			every_courtier = {
				limit = {
					var:ftr_court_promised_by ?= { has_character_flag = ftr_court_purged }
				}
				remove_variable = ftr_court_promise_council
				remove_variable = ftr_court_promise_contract
				remove_variable = ftr_court_promised_by
			}
		}
		# 清除被清算者的欠账计数与临时标记
		scope:ftr_court_story ?= {
			every_in_list = {
				list = ftr_court_participant_list
				limit = { has_character_flag = ftr_court_purged }
				remove_variable = ftr_court_owes_council
				remove_character_flag = ftr_court_purged
			}
		}
		add_dread = ftr_court_dread_medium_value
	}
	option = { # Let them keep their titles, but watch them
		name = ftr_court_struggle.0011.c
		trigger = {
			exists = scope:ftr_court_dissenter
		}
		ai_chance = { base = 25 }
		scope:ftr_court_dissenter = {
			trigger_event = {
				id = ftr_court_struggle.0012
				days = 5
			}
		}
	}
}

######### 0016 首次提醒（root = 新君） #########
# 可立刻任命（按才干分派官方五席），也可推迟一次（180-360 天后终裁）。
ftr_court_struggle.0016 = {
	type = character_event
	title = ftr_court_struggle.0016.t
	desc = ftr_court_struggle.0016.desc
	theme = stewardship

	left_portrait = {
		character = root
	}
	right_portrait = {
		character = scope:ftr_court_promised_vassal
	}

	# 无欠账则事件静默取消
	trigger = {
		has_variable = ftr_court_owes_council
		var:ftr_court_owes_council > 0
		OR = {
			any_vassal = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
			any_courtier = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
		}
	}

	immediate = {
		# 取一名被承诺者（先封臣后廷臣）
		random_vassal = {
			limit = {
				has_variable = ftr_court_promise_council
				var:ftr_court_promised_by ?= { this = prev }
			}
			save_scope_as = ftr_court_promised_vassal
		}
		if = {
			limit = { NOT = { exists = scope:ftr_court_promised_vassal } }
			random_courtier = {
				limit = {
					has_variable = ftr_court_promise_council
					var:ftr_court_promised_by ?= { this = prev }
				}
				save_scope_as = ftr_court_promised_vassal
			}
		}
	}

	option = { # Grant them the promised seat
		name = ftr_court_struggle.0016.a
		ai_chance = {
			base = 60
			modifier = { add = 30  has_trait = just }
			modifier = { add = -40  has_trait = deceitful }
		}
		trigger = { exists = scope:ftr_court_promised_vassal }
		ftr_court_appoint_promised_councillor_effect = { TARGET = scope:ftr_court_promised_vassal }
		# 还有别的欠账则串行下一个提醒
		ftr_court_queue_next_council_reminder_effect = yes
	}
	option = { # Postpone it — they will not wait forever
		name = ftr_court_struggle.0016.b
		ai_chance = {
			base = 40
			modifier = { add = 40  has_trait = deceitful }
			modifier = { add = 20  has_trait = impatient }
			modifier = { add = -30  has_trait = just }
		}
		trigger = { exists = scope:ftr_court_promised_vassal }
		# 唯一一次推迟：终裁事件只给任命或拒绝
		trigger_event = {
			id = ftr_court_struggle.0017
			days = { ftr_court_council_final_delay_min_value  ftr_court_council_final_delay_max_value }
		}
	}
}

######### 0040 是否参与继位角逐（root = 潜在继任者） #########
# 由 story 的启动 / 定期刷新（ftr_court_launch_participation_asks_effect）推给潜在继任者。
# 血缘继承：询问君主子女 / 兄弟姐妹；选举继承：按主头衔继承顺位询问。
# 选项：参与（加入角逐，可被他人支持） / 不参与（终身 flag，防止被他人主动支持）。
# 参与人数已达上限（5）时不询问；已明确表示不参与者不询问。
ftr_court_struggle.0040 = {
	type = character_event
	title = ftr_court_struggle.0040.t
	desc = ftr_court_struggle.0040.desc
	theme = intrigue

	left_portrait = {
		character = root
	}
	right_portrait = {
		character = scope:ftr_court_monarch
	}

	immediate = {
		save_scope_as = ftr_court_asked_claimant
		# 优先使用触发方（ftr_court_launch_participation_asks_effect）通过 save_scope_as 传递的 story/monarch，
		# 避免因候选人关系变化（top_liege/employer 非朝堂君主）导致重新定位失败。
		# 兜底：若未传递或失效，再按关系定位。
		if = {
			limit = { NOT = { exists = scope:ftr_court_story } }
			if = {
				limit = { is_independent_ruler = yes }
				save_scope_as = ftr_court_monarch
			}
			else_if = {
				limit = { exists = top_liege }
				top_liege = {
					save_scope_as = ftr_court_monarch
					random_owned_story = {
						type = ftr_court_struggle
						save_scope_as = ftr_court_story
					}
				}
			}
			else_if = {
				limit = { exists = employer }
				employer = {
					save_scope_as = ftr_court_monarch
					random_owned_story = {
						type = ftr_court_struggle
						save_scope_as = ftr_court_story
					}
				}
			}
			else = {
				save_scope_as = ftr_court_monarch
				random_owned_story = {
					type = ftr_court_struggle
					save_scope_as = ftr_court_story
				}
			}
		}
		# 若君主未传递，尝试从 story 反推
		if = {
			limit = {
				exists = scope:ftr_court_story
				NOT = { exists = scope:ftr_court_monarch }
			}
			scope:ftr_court_story = {
				story_owner = { save_scope_as = ftr_court_monarch }
			}
		}
	}

	option = { # I will enter the contest for the throne
		name = ftr_court_struggle.0040.a
		trigger = {
			exists = scope:ftr_court_story
			scope:ftr_court_story = {
				has_variable = ftr_court_participant_count
				var:ftr_court_participant_count < ftr_court_max_candidates_value
			}
			NOT = { has_character_flag = ftr_court_declined_contest }
		}
		ai_chance = {
			base = 35
			modifier = { add = 40  has_trait = ambitious }
			modifier = { add = 25  has_trait = brave }
			modifier = { add = 25  has_trait = greedy }
			modifier = { add = -50  has_trait = content }
			modifier = { add = -30  has_trait = craven }
		}
		# 参与角逐：加 flag + 加入最高统治者个人维护的参与者列表 ftr_court_claimants。
		# 新方案：参与者列表主存储放在最高统治者（scope:ftr_court_monarch）的个人变量列表上，

		save_scope_as = ftr_court_entering_claimant
		add_character_flag = ftr_court_participating
		remove_character_flag = ftr_court_asked_participation
		# 计算竞争力评分（在候选人身上）
		set_variable = {
			name = ftr_claim_strength
			value = ftr_court_claim_strength_value
		}
		# 把参与者加入最高统治者的 ftr_court_claimants 列表
		scope:ftr_court_monarch = {
			add_to_variable_list = {
				name = ftr_court_claimants
				target = scope:ftr_court_entering_claimant
			}
		}
		# 同步到 story 的 ftr_court_participant_list（供 story 逻辑与可视化使用）
		scope:ftr_court_story ?= {
			add_to_variable_list = {
				name = ftr_court_participant_list
				target = scope:ftr_court_entering_claimant
			}
			set_variable = {
				name = ftr_court_participant_count
				value = list_size:ftr_court_participant_list
			}
			set_variable = { name = ftr_court_candidate_count  value = var:ftr_court_participant_count }
			ftr_court_update_phase_effect = yes
		}
		# 通知朝廷
		scope:ftr_court_monarch = {
			ftr_court_send_court_toast_effect = {
				TITLE = ftr_court_notice_join_contest
				ICON = scope:ftr_court_asked_claimant

		}
