
on_title_gain = {
	on_actions = {
		ftr_ursurp_discontent_effects
		ftr_ursurp_tyrant_succession_effects
		ftr_remove_empire_for_hegemony_effects
		ftr_auto_assign_baronies_effects
		# 政权易主（退位/被废/篡夺/派系逼宫）时，朝堂权力角逐 story 跟随新君转移
		ftr_court_handover_on_title_gain_effect
	}
}

# 非死亡权力交接的朝堂故事转移守卫。
# 作用域（官方 on_title_gain）：root = 新 holder，scope:title = 易主的头衔，
# scope:previous_holder = 前 holder（可能已死）。
# 死亡继承场景前 holder 已死 → is_alive 过滤掉，交给 story 的 on_owner_death 处理；
# 此处只接管「前 holder 存活却失去政权」的退位/被废/篡夺/逼宫。
ftr_court_handover_on_title_gain_effect = {
	trigger = {
		# 仅政权级头衔（王国及以上）易主
		scope:title ?= {
			tier >= tier_kingdom
		}
		# 只处理权力交接类（排除 inheritance/granted/created 等日常流转）
		OR = {
			scope:transfer_type = flag:abdication
			scope:transfer_type = flag:stepped_down
			scope:transfer_type = flag:faction_demand
			scope:transfer_type = flag:usurped
		}
		# 前 holder 存活且其名下确有朝堂故事
		scope:previous_holder ?= {
			is_alive = yes
			any_owned_story = { story_type = ftr_court_struggle }
		}
		# 新 holder 目前没有自己的朝堂故事（防重复转移）
		NOT = { any_owned_story = { story_type = ftr_court_struggle } }
	}
	effect = {
		root = { save_scope_as = ftr_court_new_monarch }
		scope:previous_holder = {
			every_owned_story = {
				limit = { story_type = ftr_court_struggle }
				ftr_court_transfer_story_effect = yes
			}
		}
	}
}

ftr_ursurp_discontent_effects = {
	trigger = {
		is_independent_ruler = yes
		scope:title = {
			tier >= tier_kingdom
			tier = prev.primary_title.tier
		}
		OR = {
			scope:transfer_type = flag:conquest_claim
			scope:transfer_type = flag:usurped
			scope:transfer_type = flag:faction_demand
		}
	}
	effect = {
		save_scope_as = usurper_liege
		every_vassal = {
			limit = {
				primary_title = {
					tier >= tier_duchy
					trigger_if = {
						limit = {
							NOT = { scope:title = scope:usurper_liege.primary_title }
						}
						de_jure_liege = scope:title
					}
				}
				opinion = {
					value < 0
					target = scope:usurper_liege
				}
				NOT = {
					is_allied_to = scope:usurper_liege
				}
				NOT = {
					has_character_flag = ftr_block_usurper_notify
				}
				is_at_war = no
			}
			add_character_flag = {
				flag = ftr_block_usurper_notify
				days = 30
			}
			trigger_event = {
				id = ftr_realm_event.1010
				days = 5
			}
		}
	}
}

ftr_ursurp_tyrant_succession_effects = {
	trigger = {
		is_independent_ruler = yes
		government_allows = administrative
		can_have_acclamation_succession_law_trigger = yes
		OR = {
			has_realm_law = single_heir_succession_law
			has_realm_law = single_heir_succession_law_youngest
		}
		scope:transfer_type = flag:faction_demand
	}
	effect = {
		save_scope_as = usurper_liege
		#remove_realm_law = autocracy_succession_law
		add_realm_law = acclamation_succession_law
		every_vassal = {
			limit = { is_ai = no }
			send_interface_message = {
				type = event_generic_neutral
				title = liege_adapt_acclamation_succession_law_title
				desc = liege_adapt_acclamation_succession_law_desc
				right_icon = scope:usurper_liege
			}
		}
	}
}

ftr_remove_empire_for_hegemony_effects = {
	trigger = {
		scope:transfer_type = flag:created
		scope:title.tier = tier_hegemony
		is_ai = yes
	}
	effect = {
		every_held_title = {
			limit = {
				tier = tier_empire
			}
			holder = { destroy_title = prev }
		}
	}
}

ftr_auto_assign_baronies_effects = {
	trigger = {
		scope:title.tier = tier_barony
		is_ai = no
		is_independent_ruler = yes
		government_allows = administrative
		has_character_flag = ftr_auto_assign_baronies_flag
	}
	effect = {
		create_character = {
			gender = male
			location = root.location
			culture = root.culture
			faith = root.faith
			save_scope_as = barony_manager
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:title = {
			change_title_holder_include_vassals = {
				holder = scope:barony_manager
				change = scope:change
				government_base = root
			}
		}
		scope:barony_manager = {
			change_liege = {
				liege = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}
