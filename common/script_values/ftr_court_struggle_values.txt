# 求值时的作用域：this = 候选人，scope:ftr_court_monarch = 在位君主
ftr_court_claim_strength_value = {
	value = 0

	### 成年与年龄阶段
	if = {
		limit = { is_adult = yes }

	if = { limit = { has_trait = reclusive }		add = ftr_court_claim_reclusive_value }
	if = { limit = { has_trait = shy }				add = ftr_court_claim_shy_value }

	### 实力与身份
	if = { limit = { is_landed = yes }									add = ftr_court_claim_landed_value }
	if = { limit = { is_councillor_of = scope:ftr_court_monarch }		add = ftr_court_claim_councillor_value }

		limit = { opinion = { target = scope:ftr_court_monarch  value >= 0 } }

		add = ftr_court_claim_opinion_low_value

	}

	### 永不出现负分，便于展示
	max = 0
}

########## 四·B、列表人数（查询用） ##########
# 重要：以下 court 名单均为 add_to_variable_list 维护的 variable list。
# 取大小有两种合法姿势：
#   1) trigger/limit 里判断：variable_list_size = { name = 名单 value … }（原版 tests 权威）；
#   2) 需要数值（set_variable / multiply / tooltip 文本）：不能裸用 list_size:（那是 add_to_list 临时 list 的语法），
#      必须 has_variable_list 保护后用 every_in_list variable= 遍历累加（对齐原版 script value 计数范式）。
# 下面三个 count value 即按姿势 2 实现，供各调用点引用。

# 求值域：this = 某权贵（君主 / 共治者 / 参与角逐者）。数其 ftr_court_supporters 支持者。
# 调试决议 ftr_court_debug_report_claimants_decision 在 custom_tooltip 文本中引用本值，逐行显示各竞争者支持者数。
ftr_court_supporters_count_value = {
	value = 0
	if = {
		limit = { has_variable_list = ftr_court_supporters }
		every_in_list = {
			variable = ftr_court_supporters
			add = 1
		}
	}
}

# 求值域：this = 拟背书者（actor），scope:recipient = 被支持的权贵。
# AI 主动背书交互 ftr_court_back_claimant_interaction 的 ai_will_do 引用本值：
# 逐名遍历目标既有支持者名单，每名扣减一步长（首名豁免），线性压制"锦上添花"式背书。
ftr_court_backing_crowd_penalty_value = {
	value = 0
	scope:recipient = {
		if = {
			limit = { has_variable_list = ftr_court_supporters }
			every_in_list = {
				variable = ftr_court_supporters
				subtract = ftr_court_backing_crowd_penalty_step_value
			}
			# 豁免第一名：对象已有 1 名支持者时仍可正常押注
			add = ftr_court_backing_crowd_penalty_step_value
		}
	}
}
