					scope:target.primary_defender = scope:recipient

					limit = { attacker_war_score > 0 }

						save_scope_as = winning_side
					}

						save_scope_as = losing_side
					}

			}

				exists = scope:target

			who = scope:actor

			trigger = {
				exists = scope:target
				scope:target.primary_attacker = scope:winning_side
			}
			opinion_target = scope:actor
			who = scope:target.primary_attacker
			multiplier = -0.5
		}
	}

						AND = {
							primary_attacker = scope:recipient

								is_liege_or_above_of = prev.primary_defender

						}

									}
								}

						}

									opinion = {
										target = primary_defender
										value < 0
									}
								}
							}
							add = defender_war_score

							#

					}

			is_vassal_of = scope:actor
			government_has_mechanic = administrative
			primary_title.tier >= tier_duchy
			is_landed = yes
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			is_at_war = no
			is_imprisoned = no
		}
		scope:actor = {
			is_imprisoned = no
		}
	}
	populate_actor_list = {
		scope:actor = {
			every_vassal = {
				limit = {
					primary_title.tier = scope:recipient.primary_title.tier
					government_has_mechanic = administrative
					is_landed = yes
				}
				add_to_list = characters
			}
		}
	}
	can_be_picked = {
		is_at_war = no
	}
	
	cost = {
		influence = {
			value = monumental_influence_value
			if = {
				limit = { scope:secondary_actor.primary_title.tier = tier_duchy }
				add = 250
			}
			else_if = {
				limit = { scope:secondary_actor.primary_title.tier = tier_kingdom }
				add = 750
			}
		}
	}
	on_accept = {
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:secondary_actor = {
			if = {
				limit = {
					any_held_title = {
						is_noble_family_title = yes
					}
				}
				every_held_title = {
					limit = {
						is_noble_family_title = no
					}
					change_title_holder_include_vassals = {
						holder = scope:recipient
						change = scope:change
					}
				}
			}
			else = {
				every_held_title = {
					limit = {
						is_noble_family_title = no
						tier = scope:secondary_actor.primary_title.tier
					}
					change_title_holder_include_vassals = {
						holder = scope:recipient
						change = scope:change
					}
				}
				change_liege = {
					liege = scope:recipient
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:secondary_actor = {
			add_opinion = {
				target = scope:actor
				modifier = ftr_depose_bureacrat_opinion
			}
			add_opinion = {
				modifier = angry_opinion
				target = scope:recipient
				opinion = -30
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = ftr_promote_me_opinion
			}
		}
	}
	on_auto_accept = {}
	on_decline = {}
	auto_accept = yes
	ai_will_do = {
		base = -40
		modifier = {
			add = 100
			scope:actor.vassal_limit_available < 5
		}
		modifier = {
			add = -20
			OR = {
				scope:recipient = { is_powerful_vassal = yes }
				scope:secondary_actor = { is_powerful_vassal = yes }
			}
		}
		opinion_modifier = {
			who = scope:actor
			opinion_target = scope:recipient
			multiplier = 0.05
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.15
		}
		opinion_modifier = {
			who = scope:actor
			opinion_target = scope:secondary_actor
			multiplier = -0.05
		}
		opinion_modifier = {
			who = scope:secondary_actor
			opinion_target = scope:actor
			multiplier = -0.15
		}
	}
}
