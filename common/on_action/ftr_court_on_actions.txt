##############################################################
### 朝堂权力角逐 COURT STRUGGLE — 调度层
###
### 设计原则：
###   1) 故事一旦建立，其年度维护完全交给 story 的 effect_group，
###      不在 on_action 里重复做，避免双重开销。
###   2) on_action 只负责三件事：准入创建、候选池增量刷新、收场。
###   3) 君主驾崩后新君是否重启角逐，交由下一年的年度检查处理——
###      刻意避开继承结算的时序风险。
##############################################################

########## 一、准入：为符合条件的新君建立朝堂角逐 ##########

ftr_court_try_start_on_action = {
	trigger = {
		# 廉价前置过滤（AND 短路求值，绝大多数角色在此即退出，避免进入昂贵判定）
		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
		# 已有角逐则跳过
		NOT = { any_owned_story = { type = ftr_court_struggle } }
		# 完整准入 + 议题1触发（内部已含 valid / 非冷却 / 非战争 / 候选数）
		ftr_court_can_start_struggle_trigger = yes
		ftr_court_should_activate_succession_trigger = yes
	}
	effect = {
		create_story = ftr_court_struggle
	}
}

########## 二、收场 ##########

# 收紧后的兜底收场：仅当故事持有者不再是最高领主（失去独立 / 等级跌至王国以下）才收场；
# 被囚禁、战争、暂时无继承顺位等临时失效一概保留（由 story 的 effect_group fallback 同判）。
# story owner 驾崩无继承的收场由 on_owner_death → end_story 独立处理，不在此重复。
ftr_court_validate_on_action = {
	trigger = {
		# 身上确实挂着朝堂故事，且持有者已非最高领主才需要兜底收场
		any_owned_story = { type = ftr_court_struggle }
		is_alive = yes
		NOT = { ftr_court_story_owner_is_supreme_liege_trigger = yes }
	}
	effect = {
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			end_story = yes
		}
	}
}

######################################################
### 挂载到原版 on_action
######################################################

# 每年一次的全局脉冲：负责准入创建与失效收场
random_yearly_everyone_pulse = {
	on_actions = {
		ftr_court_try_start_on_action
		ftr_court_validate_on_action
	}
}


# 角色死亡时的朝堂名单清理（root = 死者）。
# 任何角色死亡都可能涉入朝堂（作为支持者或参与者），用变量前置过滤，
# 仅当死者身上确有朝堂关系（支持者登记 / 所属政权 backlink）才进入 effect。
ftr_court_death_cleanup_on_action = {
	trigger = {
		OR = {
			has_variable = ftr_court_backer_of
			has_variable = ftr_court_participant_of
		}
	}
	effect = {
		ftr_court_death_cleanup_effect = yes
	}
}

on_death = {
	on_actions = {
		ftr_court_death_cleanup_on_action
	}
}

######################################################
### 三、内斗主时钟选族（由 story 心跳每 60–120 天调用）
### story effect_group → story_owner = { trigger_event = { on_action = X } }
### → on_action 以 effect + random_list 按权重随机选一族，直接调各族既有 effect。
### 每拍只随机触发一族入口（党羽逼宫并入强出口族）；各族内部再按既有 effect 选角/派发。
######################################################

ftr_court_struggle_main_pulse = {
	effect = {
		# root = story_owner：经其 every_owned_story 定位故事（不依赖 scope:story）
		every_owned_story = {
			limit = { story_type = ftr_court_struggle }
			save_scope_as = ftr_court_pulse_story
		}
		if = {
			limit = { exists = scope:ftr_court_pulse_story }
			# 每拍按权重随机选一族，直接调各族既有 effect（原 0211–0215 hidden 调度事件已移除）
			random_list = {
				# 互殴手段族（0050–0052 / 0070 / 0080 / 0090）
				60 = {
					scope:ftr_court_pulse_story = { ftr_court_strife_slander_effect = yes }
				}
				# 强出口族（0003 / 0005 通知 + 0014 / 0015 / 0018）+ 党羽逼宫（0085 / 0086 并族）
				35 = {
					scope:ftr_court_pulse_story = { ftr_court_strife_exit_pulse_effect = yes }
					scope:ftr_court_pulse_story = { ftr_court_backer_demand_pulse_effect = yes }
				}
				# 无地拉锯（0083，需天平 ≥70 且有共治者）
				15 = {
					scope:ftr_court_pulse_story = { ftr_court_landless_swing_effect = yes }
				}
				# 共治反击（0084，需天平 ≤30 且有共治者）
				12 = {
					scope:ftr_court_pulse_story = { ftr_court_diarch_backlash_effect = yes }
				}
			}
		}
	}
}
