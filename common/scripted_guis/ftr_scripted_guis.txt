ftr_update_my_powerful_subject_list = {
	scope = character
	effect = {
		ftr_update_powerful_subjects_list_effect = yes
		clear_variable_list = ftr_my_powerful_subject_list
		every_in_list = {
			list = powerful_subjects_list
			save_scope_as = current_character
			root = {
				add_to_variable_list = {
					name = ftr_my_powerful_subject_list
					target = scope:current_character
				}
			}
		}
	}
}

# 朝堂局势 UI 数据同步：把玩家相关朝堂 story 的变量镜像到玩家域，
# 供 gui 读取（CK3 无法在 data function 中直接访问 story 变量）。
# 数据源优先级：自己的朝堂 → top_liege → employer（无地廷臣也能看自己主家的局势）。
# 在 HUD 点击与窗口打开/刷新时执行。
ftr_court_sync_ui_data = {
	scope = character
	is_shown = {
		always = yes
	}
	effect = {
		# 先复位，无朝堂时窗口显示"暂无角逐"
		set_variable = { name = ftr_court_ui_active  value = 0 }
		set_variable = { name = ftr_court_ui_swing  value = 0 }
		set_variable = { name = ftr_court_ui_phase  value = 0 }
		set_variable = { name = ftr_court_ui_count  value = 0 }
		set_variable = { name = ftr_court_ui_gap  value = 999 }
		clear_variable_list = ftr_court_ui_claimants

		# 1) 自己的朝堂
		if = {
			limit = {
				any_owned_story = { type = ftr_court_struggle }
			}
			ftr_court_sync_ui_from_story_effect = yes
		}
		# 2) 宗主的朝堂
		else_if = {
			limit = {
				exists = top_liege
				top_liege = {
					any_owned_story = { type = ftr_court_struggle }
				}
			}
			top_liege = { ftr_court_sync_ui_from_story_effect = yes }
		}
		# 3) 雇主的朝堂（无地廷臣）
		else_if = {
			limit = {
				exists = employer
				employer = {
					any_owned_story = { type = ftr_court_struggle }
				}
			}
			employer = { ftr_court_sync_ui_from_story_effect = yes }
		}
	}
}
