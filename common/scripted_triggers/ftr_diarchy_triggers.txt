### 一、政权准入

# 求值域：统治者本人。行政类政体 + 独立 + 主头衔 ≥ 王国
ftr_diarchy_realm_eligible_trigger = {
	government_allows = administrative
	is_independent_ruler = yes
	highest_held_title_tier >= tier_kingdom
}

# 求值域：统治者本人。太子共治准入：行政政体 + 独立 + ≥王国 + 当前无共治。
# 注：不再排除欢呼法——欢呼法政权若为罗马/文化解锁共帝则走 co_emperorship（分支①，更优），
#     其余欢呼法行政政权退而用太子共治，避免出现「无类型可用 → 只能固化摄政」的空档。
ftr_diarchy_can_start_crown_prince_trigger = {
	ftr_diarchy_realm_eligible_trigger = yes
	NOT = { has_active_diarchy = yes }
	is_alive = yes
	is_imprisoned = no
}

# 求值域：统治者本人。主君是否已有任何共治者（立储能入口锁定，§2.2 独占）
ftr_diarchy_liege_has_any_diarch_trigger = {
	has_active_diarchy = yes
}

### 二、身份分流（继承人线 / 官职线，§1.3）

# 求值域：共治者本人。是否属于「继承人线」——太子类型 / 指定继承人 / 候任共治者
ftr_diarchy_is_heir_line_diarch_trigger = {
	is_diarch = yes
	OR = {
		is_designated_diarch = yes
		liege = { has_diarchy_type = ftr_diarchy_crown_prince }
		liege = {
			designated_heir ?= { is_diarch_of_target = prev }
		}
	}
}

# 求值域：共治者本人。是否属于「官职线」——非继承人共治者（含原版 regency / co_monarchy / vizierate）
ftr_diarchy_is_officer_line_diarch_trigger = {
	is_diarch = yes
	NOT = { ftr_diarchy_is_heir_line_diarch_trigger = yes }
}

# 求值域：共治者本人。是否为「太子共治」类型
ftr_diarchy_is_crown_prince_trigger = {
	is_diarch = yes
	liege = { has_diarchy_type = ftr_diarchy_crown_prince }
}

# 求值域：君主本人。当前共治是否为「继承人 / 共治型」——太子 / 共治皇帝 / 共治君主 / 幼主共治。
# 用于 01 背书共治者的 AI 分流（01 §7.1）：继承人·共治型看与本人的关系，官职线看与君主的关系。
ftr_diarchy_is_heir_co_rule_type_trigger = {
	has_active_diarchy = yes
	OR = {
		has_diarchy_type = ftr_diarchy_crown_prince
		has_diarchy_type = co_emperorship
		has_diarchy_type = co_monarchy
		has_diarchy_type = junior_emperorship
	}
}

# 求值域：共治者本人。是否为官职线且已完成土地积累（§4.1 前置，A1c「先索官索地」）
ftr_diarchy_officer_line_landed_trigger = {
	ftr_diarchy_is_officer_line_diarch_trigger = yes
	is_landed = yes
}

### 三、天平门槛（求值域：共治者本人；swing 读自 liege）

ftr_diarchy_swing_at_least_depose_trigger = {
	liege = { diarchy_swing >= ftr_diarchy_depose_like_threshold_value }
}
# 独断档（≥90）：篡位落地
ftr_diarchy_swing_at_least_absolute_trigger = {
	liege = { diarchy_swing >= ftr_diarchy_coup_landing_swing_value }
}
ftr_diarchy_swing_at_least_designate_trigger = {
	liege = { diarchy_swing >= ftr_diarchy_designate_heir_swing_value }

ftr_diarchy_swing_at_least_boost_trigger = {
	liege = { diarchy_swing >= ftr_diarchy_boost_line_swing_value }
}
ftr_diarchy_swing_at_least_levy_trigger = {
	liege = { diarchy_swing >= ftr_diarchy_levy_vassal_swing_value }
}
ftr_diarchy_swing_at_least_amnesty_trigger = {
	liege = { diarchy_swing >= ftr_diarchy_amnesty_swing_value }
}

# 求值域：this = 待抽签的游说者（来自君主 / 共治者的 ftr_court_supporters 名单）。
# 口径：活着的成年可用者、未被囚、非无能——游说须亲自出面（无地廷臣同样算数）。
# 另设**个人冷却**：刚出过战的说客一年内不再被抽中（阵营内部轮换，防强手反复出战形成一边倒）。
# 用途：阶段 A 党羽游说对垒抽签（§4.4-A D13）。
ftr_diarchy_lobby_pk_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	NOT = { has_trait = incapable }
	NOT = { has_character_flag = ftr_diarchy_lobby_pk_lobbied }
}

### 四、无地共治者授土提醒（§4.0，求值域：共治者本人）

ftr_diarchy_landless_reminder_trigger = {
	is_diarch = yes
	is_alive = yes
	is_adult = yes
	is_landed = no
	# **已持有任何头衔者一律排除**：`is_landed = no` 判不出「无地头衔」——
	# 天朝制的宰相（首相府共治）持有无地的帝国级头衔，划地给他也会被制度收回，
	# 提醒与授土全部空转。故要求其名下**一个头衔都没有**。
	NOT = { any_held_title = { always = yes } }
	# 「忠心事主」后不再提醒
	NOT = { has_character_flag = ftr_diarchy_loyal_to_liege }
	# **太子一律不提醒**（2026-09）：太子须留守主君宫廷（§3.1），
	# 授土会把他变成 landed 封臣、进而解除共治——故太子不参与「索地 / 授土」这条线。
	NOT = { ftr_diarchy_is_crown_prince_trigger = yes }
}

# 求值域：主君本人。是否有可划出的地产（非首都伯爵领 或 公爵级头衔）
# 【必须与 ftr_diarchy_grant_land_effect 的筛选条件完全一致】
# 世族（贵族家族）头衔在天朝制为伯爵级、行政制为公爵级，且与「家主」身份绑定，
# 划出即等于主君静默失去家主之位，因此两处都必须以 is_noble_family_title = no 排除；
# 否则会出现「按钮可点、点了却什么也没给」的空转（或反向的静默转移）。
ftr_diarchy_liege_has_land_to_give_trigger = {
	OR = {
		any_held_title = {
			tier = tier_county
			NOT = { this = prev.capital_county } 
			is_noble_family_title = no
		}
		any_held_title = {
			tier = tier_duchy
			is_noble_family_title = no
		}
	}
}

### 五、立储候选（§2.2，求值域：主君本人）

# 是否「没有可继承的子女」——用于决定是否追加手足候选
ftr_diarchy_has_no_eligible_child_trigger = {
	NOT = {
		any_child = {
			is_alive = yes
			is_eunuch_trigger = no
			NOR = {
				has_trait = disinherited
				has_trait = devoted
				has_trait = order_member
				has_trait = bastard
			}
			house ?= prev.house
		}
	}
}

# 求值域：君主本人。是否已挂「继承人线共治者」（改储锁定锚点，§2.2 B3）
ftr_diarchy_liege_has_heir_line_diarch_trigger = {
	has_active_diarchy = yes
	OR = {
		has_diarchy_type = ftr_diarchy_crown_prince
		designated_heir ?= { is_diarch = yes }
		designated_diarch ?= { is_diarch = yes }
	}
}

### 六、阶段 A→B 解锁判定（§4.4，求值域：共治者本人）

ftr_diarchy_phase_a_no_opposition_trigger = {

### 七、战争实力对比（§4.5 拉人意愿；求值域 = war，参照 = scope:actor 我方）

# 我方处于劣势：敌我比 ≥ 2 → 需要拉人
ftr_diarchy_war_outnumbered_trigger = {
	ftr_diarchy_war_enemy_to_our_ratio_value >= ftr_diarchy_war_ratio_threshold_high_value
}

# 我方压倒性优势：敌我比 ≤ 0.5 → 不必拉人
ftr_diarchy_war_overwhelming_trigger = {
	ftr_diarchy_war_enemy_to_our_ratio_value <= ftr_diarchy_war_ratio_threshold_low_value
}

### 八、清君侧（§5，交互/决议求值域：作为宣战方的封臣本人）

# 求值域：作为战争目标的**主君本人**。是否为合法清君侧目标：
# 独立 + 主头衔 ≥ 王国 + 存在共治者。
# 注（裁定 Q3）：**不再要求行政制**——非行政制政权经 01 请愿取得 `regency` 后，
# 封臣同样需要「武力替换摄政」的入口，故沿用与太子准入相同的等级要求、去掉政体要求。
ftr_qingjunce_target_eligible_trigger = {
	is_independent_ruler = yes
	highest_held_title_tier >= tier_kingdom
	has_active_diarchy = yes
	exists = diarch
}

ftr_qingjunce_can_declare_trigger = {
	is_landed = yes
	is_capable_adult = yes
	NOT = { is_imprisoned = yes }
	is_independent_ruler = no
	exists = liege
	# 目标：主君当前共治者
	liege = {
		has_active_diarchy = yes
		exists = diarch
		NOT = { diarch ?= { this = prev } }
	}
	# 未与主君处于同一场清君侧战争
	liege = {
		NOT = { has_variable = ftr_qingjunce_current_war }
	}
}

### 九、废太子（§3.2，求值域：主君本人）

# 主君是否握有「正当理由」处罚太子。有则**废除太子不损正统性**（决议提示框内以 custom_tooltip 明示）。
# 依据：本 mod 的罪行类 opinion modifier（ftr_attempted_coup_crime / ftr_murderer_crime /
# ftr_corruption_crime 等）均带 imprisonment_reason / revoke_title_reason，
# 故原版 has_*_reason 触发器即覆盖「谋逆 / 谋杀 / 贪墨」等实罪；
# 另把「太子已被主君拿下（在押）」计为充足口实。
# 求值时机：废除动作**之前**（共治关系尚在），故可安全读 diarch。
ftr_diarchy_abolish_prince_just_cause_trigger = {
	exists = diarch
	OR = {
		has_imprisonment_reason = diarch
		has_revoke_title_reason = diarch
		diarch ?= { is_imprisoned = yes }
	}
}
